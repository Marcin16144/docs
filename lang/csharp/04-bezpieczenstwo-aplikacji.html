ASP.NET Identity

Identity to framework do zarzadzania kontami, hasłami, rolami, claimami w ASP.NET Core.

// Program.cs
builder.Services.AddIdentity<ApplicationUser, IdentityRole>(options =>
{
    // Password requirements
    options.Password.RequiredLength = 12;
    options.Password.RequireDigit = true;
    options.Password.RequireUppercase = true;
    options.Password.RequireNonAlphanumeric = true;

    // Lockout - rate limiting prob
    options.Lockout.MaxFailedAccessAttempts = 5;
    options.Lockout.DefaultLockoutTimeSpan = TimeSpan.FromMinutes(15);

    // Email confirmation
    options.SignIn.RequireConfirmedEmail = true;
})
.AddEntityFrameworkStores<AppDb>()
.AddDefaultTokenProviders();

Hashowanie hasel - PBKDF2

Identity uzywa PBKDF2 z 100,000+ iteracji. Mozna podmienic na Argon2 przez custom IPasswordHasher<TUser>.

Data Protection API

System do szyfrowania danych w aplikacji - cookies, tokeny, sensitive query strings.

// Konfiguracja - persystencja kluczy
builder.Services.AddDataProtection()
    .PersistKeysToFileSystem(new DirectoryInfo(@"/var/keys"))
    .ProtectKeysWithCertificate("thumbprint")
    .SetDefaultKeyLifetime(TimeSpan.FromDays(90));

// Uzycie - DI
public class TokenService(IDataProtectionProvider dpProvider)
{
    private readonly IDataProtector _protector =
        dpProvider.CreateProtector("MyApp.Tokens.v1");

    public string Protect(string plain) => _protector.Protect(plain);
    public string Unprotect(string cipher) => _protector.Unprotect(cipher);
}

// Time-limited tokens
var protector = dpProvider.CreateProtector("PasswordReset")
    .ToTimeLimitedDataProtector();
var token = protector.Protect(userId, TimeSpan.FromHours(1));

Anti-Forgery (CSRF)

// Program.cs
builder.Services.AddAntiforgery(options =>
{
    options.HeaderName = "X-XSRF-TOKEN";
    options.Cookie.SecurePolicy = CookieSecurePolicy.Always;
    options.Cookie.SameSite = SameSiteMode.Strict;
});

// Razor - automatyczny token w formularzu
<form asp-action="Submit" method="post">
    @* @Html.AntiForgeryToken() - automatyczne dla form tag helper *@
    <input name="title" />
    <button type="submit">OK</button>
</form>

// Controller - validacja
[HttpPost]
[ValidateAntiForgeryToken]
public IActionResult Submit(MyDto dto) { /* ... */ }

// W .NET 7+ - automatyczna walidacja dla MVC
builder.Services.AddControllersWithViews(options =>
{
    options.Filters.Add(new AutoValidateAntiforgeryTokenAttribute());
});

Secure cookies

builder.Services.ConfigureApplicationCookie(options =>
{
    options.Cookie.HttpOnly = true;            // niedostepne dla JS
    options.Cookie.SecurePolicy = CookieSecurePolicy.Always;  // tylko HTTPS
    options.Cookie.SameSite = SameSiteMode.Strict;
    options.ExpireTimeSpan = TimeSpan.FromHours(8);
    options.SlidingExpiration = true;
    options.LoginPath = "/Account/Login";
    options.AccessDeniedPath = "/Account/Forbidden";
});

JWT validation

builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuer = true,
            ValidIssuer = builder.Configuration["Jwt:Issuer"],
            ValidateAudience = true,
            ValidAudience = builder.Configuration["Jwt:Audience"],
            ValidateLifetime = true,
            ClockSkew = TimeSpan.FromMinutes(2),
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = new SymmetricSecurityKey(
                Encoding.UTF8.GetBytes(builder.Configuration["Jwt:Key"]!)),
            // Asymetryczne tez mozliwe (RsaSecurityKey)
        };

        options.MapInboundClaims = false;  // standardowe nazwy claims
    });

builder.Services.AddAuthorization();

app.UseAuthentication();
app.UseAuthorization();

Authorization policies

// Definicja polityk
builder.Services.AddAuthorization(options =>
{
    options.AddPolicy("AdminOnly", p => p.RequireRole("Admin"));

    options.AddPolicy("AtLeast21", p => p.RequireAssertion(ctx =>
    {
        var birth = ctx.User.FindFirst("DateOfBirth")?.Value;
        return DateTime.TryParse(birth, out var dob)
            && (DateTime.Today - dob).TotalDays / 365 >= 21;
    }));

    options.AddPolicy("CanEditPost", p => p
        .RequireAuthenticatedUser()
        .AddRequirements(new SameAuthorRequirement()));
});

// Custom requirement + handler
public class SameAuthorRequirement : IAuthorizationRequirement { }

public class SameAuthorHandler(IPostRepo repo)
    : AuthorizationHandler<SameAuthorRequirement, Post>
{
    protected override async Task HandleRequirementAsync(
        AuthorizationHandlerContext ctx,
        SameAuthorRequirement req,
        Post post)
    {
        if (ctx.User.FindFirst("sub")?.Value == post.AuthorId)
            ctx.Succeed(req);
    }
}

// Uzycie
[Authorize(Policy = "AdminOnly")]
public IActionResult Admin() => Ok();

Rate limiting (.NET 7+)

builder.Services.AddRateLimiter(options =>
{
    options.AddFixedWindowLimiter("auth", o =>
    {
        o.PermitLimit = 5;
        o.Window = TimeSpan.FromMinutes(1);
        o.QueueProcessingOrder = QueueProcessingOrder.OldestFirst;
        o.QueueLimit = 0;
    });

    options.RejectionStatusCode = 429;
});

app.UseRateLimiter();

app.MapPost("/api/login", LoginHandler)
   .RequireRateLimiting("auth");

SQL Injection - parametryzacja

// ZLE - SQL Injection
var name = Request.Query["name"];
var sql = $"SELECT * FROM Users WHERE Name = '{name}'";

// DOBRZE - EF Core
var users = await db.Users
    .Where(u => u.Name == name)
    .ToListAsync();

// DOBRZE - FromSqlInterpolated
var users = await db.Users
    .FromSqlInterpolated($"SELECT * FROM Users WHERE Name = {name}")
    .ToListAsync();

// DOBRZE - Dapper z parametrami
var users = await connection.QueryAsync<User>(
    "SELECT * FROM Users WHERE Name = @Name",
    new { Name = name });

Secure-by-default checklist

  • HTTPS only (UseHttpsRedirection + HSTS)
  • CSP headers (Content-Security-Policy)
  • Anti-Forgery tokens dla form-based
  • Rate limiting na endpointach auth
  • Brak BinaryFormatter (deprecated)
  • Secrets w User Secrets / Key Vault
  • dotnet list package --vulnerable w CI
  • Signed NuGet packages
