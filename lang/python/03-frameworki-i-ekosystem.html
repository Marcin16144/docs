FastAPI - nowoczesny standard 2026

FastAPI laczy: szybkosc Starlette (ASGI), walidacja przez Pydantic v2, automatyczna dokumentacja OpenAPI/Swagger, wsparcie dla async/await.

# Instalacja
uv add 'fastapi[standard]'

# main.py
from fastapi import FastAPI, HTTPException, Depends
from pydantic import BaseModel, Field
from datetime import datetime

app = FastAPI(title="My API", version="1.0")

class UserCreate(BaseModel):
    name: str = Field(min_length=2, max_length=50)
    email: str
    age: int = Field(ge=0, le=150)

class User(UserCreate):
    id: int
    created_at: datetime

# In-memory store dla przykladu
users: dict[int, User] = {}

@app.get("/users/{user_id}")
async def get_user(user_id: int) -> User:
    if user_id not in users:
        raise HTTPException(404, "User not found")
    return users[user_id]

@app.post("/users", status_code=201)
async def create_user(data: UserCreate) -> User:
    user_id = len(users) + 1
    user = User(
        id=user_id,
        created_at=datetime.utcnow(),
        **data.model_dump()
    )
    users[user_id] = user
    return user

# Uruchomienie
# fastapi dev main.py
# fastapi run main.py

Django 5

Django to "batteries included" framework - ORM, admin, auth, templates, sesje, forms - wszystko w jednym. Idealne dla pelnych aplikacji webowych.

# Instalacja i nowy projekt
uv add django
uv run django-admin startproject mysite
cd mysite
uv run python manage.py startapp blog

# models.py
from django.db import models

class Post(models.Model):
    title = models.CharField(max_length=200)
    content = models.TextField()
    created_at = models.DateTimeField(auto_now_add=True)
    author = models.ForeignKey('auth.User', on_delete=models.CASCADE)

    def __str__(self):
        return self.title

# views.py - class-based
from django.views.generic import ListView, DetailView

class PostList(ListView):
    model = Post
    template_name = 'blog/list.html'
    paginate_by = 10

class PostDetail(DetailView):
    model = Post

# Migracje
# python manage.py makemigrations
# python manage.py migrate
# python manage.py runserver

Litestar - FastAPI's konkurent

Litestar jest bardzo podobny do FastAPI, ale szybszy (~30%), z lepsza struktura controllerow.

from litestar import Litestar, get, post
from msgspec import Struct

class User(Struct):
    id: int
    name: str

@get("/users/{user_id:int}")
async def get_user(user_id: int) -> User:
    return User(id=user_id, name="Anna")

@post("/users")
async def create_user(data: User) -> User:
    return data

app = Litestar([get_user, create_user])

Pydantic v2 - walidacja danych

Pydantic v2 jest ~10x szybszy niz v1 (rdzen napisany w Rust). Bazowy budulec FastAPI, ale tez uzywany standalone.

from pydantic import BaseModel, Field, EmailStr, field_validator
from datetime import datetime

class User(BaseModel):
    id: int
    email: EmailStr
    name: str = Field(min_length=2, max_length=100)
    tags: list[str] = []
    created_at: datetime = Field(default_factory=datetime.utcnow)

    @field_validator('name')
    @classmethod
    def name_no_digits(cls, v: str) -> str:
        if any(c.isdigit() for c in v):
            raise ValueError("Imie nie moze zawierac cyfr")
        return v.title()

    model_config = {
        "str_strip_whitespace": True,
        "validate_assignment": True,
    }

# Uzycie
u = User(id=1, email="anna@example.com", name="anna kowalska")
print(u.name)             # 'Anna Kowalska'
print(u.model_dump())     # dict
print(u.model_dump_json()) # json string

# Walidacja w runtime
try:
    User(id="bad", email="invalid", name="Anna 123")
except ValidationError as e:
    print(e.errors())

SQLAlchemy 2 - typed ORM

from sqlalchemy import create_engine, ForeignKey, select
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship, Session

class Base(DeclarativeBase):
    pass

class User(Base):
    __tablename__ = "users"

    id: Mapped[int] = mapped_column(primary_key=True)
    email: Mapped[str] = mapped_column(unique=True, index=True)
    name: Mapped[str]
    posts: Mapped[list["Post"]] = relationship(back_populates="author")

class Post(Base):
    __tablename__ = "posts"

    id: Mapped[int] = mapped_column(primary_key=True)
    title: Mapped[str]
    content: Mapped[str | None]
    author_id: Mapped[int] = mapped_column(ForeignKey("users.id"))
    author: Mapped["User"] = relationship(back_populates="posts")

# Engine + Session
engine = create_engine("postgresql://user:pass@localhost/mydb")
Base.metadata.create_all(engine)

with Session(engine) as session:
    user = User(email="a@b.pl", name="Anna")
    session.add(user)
    session.commit()

    # Typed query (2.0 style)
    stmt = select(User).where(User.email == "a@b.pl")
    user = session.scalars(stmt).one()
    print(user.name)

hypothesis - property-based testing

from hypothesis import given, strategies as st

@given(st.integers(), st.integers())
def test_dodawanie_przemiennne(a: int, b: int):
    assert a + b == b + a

@given(st.lists(st.integers()))
def test_sortowanie(lst):
    assert sorted(sorted(lst)) == sorted(lst)

@given(st.text())
def test_round_trip(s: str):
    assert decode(encode(s)) == s

# Hypothesis automatycznie generuje setki random inputs
# i znajduje minimalne contrprzyklady gdy testy faliuja
