1. App Router vs Pages Router

Next.js 13 wprowadzil App Router — nowy system routingu oparty na React Server Components. Pages Router (stary system) nadal dziala i jest wspierany, ale App Router jest zalecanym podejsciem dla nowych projektow.

2. Server Components vs Client Components

W App Router kazdy komponent jest domyslnie Server Component. Server Components sa renderowane na serwerze i nie wymagaja JavaScriptu po stronie klienta. Dyrektywa 'use client' oznacza granice klient-serwer.

// app/users/page.tsx — Server Component (domyslnie)
// Moze byc async! Moze uzywac await bezposrednio!
// Nie trafia do bundle po stronie klienta

import { db } from '@/lib/db';
import { UserCard } from './UserCard';  // moze byc client component

export default async function UsersPage() {
  // Bezposredni dostep do bazy danych — dziala tylko na serwerze
  const users = await db.user.findMany({
    orderBy: { createdAt: 'desc' },
    take: 20
  });

  return (
    <main>
      <h1>Uzytkownicy ({users.length})</h1>
      <div className="grid">
        {users.map(user => (
          <UserCard key={user.id} user={user} />
        ))}
      </div>
    </main>
  );
}

// app/users/UserCard.tsx — Client Component
// Uzywaj 'use client' tylko gdy potrzebujesz:
// - useState, useEffect, useReducer
// - onClick, onChange i inne event handlery
// - Dostep do window, document, localStorage
// - Biblioteki zewnetrzne bez wsparcia RSC

'use client';

import { useState } from 'react';

interface User { id: string; name: string; email: string; }

export function UserCard({ user }: { user: User }) {
  const [expanded, setExpanded] = useState(false);

  return (
    <div className="card" onClick={() => setExpanded(!expanded)}>
      <h3>{user.name}</h3>
      {expanded && <p>{user.email}</p>}
    </div>
  );
}

3. Server Actions

Server Actions to funkcje wykonywane na serwerze, wywoływane bezposrednio z komponentow klienckich lub formularzy. Zastepuja tradycyjne API routes dla mutacji danych.

// app/actions/users.ts — Server Actions
'use server';

import { revalidatePath, revalidateTag } from 'next/cache';
import { redirect } from 'next/navigation';
import { db } from '@/lib/db';
import { z } from 'zod';

const CreateUserSchema = z.object({
  name: z.string().min(2),
  email: z.string().email()
});

export async function createUser(formData: FormData) {
  // Walidacja
  const parsed = CreateUserSchema.safeParse({
    name: formData.get('name'),
    email: formData.get('email')
  });

  if (!parsed.success) {
    return { error: parsed.error.flatten().fieldErrors };
  }

  // Zapis do bazy
  await db.user.create({ data: parsed.data });

  // Inwalidacja cache — strona /users zostanie przeladowana
  revalidatePath('/users');

  // Przekierowanie
  redirect('/users');
}

// Uzycie w Server Component (formularz z progressywnym usprawnieniem)
import { createUser } from '../actions/users';

export default function NewUserPage() {
  return (
    <form action={createUser}>
      <input name="name" placeholder="Imie" required />
      <input name="email" type="email" placeholder="Email" required />
      <button type="submit">Dodaj uzytkownika</button>
    </form>
  );
}

// Uzycie w Client Component (z useActionState)
'use client';
import { useActionState } from 'react';
import { createUser } from '../actions/users';

export function NewUserForm() {
  const [state, action, isPending] = useActionState(createUser, null);
  return (
    <form action={action}>
      <input name="name" />
      <input name="email" />
      <button disabled={isPending}>
        {isPending ? 'Zapisywanie...' : 'Zapisz'}
      </button>
      {state?.error && <p>Blad: {JSON.stringify(state.error)}</p>}
    </form>
  );
}

4. Strategie renderowania

SSG (Static)Build timeStrony bez zmiennych danychCDN na zawsze
SSR (Dynamic)Kazde zadanieDane personalizowane, real-timeBrak lub krotki TTL
ISR (Incremental)Build + rewalidacjaBlog, e-commerce z rzadkimi zmianamiCDN + rewalidacja co N sekund
PPR (Partial Prerendering)Build (shell) + Runtime (dynamic parts)Strony z mieszana tresciaCDN dla shella
// Kontrola cache przez fetch options
async function getData() {
  // SSG: cache na zawsze (domyslne w Next.js 14, NIE w 15)
  const res1 = await fetch('https://api.example.com/static', {
    cache: 'force-cache'
  });

  // SSR: bez cache, zawsze swiezy
  const res2 = await fetch('https://api.example.com/live', {
    cache: 'no-store'
  });

  // ISR: rewalidacja co 60 sekund
  const res3 = await fetch('https://api.example.com/posts', {
    next: { revalidate: 60 }
  });

  // ISR z tagami (rewalidacja on-demand)
  const res4 = await fetch('https://api.example.com/products', {
    next: { tags: ['products'] }
  });
}

// Rewalidacja taga z Server Action lub API route
import { revalidateTag } from 'next/cache';
revalidateTag('products');  // inwaliduje wszystkie fetche z tym tagiem

// Wymuszenie dynamicznosci calej strony
export const dynamic = 'force-dynamic';
export const revalidate = 0;

6. Data fetching z cache i revalidate

W Next.js 15 domyslne zachowanie cache dla fetch zmienilo sie — teraz domyslnie jest no-store (brak cache). Trzeba jawnie podac cache: 'force-cache' lub next: { revalidate } dla ISR.

7. Middleware

// middleware.ts — uruchamiany na Edge przed kazdym requestem
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  const token = request.cookies.get('auth-token')?.value;

  // Przekierowanie niezalogowanych uzytkownikow
  if (!token && request.nextUrl.pathname.startsWith('/dashboard')) {
    return NextResponse.redirect(new URL('/login', request.url));
  }

  // Dodanie custom headera
  const response = NextResponse.next();
  response.headers.set('X-Custom-Header', 'wartość');
  return response;
}

// Konfiguracja — ktore trasy obejmuje middleware
export const config = {
  matcher: [
    '/dashboard/:path*',
    '/admin/:path*',
    '/((?!api|_next/static|_next/image|favicon.ico).*)'
  ]
};

9. API Routes (route.ts)

// app/api/users/route.ts — GET i POST
import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';

export async function GET(request: NextRequest) {
  const { searchParams } = request.nextUrl;
  const page = Number(searchParams.get('page') ?? '1');

  const users = await db.user.findMany({
    skip: (page - 1) * 20,
    take: 20
  });

  return NextResponse.json({ users, page });
}

export async function POST(request: NextRequest) {
  const body = await request.json();
  const user = await db.user.create({ data: body });
  return NextResponse.json(user, { status: 201 });
}

// app/api/users/[id]/route.ts — dynamiczny parametr
export async function GET(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;  // params sa teraz Promise w Next.js 15
  const user = await db.user.findUnique({ where: { id } });
  if (!user) return NextResponse.json({ error: 'Nie znaleziono' }, { status: 404 });
  return NextResponse.json(user);
}
