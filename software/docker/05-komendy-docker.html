Komendy Docker

Pelna referencja CLI z przykladami

docker run — uruchomienie kontenera

docker run [OPTIONS] IMAGE [COMMAND] [ARGS...]

# Podstawowe
docker run hello-world
docker run -it ubuntu bash                  # interaktywnie z TTY
docker run -d nginx                         # detached (w tle)
docker run --rm alpine echo hi              # auto usun po zakonczeniu
docker run --name web -d nginx              # nazwa kontenera

# Porty
docker run -p 8080:80 nginx                 # host:container
docker run -p 127.0.0.1:8080:80 nginx       # bind tylko na localhost
docker run -P nginx                         # losowe porty na hoscie

# Wolumeny
docker run -v /host/data:/data nginx        # bind mount
docker run -v mydata:/data nginx            # named volume
docker run -v $(pwd):/app -w /app node:20 npm test

# Zmienne
docker run -e NODE_ENV=production myapp
docker run --env-file .env myapp

# Zasoby
docker run --cpus 1.5 --memory 512m myapp

# Restart policy
docker run --restart unless-stopped -d nginx
# zawsze | unless-stopped | on-failure | no

# Kompletny przyklad
docker run -d \
  --name api \
  --restart unless-stopped \
  -p 8080:80 \
  -v $(pwd)/config:/etc/api:ro \
  -v api-data:/var/api \
  -e DATABASE_URL=$DB_URL \
  --memory 512m --cpus 1 \
  --health-cmd "curl -f http://localhost/health" \
  --health-interval 30s \
  ghcr.io/user/api:1.2.0

docker ps — lista kontenerow

docker ps                         # uruchomione
docker ps -a                      # wszystkie (rowniez stopped)
docker ps -q                      # tylko ID
docker ps --size                  # z rozmiarem warstw
docker ps --filter "status=exited"
docker ps --filter "name=web"
docker ps --filter "label=env=prod"
docker ps --format "table {{.ID}}\t{{.Names}}\t{{.Status}}"

docker exec — wykonaj w dzialajacym kontenerze

docker exec -it web bash                # shell
docker exec -it web sh                  # gdy bash niedostepny (alpine)
docker exec web ls /app                 # jednorazowa komenda
docker exec -u root web apt update      # jako root
docker exec -e DEBUG=1 web /script.sh   # z env var

docker logs

docker logs web                     # cale logi
docker logs -f web                  # follow (jak tail -f)
docker logs --tail 100 web          # ostatnie 100 linii
docker logs --since 10m web         # od 10 minut wstecz
docker logs --since 2024-01-01 web
docker logs --until 1h web          # az do godziny temu
docker logs -t web                  # z timestampami

docker build

docker build -t myapp:1.0 .                       # tag
docker build -t myapp -f Dockerfile.prod .         # niestandardowy Dockerfile
docker build --no-cache -t myapp .                 # ignoruj cache
docker build --build-arg VERSION=1.0 -t myapp .    # zmienne ARG
docker build --target builder -t myapp:build .     # multi-stage — tylko stage builder
docker build --platform linux/amd64,linux/arm64 . # multi-arch (z buildx)
docker build --secret id=npm,src=$HOME/.npmrc .    # secret w build (BuildKit)

# BuildKit (default w nowoczesnym Dockerze, lepszy)
DOCKER_BUILDKIT=1 docker build .

# Przyklad z buildx (multi-arch + push)
docker buildx create --use
docker buildx build \
  --platform linux/amd64,linux/arm64 \
  --tag user/myapp:1.0 \
  --push .

docker pull / push / tag

docker pull nginx                       # latest
docker pull nginx:1.27-alpine
docker pull --platform linux/arm64 nginx

docker tag myapp:latest registry.example.com/myapp:1.0
docker push registry.example.com/myapp:1.0

# Login
docker login                            # Docker Hub
docker login ghcr.io                    # GitHub Container Registry
docker login -u user -p $TOKEN registry.example.com

docker images

docker images                       # lista obrazow
docker images -a                    # rowniez intermediate
docker images --filter dangling=true
docker images --format "{{.Repository}}:{{.Tag}} {{.Size}}"

docker rmi nginx:latest             # usun obraz
docker rmi $(docker images -q)      # usun wszystkie

docker image prune                  # usun dangling
docker image prune -a               # usun wszystkie nieuzywane
docker image prune -a --filter "until=24h"

docker stop / start / restart / kill / rm

docker stop web                     # graceful (SIGTERM, potem SIGKILL po 10s)
docker stop -t 30 web               # czas na cleanup 30s
docker start web
docker restart web
docker kill web                     # SIGKILL od razu
docker kill -s SIGUSR1 web          # custom signal

docker rm web                       # usun stopped
docker rm -f web                    # force (kill + rm)
docker rm $(docker ps -aq)          # usun wszystkie

docker inspect

docker inspect web                              # pelny JSON
docker inspect --format '{{.State.Status}}' web
docker inspect --format '{{.NetworkSettings.IPAddress}}' web
docker inspect --format '{{json .Mounts}}' web | jq

docker stats / top

docker stats                        # live CPU/RAM/IO wszystkich
docker stats web                    # tylko jeden
docker stats --no-stream            # snapshot raz
docker stats --format "table {{.Name}}\t{{.CPUPerc}}\t{{.MemUsage}}"

docker top web                      # procesy w kontenerze (jak ps)

docker network

docker network ls
docker network inspect bridge

# Tworzenie sieci uzytkownika — auto DNS miedzy kontenerami
docker network create mynet
docker network create --driver overlay --attachable mynet
docker network create --subnet 172.30.0.0/16 mynet

# Polaczenie kontenera
docker run --network mynet --name web nginx
docker network connect mynet existing-container
docker network disconnect mynet web

docker network rm mynet
docker network prune                # usun nieuzywane

docker volume

docker volume create mydata
docker volume create --driver local --opt type=nfs --opt o=addr=10.0.0.1 nfs-vol
docker volume ls
docker volume inspect mydata
docker volume rm mydata
docker volume prune                 # usun nieuzywane

# Backup volume
docker run --rm -v mydata:/data -v $(pwd):/backup alpine \
  tar czf /backup/mydata.tar.gz -C /data .

# Restore
docker run --rm -v mydata:/data -v $(pwd):/backup alpine \
  tar xzf /backup/mydata.tar.gz -C /data

docker system — konserwacja

docker system df              # ile miejsca zajmuje Docker
docker system info            # info o systemie

docker system prune           # cleanup: stopped containers, dangling images, networks, build cache
docker system prune -a        # rowniez nieuzywane obrazy
docker system prune -a --volumes  # rowniez wolumeny (UWAGA — usuwa dane!)
docker system prune -a --filter "until=72h"
