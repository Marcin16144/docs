Docker Compose

Multi-container apps z compose.yaml — dev i prod

Co to jest Compose?

Docker Compose to tool do orkiestracji wielu kontenerow. Definiujesz aplikacje (services, networks, volumes) w jednym pliku YAML, uruchamiasz jednym poleceniem.

W 2026 roku Compose v2 (jako plugin docker compose, nie standalone docker-compose) jest standardem. Plik nazywa sie compose.yaml (bez prefiksu "docker-").

Pierwszy compose.yaml

# compose.yaml
services:
  web:
    image: nginx:alpine
    ports:
      - "8080:80"
    volumes:
      - ./html:/usr/share/nginx/html:ro

  api:
    build: ./api
    environment:
      DATABASE_URL: postgres://user:pass@db:5432/myapp
    depends_on:
      db:
        condition: service_healthy
    ports:
      - "3000:3000"

  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_USER: user
      POSTGRES_PASSWORD: pass
      POSTGRES_DB: myapp
    volumes:
      - db-data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U user"]
      interval: 5s
      timeout: 3s
      retries: 5

volumes:
  db-data:
# Uruchomienie
docker compose up -d           # detached
docker compose logs -f         # follow logi
docker compose ps              # status
docker compose down            # stop + rm
docker compose down -v         # rowniez wolumeny
docker compose restart api

Anatomia compose.yaml

services — kontenery

services:
  myservice:
    image: nginx:1.27-alpine          # gotowy obraz
    # LUB
    build:
      context: ./api
      dockerfile: Dockerfile.prod
      args:
        VERSION: 1.0
      target: runtime                  # multi-stage target

    container_name: api-server         # opcjonalnie
    hostname: api

    ports:
      - "3000:3000"                    # host:container
      - "127.0.0.1:5432:5432"          # bind localhost
      - "8080-8090:80"                 # zakres

    volumes:
      - ./src:/app/src                 # bind mount
      - mydata:/var/data               # named volume
      - /var/run/docker.sock:/var/run/docker.sock:ro

    environment:
      NODE_ENV: production
      DB_HOST: db
    env_file:
      - .env
      - .env.${ENV:-development}

    networks:
      - backend
      - frontend

    depends_on:
      db:
        condition: service_healthy
      redis:
        condition: service_started

    restart: unless-stopped
    # no | always | on-failure | unless-stopped

    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3000/health"]
      interval: 30s
      timeout: 3s
      retries: 3
      start_period: 10s

    deploy:
      resources:
        limits:
          cpus: '1.5'
          memory: 512M
        reservations:
          memory: 256M

    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"

networks

networks:
  frontend:
    driver: bridge
  backend:
    driver: bridge
    internal: true                 # brak dostepu do internetu
  external-net:
    external: true
    name: shared-network

volumes

volumes:
  db-data:                         # named volume
  cache:
    driver: local
    driver_opts:
      type: tmpfs
      device: tmpfs
  shared:
    external: true
    name: my-shared-volume         # juz istnieje

depends_on — z healthcheck

depends_on bez warunku tylko ustala kolejnosc startu, nie czeka az service jest gotowy. Z condition czeka az healthcheck przejdzie.

services:
  api:
    depends_on:
      db:
        condition: service_healthy        # czeka na zdrowie
      migrations:
        condition: service_completed_successfully  # czeka az skonczy sie

  db:
    image: postgres:16
    healthcheck:
      test: ["CMD-SHELL", "pg_isready"]
      interval: 5s

  migrations:
    build: ./api
    command: npm run migrate
    depends_on:
      db:
        condition: service_healthy

Watch mode (Compose 2.22+)

Hot reload na poziomie compose — kopiuje pliki do kontenera lub rebuildujje przy zmianach. Zastapuje czesto rezczne bind mounts.

services:
  api:
    build: ./api
    develop:
      watch:
        - action: sync                  # kopiuj pliki do kontenera
          path: ./api/src
          target: /app/src
          ignore:
            - "**/*.test.ts"
        - action: rebuild               # rebuild gdy package.json sie zmieni
          path: ./api/package.json

# Uruchomienie
docker compose up --watch

Profiles — rozne srodowiska

services:
  api:
    image: api:latest
    # zawsze startuje

  db:
    image: postgres:16
    profiles: ["dev", "test"]

  pgadmin:
    image: dpage/pgadmin4
    profiles: ["dev"]                   # tylko dla dev

  loadtest:
    image: grafana/k6
    profiles: ["test"]                   # tylko dla testow

# Uruchom konkretne profile
docker compose --profile dev up
docker compose --profile test up
docker compose up                        # tylko bez profilu (api)

Override files

Compose automatycznie laduje compose.override.yaml (jesli istnieje) na compose.yaml. Pozwala mieci osobny dev/prod.

# compose.yaml — baza
services:
  api:
    image: ghcr.io/user/api:1.0
    restart: unless-stopped

# compose.override.yaml — dev (auto-merge)
services:
  api:
    build: ./api                        # build zamiast image w dev
    volumes:
      - ./api/src:/app/src
    environment:
      NODE_ENV: development
      DEBUG: "*"

# compose.prod.yaml — prod (manualny)
services:
  api:
    deploy:
      replicas: 3
      resources:
        limits:
          memory: 1G

# Uruchomienia:
docker compose up                                    # baza + override (dev)
docker compose -f compose.yaml -f compose.prod.yaml up    # baza + prod

env_file — zmienne

# .env (auto-loaded przez compose)
ENV=production
DB_HOST=db
DB_PORT=5432

# compose.yaml
services:
  api:
    image: api:${TAG:-latest}             # ${TAG} z .env, default latest
    environment:
      NODE_ENV: ${ENV}
      DB_HOST: ${DB_HOST}
    env_file:
      - .env.${ENV}                       # np. .env.production
      - path: ./secrets.env
        required: false                   # nie blad gdy brak (Compose 2.24+)

Pelny przyklad — full stack

# compose.yaml
services:
  caddy:
    image: caddy:2-alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./Caddyfile:/etc/caddy/Caddyfile:ro
      - caddy-data:/data
      - caddy-config:/config
    depends_on:
      - frontend
      - api

  frontend:
    build: ./frontend
    expose: ["3000"]
    environment:
      NEXT_PUBLIC_API_URL: https://api.example.com
    restart: unless-stopped

  api:
    build: ./api
    expose: ["3000"]
    environment:
      DATABASE_URL: postgres://user:pass@db:5432/app
      REDIS_URL: redis://redis:6379
    depends_on:
      db:
        condition: service_healthy
      redis:
        condition: service_started
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "wget", "-qO-", "http://localhost:3000/health"]
      interval: 30s

  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_USER: user
      POSTGRES_PASSWORD_FILE: /run/secrets/db_password
      POSTGRES_DB: app
    secrets: [db_password]
    volumes:
      - db-data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U user"]
      interval: 5s
    restart: unless-stopped

  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes
    volumes:
      - redis-data:/data
    restart: unless-stopped

volumes:
  db-data:
  redis-data:
  caddy-data:
  caddy-config:

secrets:
  db_password:
    file: ./secrets/db_password.txt

KomendaCel

docker compose up --watchWatch mode (hot reload)

docker compose logs -f apiLogi konkretnej service
docker compose exec api bashShell w running service
docker compose run --rm api npm testJednorazowe (np. test, migracje)

docker compose topProcesy w services

Zlote zasady

  • Uzywaj compose.yaml (nie docker-compose.yml)
  • Pomijaj version: (Compose Spec, nieprzestarzaly)
  • Uzywaj healthcheck + depends_on condition
  • Sekrety przez secrets: nie environment
  • Override files dla dev vs prod
  • Watch mode zamiast bind mount + nodemon dla nowoczesnego DX
  • Profiles dla optionalnych services
