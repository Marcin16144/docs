Minimal APIs (.NET 6+)

Minimalna ilosc kodu do uruchomienia API. Idealny dla mikroserwisow.

# Tworzenie
dotnet new webapi -minimal -o MyApi
cd MyApi
dotnet run
// Program.cs
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddDbContext<AppDb>(opt =>
    opt.UseSqlite("Data Source=app.db"));

var app = builder.Build();

// GET endpoint
app.MapGet("/users/{id:int}", async (int id, AppDb db) =>
    await db.Users.FindAsync(id) is User u
        ? Results.Ok(u)
        : Results.NotFound());

// POST endpoint
app.MapPost("/users", async (User u, AppDb db) =>
{
    db.Users.Add(u);
    await db.SaveChangesAsync();
    return Results.Created($"/users/{u.Id}", u);
});

app.Run();

Klasyczne kontrolery (MVC / Web API)

[ApiController]
[Route("api/[controller]")]
public class UsersController(IUserService service) : ControllerBase
{
    [HttpGet("{id:int}")]
    public async Task<ActionResult<User>> GetById(int id, CancellationToken ct)
    {
        var user = await service.FindAsync(id, ct);
        return user is null ? NotFound() : Ok(user);
    }

    [HttpPost]
    [Authorize(Policy = "AdminOnly")]
    public async Task<IActionResult> Create(CreateUserDto dto)
    {
        var user = await service.CreateAsync(dto);
        return CreatedAtAction(nameof(GetById), new { id = user.Id }, user);
    }
}

Entity Framework Core

EF Core to ORM dla .NET. Code-first, migracje, LINQ to SQL, change tracking.

// Model
public class User
{
    public int Id { get; set; }
    public string Email { get; set; } = "";
    public List<Order> Orders { get; set; } = [];
}

public class Order
{
    public int Id { get; set; }
    public int UserId { get; set; }
    public User User { get; set; } = null!;
    public decimal Total { get; set; }
}

// DbContext
public class AppDb(DbContextOptions<AppDb> opt) : DbContext(opt)
{
    public DbSet<User> Users => Set<User>();
    public DbSet<Order> Orders => Set<Order>();
}

// Konfiguracja w Program.cs
builder.Services.AddDbContext<AppDb>(opt =>
    opt.UseNpgsql(builder.Configuration.GetConnectionString("DefaultConnection")));

// Migracje
// dotnet ef migrations add InitialCreate
// dotnet ef database update

// Zapytania - LINQ
var users = await db.Users
    .Where(u => u.Email.EndsWith("@example.com"))
    .Include(u => u.Orders)
    .OrderByDescending(u => u.Orders.Count)
    .Take(10)
    .AsNoTracking()
    .ToListAsync();

// Compiled queries (wydajnosc)
private static readonly Func<AppDb, int, Task<User?>> GetUserById =
    EF.CompileAsyncQuery((AppDb db, int id) =>
        db.Users.FirstOrDefault(u => u.Id == id));

gRPC w .NET

gRPC to high-performance RPC framework od Google. .NET ma pierwszorzedne wsparcie.

// proto/greet.proto
syntax = "proto3";
service Greeter {
  rpc SayHello (HelloRequest) returns (HelloReply);
}
message HelloRequest { string name = 1; }
message HelloReply { string message = 1; }
// Server
public class GreeterService : Greeter.GreeterBase
{
    public override Task<HelloReply> SayHello(
        HelloRequest request, ServerCallContext context)
    {
        return Task.FromResult(new HelloReply
        {
            Message = $"Hello {request.Name}"
        });
    }
}

// Client
using var channel = GrpcChannel.ForAddress("https://localhost:5001");
var client = new Greeter.GreeterClient(channel);
var reply = await client.SayHelloAsync(new HelloRequest { Name = "Anna" });

SignalR - real-time communication

// Hub
public class ChatHub : Hub
{
    public async Task SendMessage(string user, string message)
    {
        await Clients.All.SendAsync("ReceiveMessage", user, message);
    }
}

// Konfiguracja
builder.Services.AddSignalR();
app.MapHub<ChatHub>("/chat");

// JavaScript client
const conn = new signalR.HubConnectionBuilder()
    .withUrl("/chat")
    .build();
conn.on("ReceiveMessage", (user, msg) => console.log(user, msg));
await conn.start();
await conn.invoke("SendMessage", "Anna", "Hello!");

MSBuild i Directory.Packages.props

Centralne zarzadzanie wersjami paczek (Central Package Management - .NET 7+):

<!-- Directory.Packages.props - w katalogu solution -->
<Project>
  <PropertyGroup>
    <ManagePackageVersionsCentrally>true</ManagePackageVersionsCentrally>
  </PropertyGroup>
  <ItemGroup>
    <PackageVersion Include="Microsoft.EntityFrameworkCore" Version="9.0.0" />
    <PackageVersion Include="Serilog" Version="4.0.0" />
  </ItemGroup>
</Project>

<!-- W projekcie .csproj - bez Version -->
<PackageReference Include="Microsoft.EntityFrameworkCore" />
