Krotka historia C++

C++ powstal w 1985 roku jako rozszerzenie C o klasy, dzielo Bjarne Stroustrupa (Bell Labs). Pierwsza nazwa — "C with Classes" (1979). Standaryzacja przez ISO od 1998.

Typy podstawowe

// Typy calkowite (rozmiar zalezny od platformy!)
short s = 1;            // co najmniej 16 bit
int i = 1;              // co najmniej 16 bit (zwykle 32)
long l = 1L;            // co najmniej 32 bit
long long ll = 1LL;     // co najmniej 64 bit

// Typy o stalym rozmiarze - <cstdint> (preferowane!)
#include <cstdint>
int8_t  i8 = 127;
int16_t i16 = 32767;
int32_t i32 = 2'147'483'647;
int64_t i64 = 9'223'372'036'854'775'807;
uint8_t u8 = 255;       // unsigned

// Zmiennoprzecinkowe
float f = 3.14f;        // 32 bit (single precision)
double d = 3.14;        // 64 bit (double precision)
long double ld = 3.14L; // 80/128 bit

// Inne
bool b = true;          // 1 byte
char c = 'A';           // 1 byte
char8_t c8 = u8'A';     // C++20 - UTF-8
char16_t c16 = u'A';    // UTF-16
char32_t c32 = U'A';    // UTF-32

// auto - dedukcja typu (C++11+)
auto x = 42;            // int
auto y = 3.14;          // double
auto z = "tekst";       // const char*
auto v = std::vector<int>{1, 2, 3};

// if / else
if (x > 0) {
    std::cout << "dodatnia";

    std::cout << "ujemna";

    std::cout << "zero";

// if z init (C++17)

    std::cout << it->second;

    case 200:
    case 201:
        std::cout << "OK\n";
        break;
    case 404:
        std::cout << "Not Found\n";
        break;
    default:
        std::cout << "Other\n";

std::vector<int> nums = {1, 2, 3, 4, 5};
for (int n : nums) {
    std::cout << n << ' ';
}

// Z structured bindings (C++17)
std::map<std::string, int> m = {{"a", 1}, {"b", 2}};
for (const auto& [klucz, wartosc] : m) {
    std::cout << klucz << '=' << wartosc;

Klasy i OOP

#include <string>
#include <iostream>

class Osoba {
public:
    // Constructor
    Osoba(std::string imie, int wiek)
        : imie_(std::move(imie)), wiek_(wiek) {}

    // Destruktor
    ~Osoba() = default;

    // Metoda
    void przedstaw() const {
        std::cout << "Jestem " << imie_
                  << ", lat " << wiek_ << '\n';
    }

    // Geter / seter
    const std::string& imie() const { return imie_; }
    int wiek() const { return wiek_; }

private:
    std::string imie_;
    int wiek_;
};

// Dziedziczenie + virtual
class Pracownik : public Osoba {
public:
    Pracownik(std::string imie, int wiek, std::string stanowisko)
        : Osoba(std::move(imie), wiek)
        , stanowisko_(std::move(stanowisko)) {}

    void przedstaw_sie() const {
        Osoba::przedstaw();
        std::cout << "Stanowisko: " << stanowisko_ << '\n';
    }

private:
    std::string stanowisko_;
};

RAII - Resource Acquisition Is Initialization

Centralny idiom C++. Zasob jest pozyskany w konstruktorze i zwolniony w destruktorze. Niezawodny exception-safe management.

class FileHandle {
public:
    explicit FileHandle(const std::string& path)
        : file_(std::fopen(path.c_str(), "r"))
    {
        if (!file_) {
            throw std::runtime_error("Cannot open file");
        }
    }

    ~FileHandle() {
        if (file_) std::fclose(file_);
    }

    // Bez kopiowania
    FileHandle(const FileHandle&) = delete;
    FileHandle& operator=(const FileHandle&) = delete;

    // Move OK
    FileHandle(FileHandle&& o) noexcept : file_(o.file_) { o.file_ = nullptr; }

private:
    std::FILE* file_;
};

// Uzycie - automatyczne zamkniecie pliku przy wyjsciu z zakresu
void process() {
    FileHandle f("dane.txt");
    // ... uzycie f
} // automatyczny fclose w destruktorze

STL - std namespace

#include <vector>
#include <string>
#include <map>
#include <set>
#include <algorithm>

// std::string
std::string s = "Witaj";
s += ", swiecie!";
std::cout << s.length();           // 16
std::cout << s.substr(0, 5);       // "Witaj"

// std::vector - dynamiczna tablica
std::vector<int> v = {1, 2, 3};
v.push_back(4);
v.emplace_back(5);
std::cout << v.size();             // 5
std::cout << v[0];                 // 1

// std::map - posortowany slownik (drzewo czerwono-czarne)
std::map<std::string, int> m;
m["jeden"] = 1;
m["dwa"] = 2;

// std::unordered_map - hash table (szybsze)
std::unordered_map<std::string, int> um;

// Algorytmy
std::sort(v.begin(), v.end());
auto it = std::find(v.begin(), v.end(), 3);
int sum = std::accumulate(v.begin(), v.end(), 0);

Smart pointers - podstawy

Od C++11 surowe wskazniki (new/delete) sa rzadko potrzebne. Smart pointers automatycznie zarzadzaja czasem zycia.

#include <memory>

// std::unique_ptr - wylaczna wlasnosc
auto u = std::make_unique<Osoba>("Anna", 30);
u->przedstaw();
// Automatyczne usuniecie przy wyjsciu z zakresu

// std::shared_ptr - wspoldzielona wlasnosc (reference counting)
auto s1 = std::make_shared<Osoba>("Bob", 25);
auto s2 = s1;  // ref count = 2
// Usuniecie kiedy ostatni ref zniknie

// std::weak_ptr - bez wlasnosci, do przerywania cykli
std::weak_ptr<Osoba> w = s1;
if (auto locked = w.lock()) {
    locked->przedstaw();
}

Wskazowki dla nowych w C++

  • Zawsze -Wall -Wextra -Wpedantic przy kompilacji
  • Uzywaj RAII — zaden new/delete recznie
  • const correctness — const wszedzie gdzie mozna
  • Preferuj std::array / std::vector nad surowymi tablicami
  • Standardowe wytyczne: C++ Core Guidelines (Stroustrup & Sutter)
  • Wlacz sanitizers (ASan, UBSan) w trybie debug
