Qt 6 - GUI framework

Qt to najpopularniejszy cross-platform GUI framework. Qt 6 wprowadza nowoczesny C++17, lepszy QML, modulowa architektura.

// main.cpp - prosty Qt 6
#include <QApplication>
#include <QPushButton>
#include <QVBoxLayout>
#include <QWidget>

int main(int argc, char* argv[]) {
    QApplication app(argc, argv);

    QWidget window;
    window.setWindowTitle("Hello Qt 6");

    auto* layout = new QVBoxLayout(&window);
    auto* button = new QPushButton("Kliknij mnie");
    layout->addWidget(button);

    QObject::connect(button, &QPushButton::clicked, [] {
        qDebug() << "Button kliknij!";
    });

    window.show();
    return app.exec();
}
# CMakeLists.txt dla Qt 6
cmake_minimum_required(VERSION 3.20)
project(MyQtApp)

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_CXX_STANDARD 23)

find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

add_executable(myapp main.cpp)
target_link_libraries(myapp PRIVATE Qt6::Core Qt6::Widgets)

Abseil (Google)

Abseil to kolekcja bibliotek wykorzystywanych w Google — dropping-in alternatywy dla niektorych std typu, lepsze zachowanie w niektorych przypadkach.

// absl::flat_hash_map - duzo szybsze niz std::unordered_map
#include <absl/container/flat_hash_map.h>

absl::flat_hash_map<std::string, int> m;
m["a"] = 1;
m["b"] = 2;

// absl::Status - wynik operacji (jak std::expected)
absl::StatusOr<int> readNumber(const std::string& s);

// absl::Mutex z guard
absl::Mutex mu;
{
    absl::MutexLock lock(&mu);
    // critical section
}

fmt (i std::format)

Biblioteka fmt to "ojciec" std::format. Nadal popularna, bo dziala w starszych standardach.

#include <fmt/core.h>
#include <fmt/color.h>

fmt::print("{} ma {} lat\n", "Anna", 30);

// Kolory w terminalu
fmt::print(fg(fmt::color::red), "Blad!\n");
fmt::print(fg(fmt::color::green) | fmt::emphasis::bold, "OK\n");

// Specifiers
fmt::print("{:.2f}\n", 3.14159);    // 3.14
fmt::print("{:>10}\n", "abc");      // "       abc"

spdlog - logging

Bardzo szybki structured logger.

#include <spdlog/spdlog.h>
#include <spdlog/sinks/rotating_file_sink.h>

int main() {
    auto file_logger = spdlog::rotating_logger_mt(
        "file_logger",
        "logs/log.txt",
        1024 * 1024 * 5,    // 5 MB
        3                    // 3 pliki rotacja
    );
    spdlog::set_default_logger(file_logger);
    spdlog::set_level(spdlog::level::debug);
    spdlog::set_pattern("[%Y-%m-%d %H:%M:%S.%e] [%^%l%$] %v");

    spdlog::info("Aplikacja startuje");
    spdlog::warn("Wartosc {} przekracza limit", 100);
    spdlog::error("Blad: {}", "connection failed");
    spdlog::critical("Krytyczny blad!");
}

gRPC C++

// helloworld.proto
syntax = "proto3";
service Greeter {
  rpc SayHello (HelloRequest) returns (HelloReply);
}
message HelloRequest { string name = 1; }
message HelloReply { string message = 1; }
// Server
class GreeterServiceImpl final : public Greeter::Service {
    Status SayHello(ServerContext* context,
                    const HelloRequest* request,
                    HelloReply* reply) override {
        reply->set_message("Hello " + request->name());
        return Status::OK;
    }
};

// Client
auto channel = grpc::CreateChannel("localhost:50051",
    grpc::InsecureChannelCredentials());
auto stub = Greeter::NewStub(channel);
HelloRequest req;
req.set_name("Anna");
HelloReply reply;
ClientContext ctx;
auto status = stub->SayHello(&ctx, req, &reply);

Testowanie - Catch2 i Google Test

Catch2 (header-only, BDD-style)

#include <catch2/catch_test_macros.hpp>

TEST_CASE("Dodawanie", "[matematyka]") {
    REQUIRE(2 + 2 == 4);

    SECTION("ujemne") {
        REQUIRE(-1 + -1 == -2);
    }
}

SCENARIO("Vector resize", "[vector]") {
    GIVEN("Pusty wektor") {
        std::vector<int> v;

        WHEN("dodajemy element") {
            v.push_back(1);

            THEN("rozmiar jest 1") {
                REQUIRE(v.size() == 1);
            }
        }
    }
}

Google Test

#include <gtest/gtest.h>

TEST(MathTest, Addition) {
    EXPECT_EQ(2 + 2, 4);
    EXPECT_NE(1, 2);
    ASSERT_GT(5, 3);
}

// Test fixture
class DatabaseTest : public ::testing::Test {
protected:
    void SetUp() override {
        db.connect();
    }
    void TearDown() override {
        db.disconnect();
    }
    Database db;
};

TEST_F(DatabaseTest, CanInsert) {
    EXPECT_TRUE(db.insert("data"));
}

CMake (modern, target-based)

# CMakeLists.txt
cmake_minimum_required(VERSION 3.25)
project(MyApp VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Find dependencies
find_package(fmt CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)

# Library
add_library(mylib
    src/foo.cpp
    src/bar.cpp
)
target_include_directories(mylib PUBLIC include)
target_link_libraries(mylib PUBLIC fmt::fmt spdlog::spdlog)
target_compile_features(mylib PUBLIC cxx_std_23)

# Executable
add_executable(myapp src/main.cpp)
target_link_libraries(myapp PRIVATE mylib)

# Tests
enable_testing()
find_package(Catch2 3 CONFIG REQUIRED)
add_executable(tests tests/test_foo.cpp)
target_link_libraries(tests PRIVATE mylib Catch2::Catch2WithMain)
add_test(NAME tests COMMAND tests)

Package managers - vcpkg vs Conan

vcpkg (Microsoft)

# Instalacja
git clone https://github.com/microsoft/vcpkg
./vcpkg/bootstrap-vcpkg.sh

# Manifest mode - vcpkg.json
{
  "name": "myapp",
  "version": "1.0.0",
  "dependencies": [
    "fmt",
    "spdlog",
    {
      "name": "boost",
      "features": ["asio", "beast"]
    }
  ]
}

# CMake integration
cmake -B build \
  -DCMAKE_TOOLCHAIN_FILE=vcpkg/scripts/buildsystems/vcpkg.cmake

Conan 2

# conanfile.py
from conan import ConanFile
from conan.tools.cmake import CMake

class MyAppRecipe(ConanFile):
    settings = "os", "compiler", "build_type", "arch"
    generators = "CMakeDeps", "CMakeToolchain"

    def requirements(self):
        self.requires("fmt/10.2.1")
        self.requires("spdlog/1.13.0")
        self.requires("boost/1.84.0")

# Instalacja
conan install . --output-folder=build --build=missing
cmake -B build -DCMAKE_TOOLCHAIN_FILE=build/conan_toolchain.cmake

Sanitizers - znajdz bugi runtime

SanitizerWykrywa
AddressSanitizer (ASan)Use-after-free, buffer overflow, double-free
UndefinedBehaviorSanitizer (UBSan)Signed overflow, null deref, alignment
ThreadSanitizer (TSan)Data races, locks order
MemorySanitizer (MSan)Niezainicjalizowana pamiec (Clang only)
# Wlaczenie w GCC/Clang
g++ -fsanitize=address,undefined -g -O1 main.cpp
clang++ -fsanitize=thread -g main.cpp

# CMake
target_compile_options(myapp PRIVATE
    $<$<CONFIG:Debug>:-fsanitize=address,undefined>)
target_link_options(myapp PRIVATE
    $<$<CONFIG:Debug>:-fsanitize=address,undefined>)

Valgrind (Linux)

# Memcheck - memory errors
valgrind --leak-check=full --show-leak-kinds=all ./myapp

# Callgrind - profiling
valgrind --tool=callgrind ./myapp
kcachegrind callgrind.out.<pid>

# Helgrind - thread races
valgrind --tool=helgrind ./myapp
