Kompletny CMake template

# CMakeLists.txt - production-ready
cmake_minimum_required(VERSION 3.25)
project(MyApp
    VERSION 1.0.0
    DESCRIPTION "Modern C++ application"
    LANGUAGES CXX
)

# Standard
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Out-of-source builds
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "In-source build forbidden")
endif()

# Default build type
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Compile commands dla clangd
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Warnings
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Wshadow)
endif()

# Sanitizers w Debug
option(ENABLE_SANITIZERS "Enable ASan + UBSan" ON)
if(ENABLE_SANITIZERS AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

# Hardening
add_compile_definitions($<$<CONFIG:Release>:_FORTIFY_SOURCE=3>)

# Subdirectories
add_subdirectory(src)
add_subdirectory(tests)

# Install
include(GNUInstallDirs)
install(TARGETS myapp
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

Conan 2 - conanfile.py

from conan import ConanFile
from conan.tools.cmake import CMakeDeps, CMakeToolchain, CMake

class MyAppConan(ConanFile):
    name = "myapp"
    version = "1.0.0"
    settings = "os", "compiler", "build_type", "arch"
    options = {"shared": [True, False], "fPIC": [True, False]}
    default_options = {"shared": False, "fPIC": True}

    def requirements(self):
        self.requires("fmt/10.2.1")
        self.requires("spdlog/1.13.0")
        self.requires("nlohmann_json/3.11.3")

    def build_requirements(self):
        self.tool_requires("cmake/3.28.0")
        self.test_requires("catch2/3.5.0")

    def layout(self):
        from conan.tools.cmake import cmake_layout
        cmake_layout(self)

    def generate(self):
        deps = CMakeDeps(self)
        deps.generate()
        tc = CMakeToolchain(self)
        tc.generate()

Cross-compilation

Mac/Linux → Linux ARM

# Toolchain file - aarch64-toolchain.cmake
set(CMAKE_SYSTEM_NAME Linux)
set(CMAKE_SYSTEM_PROCESSOR aarch64)

set(CMAKE_C_COMPILER aarch64-linux-gnu-gcc)
set(CMAKE_CXX_COMPILER aarch64-linux-gnu-g++)

set(CMAKE_FIND_ROOT_PATH /usr/aarch64-linux-gnu)
set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)

# Build
cmake -B build -DCMAKE_TOOLCHAIN_FILE=aarch64-toolchain.cmake

Windows → Linux (MinGW)

# Z Windows uzywajac MinGW-w64 cross compiler
# Lub WSL2 - uruchom Linux toolchain w WSL

# Z Linux: cross-compile dla Windows (mingw-w64)
sudo apt install mingw-w64
x86_64-w64-mingw32-g++ -static-libgcc -static-libstdc++ main.cpp -o app.exe

zig cc - uniwersalny cross compiler

# zig instaluje sie jako jeden binary i ma vendoring libc dla wszystkich platform
zig c++ -target aarch64-linux-gnu main.cpp
zig c++ -target x86_64-windows-gnu main.cpp -o app.exe
zig c++ -target x86_64-macos main.cpp

Static vs dynamic linking

AspektStaticDynamic
Rozmiar binaryWiekszy (wszystko wbudowane)Mniejszy
Pamiec systemuKazdy proces ma swoja kopieWspoldzielenie kodu
Update bibliotekiWymaga re-linkWystarczy podmiana .so/.dll
Dependency hellBrak (samowystarczalne)Tak
Best dlaCLI tools, distrolessWieksze aplikacje
# GCC/Clang
g++ -static main.cpp -o myapp                # full static (jesli libc pozwala)
g++ -static-libgcc -static-libstdc++ main.cpp # tylko C++ runtime static

# CMake
add_executable(myapp main.cpp)
target_link_options(myapp PRIVATE -static-libgcc -static-libstdc++)

Docker dla C++ - multi-stage

# syntax=docker/dockerfile:1.7
# Stage 1: build
FROM ubuntu:24.04 AS build

RUN apt-get update && apt-get install -y --no-install-recommends \
    g++-13 \
    cmake \
    ninja-build \
    git \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /src
COPY . .

# vcpkg cache mount
RUN --mount=type=cache,target=/vcpkg/buildtrees \
    --mount=type=cache,target=/vcpkg/downloads \
    cmake -B build -G Ninja \
        -DCMAKE_BUILD_TYPE=Release \
        -DCMAKE_TOOLCHAIN_FILE=/vcpkg/scripts/buildsystems/vcpkg.cmake \
    && cmake --build build --parallel

# Stage 2: minimal runtime
FROM gcr.io/distroless/cc-debian12

COPY --from=build /src/build/myapp /usr/local/bin/myapp

USER nonroot:nonroot
ENTRYPOINT ["/usr/local/bin/myapp"]

Debugowanie - GDB

# Kompilacja z symbolami
g++ -g -O0 main.cpp -o myapp

# Uruchomienie
gdb ./myapp
(gdb) break main             # breakpoint
(gdb) run                    # start
(gdb) next                   # step over
(gdb) step                   # step into
(gdb) print x                # zmienna
(gdb) backtrace              # stack trace
(gdb) info locals            # lokalne
(gdb) continue               # do nast. break
(gdb) watch x                # watch zmiennej
(gdb) thread apply all bt    # bt wszystkich watkow

# .gdbinit - automatyczne komendy
echo "set print pretty on" > ~/.gdbinit
echo "set pagination off" >> ~/.gdbinit

GitHub Actions - CI dla C++

name: CI
on: [push, pull_request]

jobs:
  build:
    strategy:
      matrix:
        os: [ubuntu-latest, macos-latest, windows-latest]
        compiler: [gcc-13, clang-17, msvc]
        exclude:
          - os: windows-latest
            compiler: gcc-13

    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4

      - name: Setup vcpkg
        uses: lukka/run-vcpkg@v11

      - name: Configure
        run: cmake --preset=ci

      - name: Build
        run: cmake --build build --parallel

      - name: Test (z ASan)
        run: ctest --test-dir build --output-on-failure

      - name: Sanitizer build
        if: matrix.os == 'ubuntu-latest'
        run: |
          cmake -B build-asan -DENABLE_SANITIZERS=ON -DCMAKE_BUILD_TYPE=Debug
          cmake --build build-asan
          ctest --test-dir build-asan

Production checklist C++

  • Build z hardening flags (-D_FORTIFY_SOURCE=3, -fstack-protector-strong)
  • Symbole debug (-g) zachowane offline (split debug)
  • strip binary w produkcji
  • Crash reporting (Sentry, Crashpad, Breakpad)
  • Static analysis w CI (clang-tidy, cppcheck)
  • Sanitizers w testach (ASan, UBSan, TSan w nightly)
  • Reproducible builds (-fdebug-prefix-map, -frandom-seed)
  • Locked dependencies (vcpkg baseline / Conan lockfile)
