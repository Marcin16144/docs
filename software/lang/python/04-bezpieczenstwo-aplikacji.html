Deserialization attacks - pickle

pickle to najwiekszy security footgun w Pythonie. Pickle.loads() na danych z internetu = RCE.

# NIGDY na user input
import pickle
import os

# Atakujacy moze stworzyc malicious pickle
class Exploit:
    def __reduce__(self):
        return (os.system, ('rm -rf /',))

malicious = pickle.dumps(Exploit())
pickle.loads(malicious)   # uruchamia rm -rf!

# DOBRZE - JSON dla untrusted data
import json
data = json.loads(user_input)

# Inne bezpieczne formaty
import msgpack
data = msgpack.unpackb(bytes_input, raw=False)

# Cap'n Proto, MessagePack, Protocol Buffers - wszystkie bezpieczne

YAML - safe_load

import yaml

# ZLE - yaml.load wykonuje arbitrary code
data = yaml.load(user_input)   # niebezpieczne

# DOBRZE - safe_load (od PyYAML 6.0 jest default)
data = yaml.safe_load(user_input)

# Lub uzyj Loader=SafeLoader explicit
data = yaml.load(user_input, Loader=yaml.SafeLoader)

SQL Injection

# ZLE - string concat
import sqlite3
conn = sqlite3.connect("app.db")
cur = conn.cursor()

name = input("Imie: ")
cur.execute(f"SELECT * FROM users WHERE name = '{name}'")  # SQLi!

# DOBRZE - parametry
cur.execute("SELECT * FROM users WHERE name = ?", (name,))

# psycopg / postgres
cur.execute("SELECT * FROM users WHERE id = %s", (user_id,))

# SQLAlchemy ORM - automatyczna parametryzacja
session.scalars(select(User).where(User.name == name)).all()

# SQLAlchemy text() - z parametrami
session.execute(
    text("SELECT * FROM users WHERE name = :name"),
    {"name": name}
)

# Django ORM - automatyczne
User.objects.filter(name=name)

Command Injection

import subprocess
import shlex

# ZLE - shell=True z user input
filename = input("Plik: ")
subprocess.run(f"cat {filename}", shell=True)   # injection!

# DOBRZE - lista argumentow, shell=False (default)
subprocess.run(["cat", filename])

# Jesli musisz shell - shlex.quote
subprocess.run(f"cat {shlex.quote(filename)}", shell=True)

# os.system jest jeszcze gorsze
os.system(f"cat {filename}")   # NIGDY z user input

# DOBRZE - dla matematycznych wyrazen
import ast
import operator

OPS = {
    ast.Add: operator.add, ast.Sub: operator.sub,
    ast.Mult: operator.mul, ast.Div: operator.truediv,
}

def safe_eval(expr):
    tree = ast.parse(expr, mode='eval')
    return _eval(tree.body)

def _eval(node):
    if isinstance(node, ast.Constant):
        return node.value
    if isinstance(node, ast.BinOp):
        return OPS[type(node.op)](_eval(node.left), _eval(node.right))
    raise ValueError("Niedozwolony")

# Lub z biblioteki: simpleeval, asteval

Path Traversal

import os
from pathlib import Path

# ZLE
filename = request.args.get("file")

# atakujacy: ../../etc/passwd

# DOBRZE - kanonizacja + sprawdzenie prefix

# Sprawdz czy jest pod base

    raise ValueError("Path traversal!")

Hashing hasel - Argon2 / bcrypt

# Argon2 (rekomendowane 2026)
from argon2 import PasswordHasher

ph = PasswordHasher(
    time_cost=4,
    memory_cost=65536,    # 64 MB
    parallelism=2,
)

hash = ph.hash("user_password")

try:
    ph.verify(hash, "user_password")
    if ph.check_needs_rehash(hash):
        new_hash = ph.hash("user_password")
except VerifyMismatchError:
    print("Bledne haslo")

# bcrypt (alternatywa)
import bcrypt

hash = bcrypt.hashpw(b"password", bcrypt.gensalt(rounds=12))
if bcrypt.checkpw(b"password", hash):
    print("OK")

# NIGDY: md5, sha1, sha256 dla hasel

JWT - bezpieczne uzycie

import jwt
from datetime import datetime, timedelta

SECRET = "supersecret-min-32-chars"

# Tworzenie
def create_token(user_id: int) -> str:
    payload = {
        "sub": str(user_id),
        "iat": datetime.utcnow(),
        "exp": datetime.utcnow() + timedelta(hours=1),
    }
    return jwt.encode(payload, SECRET, algorithm="HS256")

# Weryfikacja - ZAWSZE z algorithms
def verify_token(token: str) -> dict:
    return jwt.decode(
        token,
        SECRET,
        algorithms=["HS256"],   # WYMOG! bez tego algorithm confusion attack
    )

# Najlepiej - asymetryczne (RS256, ES256)
# klucz publiczny do walidacji, prywatny do podpisu

Security checklist Python

  • NIGDY pickle.loads na user input
  • SQL: zawsze parametryzacja
  • subprocess: shell=False (default)
  • Hashowanie: Argon2 / bcrypt
  • JWT z eksplicytnym algorithm
  • pip-audit + Dependabot w CI
  • bandit / ruff S w pre-commit
  • Secrets w env vars, nie w kodzie
  • HTTPS only + HSTS w produkcji
  • Pin wersje + hash w requirements
