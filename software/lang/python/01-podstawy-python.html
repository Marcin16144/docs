OOP w Python

class Osoba:
    # Class variable
    gatunek: str = "Homo sapiens"

    # Constructor
    def __init__(self, imie: str, wiek: int) -> None:
        self.imie = imie
        self.wiek = wiek

    # Method
    def przedstaw(self) -> str:
        return f"Jestem {self.imie}, lat {self.wiek}"

    # Dunder methods
    def __repr__(self) -> str:
        return f"Osoba({self.imie!r}, {self.wiek})"

    def __eq__(self, other) -> bool:
        if not isinstance(other, Osoba):
            return NotImplemented
        return self.imie == other.imie and self.wiek == other.wiek

# Dziedziczenie
class Pracownik(Osoba):
    def __init__(self, imie: str, wiek: int, stanowisko: str) -> None:
        super().__init__(imie, wiek)
        self.stanowisko = stanowisko

    def przedstaw(self) -> str:
        base = super().przedstaw()
        return f"{base}, {self.stanowisko}"

# Properties
class Konto:
    def __init__(self, saldo: float = 0):
        self._saldo = saldo

    @property
    def saldo(self) -> float:
        return self._saldo

    @saldo.setter
    def saldo(self, wartosc: float) -> None:
        if wartosc < 0:
            raise ValueError("Saldo nie moze byc ujemne")
        self._saldo = wartosc

Dataclasses (basic)

from dataclasses import dataclass, field

@dataclass
class User:
    imie: str
    email: str
    wiek: int = 0
    tagi: list[str] = field(default_factory=list)

u = User(imie="Anna", email="a@b.pl", wiek=30)
print(u)        # User(imie='Anna', email='a@b.pl', wiek=30, tagi=[])

# Frozen (immutable)
@dataclass(frozen=True, slots=True)
class Point:
    x: float
    y: float

Async/await - podstawy

import asyncio

async def pobierz_dane(url: str) -> str:
    print(f"Pobieram {url}")
    await asyncio.sleep(1)        # symulacja I/O
    return f"Dane z {url}"

async def main():
    # Sekwencyjnie
    a = await pobierz_dane("a.pl")
    b = await pobierz_dane("b.pl")

    # Rownolegle
    a, b = await asyncio.gather(
        pobierz_dane("a.pl"),
        pobierz_dane("b.pl"),
    )

    # TaskGroup (3.11+)
    async with asyncio.TaskGroup() as tg:
        t1 = tg.create_task(pobierz_dane("a.pl"))
        t2 = tg.create_task(pobierz_dane("b.pl"))

asyncio.run(main())

Type hints - fundamentals

from typing import Optional, Union, Callable, Any
from collections.abc import Sequence, Iterable

# Podstawowe
def fn(x: int, y: str) -> bool: ...

# Generic types
def pierwsza(items: list[int]) -> int:
    return items[0]

# Optional / Union (3.10+ - skladnia x | None)
def find(id: int) -> User | None: ...

def parse(value: int | str) -> int: ...

# Callable
def apply(fn: Callable[[int, int], int], a: int, b: int) -> int:
    return fn(a, b)

# Generic protocol
from typing import Protocol

class Drawable(Protocol):
    def draw(self) -> None: ...

def render(obj: Drawable) -> None:
    obj.draw()
