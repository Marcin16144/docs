Protocol Buffers — definicja typow

syntax = "proto3";
package ecommerce;

message User {
  int32 id = 1;               // numer pola (nie wartosc!)
  string name = 2;
  string email = 3;
  UserRole role = 4;
  repeated Order orders = 5;  // lista
  optional string phone = 6;  // opcjonalne
}

enum UserRole {
  USER_ROLE_UNSPECIFIED = 0;  // zawsze 0 jako domyslna
  USER_ROLE_USER = 1;
  USER_ROLE_ADMIN = 2;
  USER_ROLE_MODERATOR = 3;
}

message Order {
  int32 id = 1;
  int32 user_id = 2;
  repeated OrderItem items = 3;
  double total = 4;
  OrderStatus status = 5;
}

enum OrderStatus {
  ORDER_STATUS_UNSPECIFIED = 0;
  ORDER_STATUS_PENDING = 1;
  ORDER_STATUS_SHIPPED = 3;
  ORDER_STATUS_DELIVERED = 4;
  ORDER_STATUS_CANCELLED = 5;
}

Reguly numeracji pol

  • Numery pol to identyfikatory binarne — nigdy ich nie zmieniaj po wdrozeniu
  • Numery 1-15 zajmuja 1 bajt (uzywaj dla czestych pol)
  • Numery 16-2047 zajmuja 2 bajty
  • Usuniete pola oznacz jako reserved

Definicja serwisu (.proto)

service UserService {
  // Unary — request/response
  rpc GetUser(GetUserRequest) returns (GetUserResponse);
  rpc CreateUser(CreateUserRequest) returns (CreateUserResponse);

  // Server streaming — serwer wysyla strumien
  rpc ListUsers(ListUsersRequest) returns (stream User);

  // Client streaming — klient wysyla strumien
  rpc UploadPhotos(stream UploadPhotoRequest)
      returns (UploadPhotoResponse);

  // Bidirectional streaming
  rpc Chat(stream ChatMessage) returns (stream ChatMessage);
}

message GetUserRequest { int32 id = 1; }
message GetUserResponse { User user = 1; }
message ListUsersRequest {
  int32 page_size = 1;
  string page_token = 2;
}

Generowanie kodu

# Go
protoc --go_out=. --go-grpc_out=. proto/user.proto

# Python
protoc --python_out=. --grpc_python_out=. proto/user.proto

# C#
protoc --csharp_out=. --grpc_csharp_out=. proto/user.proto

# Java
protoc --java_out=. --grpc_java_out=. proto/user.proto

Kompilator protoc generuje klasy klienta i serwera z pliku .proto — zapewnia spojnosc kontraktu miedzy jezykami.

4 typy komunikacji (streaming)

1. Unary RPC (Request-Response)

Klient wysyla jedno zadanie, serwer zwraca jedna odpowiedz.

// Serwer (Go)
func (s *server) GetUser(ctx context.Context,
    req *pb.GetUserRequest) (*pb.GetUserResponse, error) {
    user, err := s.db.FindUser(req.Id)
    if err != nil {
        return nil, status.Errorf(codes.NotFound,
            "user %d not found", req.Id)
    }
    return &pb.GetUserResponse{User: user}, nil
}

// Klient
resp, err := client.GetUser(ctx, &pb.GetUserRequest{Id: 123})
fmt.Println(resp.User.Name)

2. Server Streaming

Klient wysyla jedno zadanie, serwer odpowiada strumieniem wiadomosci.

// Serwer
func (s *server) ListUsers(req *pb.ListUsersRequest,
    stream pb.UserService_ListUsersServer) error {
    users, _ := s.db.GetAllUsers(req.Filter)
    for _, user := range users {
        if err := stream.Send(user); err != nil {
            return err
        }
    }
    return nil
}

// Klient
stream, _ := client.ListUsers(ctx, req)
for {
    user, err := stream.Recv()
    if err == io.EOF { break }
    fmt.Println(user.Name)
}

3. Client Streaming

Klient wysyla strumien wiadomosci, serwer odpowiada jedna odpowiedzia.

// Serwer
func (s *server) UploadPhotos(
    stream pb.UserService_UploadPhotosServer) error {
    var totalSize int64
    for {
        photo, err := stream.Recv()
        if err == io.EOF {
            return stream.SendAndClose(
                &pb.UploadPhotoResponse{Count: totalSize})
        }
        totalSize += int64(len(photo.Data))
        s.storage.Save(photo)
    }
}

// Klient
stream, _ := client.UploadPhotos(ctx)
for _, photo := range photos {
    stream.Send(&pb.UploadPhotoRequest{Data: photo})
}
resp, _ := stream.CloseAndRecv()

4. Bidirectional Streaming

Klient i serwer wysylaja strumienie niezaleznie — idealny dla czatu, danych real-time.

// Serwer
func (s *server) Chat(
    stream pb.UserService_ChatServer) error {
    for {
        msg, err := stream.Recv()
        if err == io.EOF { return nil }
        response := processMessage(msg)
        stream.Send(response)
    }
}

// Klient — odczyt i zapis rownolegle
stream, _ := client.Chat(ctx)
go func() {
    for _, msg := range messages {
        stream.Send(msg)
    }
    stream.CloseSend()
}()
for {
    resp, err := stream.Recv()
    if err == io.EOF { break }
    fmt.Println(resp.Text)
}

Deadlines i Timeouts

gRPC uzywa deadlines zamiast timeoutow — deadline to absolutny punkt w czasie, ktory propaguje sie w lancuchu wywolan.

// Klient ustawia deadline (5 sekund)
ctx, cancel := context.WithTimeout(
    context.Background(), 5*time.Second)
defer cancel()

resp, err := client.GetUser(ctx, req)
if err != nil {
    st, _ := status.FromError(err)
    if st.Code() == codes.DeadlineExceeded {
        log.Println("Timeout — serwer nie odpowiedzial w 5s")
    }
}

// Deadline propaguje sie: A → B → C
// Jesli A ustawil 5s, B i C znaja pozostaly czas

Load Balancing w gRPC

HTTP/2 multipleksuje wiele zadan na jednym polaczeniu TCP — L4 load balancer widzi jedno polaczenie. Potrzebny jest L7 load balancer (Envoy, Nginx) rozumiejacy ramki HTTP/2.

Strategie

  1. Proxy-based (L7) — Envoy/Nginx terminuje gRPC, tworzy nowe polaczenia do backendu
  2. Client-side — klient zna liste serwerow, sam rozdziela zadania (round-robin)
  3. Look-aside — klient odpytuje service discovery (Consul, etcd)

gRPC-Web dla przegladarek

Architektura:

Przegladarka (gRPC-Web klient)
    |
    | HTTP/1.1 lub HTTP/2 (bez trailerow)
    v
Envoy Proxy (translacja)
    |
    | gRPC natywny (HTTP/2)
    v
gRPC Serwer

Ograniczenia: brak client streaming i bidirectional, wymaga proxy (Envoy).

Kiedy uzywac gRPC vs REST?

Uzywaj gRPC:

  • Komunikacja wewnetrzna (mikroserwisy)
  • Wysoka wydajnosc, niskie opoznienia
  • Streaming danych real-time
  • Scisly kontrakt API
  • Wiele jezykow programowania
  • Duze wolumeny danych

Uzywaj REST:

  • API publiczne
  • Aplikacje przegladarkowe (bez proxy)
  • Proste operacje CRUD
  • Latwe debugowanie (JSON)
  • Zespol nie zna gRPC
  • Caching HTTP krytyczny

Podejscie hybrydowe

[Przegladarka] → REST  → [API Gateway] → gRPC → [Serwis A]
[Mobile App]   → REST  → [API Gateway] → gRPC → [Serwis B]
                                              → gRPC → [Serwis C]

REST na zewnatrz (latwosc), gRPC wewnatrz (wydajnosc)

Dobre praktyki

  1. Projektuj API-first — .proto jest kontraktem, projektuj swiadomie
  2. Wersjonuj .proto — uzywaj package versioning (v1, v2)
  3. Nigdy nie zmieniaj numerow pol — dodawaj nowe, stare oznacz jako reserved
  4. Ustawiaj deadlines — zawsze, na kazdym poziomie lancucha
  5. Uzywaj interceptorow — middleware dla logowania, auth, metryk
  6. Health checking — standard gRPC Health Checking Protocol
  7. Monitoruj — Prometheus + gRPC metrics
  8. Uzywaj reflection — ulatwia debugging (grpcurl)
