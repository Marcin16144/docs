Interface Segregation Principle (ISP)

Wiele malych, specyficznych interfejsow zamiast jednego duzego — klient nie powinien zalezec od metod, ktorych nie uzywa.

Problem "tlustego" interfejsu

// ZLE — robot musi implementowac metody, ktorych nie obsluguje
interface Worker {
    work(): void;
    eat(): void;
    sleep(): void;
    attendMeeting(): void;
    writeReport(): void;
    driveCompanyCar(): void;
}

class RobotWorker implements Worker {
    work() { /* OK */ }
    eat() { throw new Error("Robot nie je!"); }
    sleep() { throw new Error("Robot nie spi!"); }
    attendMeeting() { throw new Error("Nie dotyczy"); }
    writeReport() { throw new Error("Nie dotyczy"); }
    driveCompanyCar() { throw new Error("Nie dotyczy"); }
}

Problemy: rzucanie wyjatkow narusza LSP, zmiana w Worker wymusza zmiany we wszystkich implementacjach, testy wymagaja mockowania niepotrzebnych metod.

Rozwiazanie: Male, rolowe interfejsy

interface Workable {
    work(): void;
}

interface Feedable {
    eat(): void;
    sleep(): void;
}

interface MeetingAttendee {
    attendMeeting(): void;
}

interface ReportWriter {
    writeReport(): void;
}

// Pracownik biurowy — wszystkie role
class OfficeWorker implements Workable, Feedable,
    MeetingAttendee, ReportWriter {
    work() { /* ... */ }
    eat() { /* ... */ }
    sleep() { /* ... */ }
    attendMeeting() { /* ... */ }
    writeReport() { /* ... */ }
}

// Robot — tylko Workable
class RobotWorker implements Workable {
    work() { /* ... */ }
}

// Funkcje przyjmuja dokladnie to, czego potrzebuja
function scheduleWork(worker: Workable) { worker.work(); }
function scheduleLunch(worker: Feedable) { worker.eat(); }

Przyklad 2: Repozytorium danych (Python)

Zle — monolityczne repozytorium

class UserRepository:
    def find_by_id(self, id): ...
    def find_all(self): ...
    def find_by_email(self, email): ...
    def save(self, user): ...
    def update(self, user): ...
    def delete(self, id): ...
    def bulk_insert(self, users): ...
    def export_to_csv(self): ...         # nie pasuje do repo
    def send_welcome_email(self, user): ... # zdecydowanie nie

Dobrze — interfejsy wedlug rol

from abc import ABC, abstractmethod

class UserReader(ABC):
    @abstractmethod
    def find_by_id(self, id: int) -> User: ...
    @abstractmethod
    def find_by_email(self, email: str) -> User: ...
    @abstractmethod
    def find_all(self) -> list[User]: ...

class UserWriter(ABC):
    @abstractmethod
    def save(self, user: User) -> None: ...
    @abstractmethod
    def update(self, user: User) -> None: ...

class UserRemover(ABC):
    @abstractmethod
    def delete(self, id: int) -> None: ...

# Implementacja realizuje wiele interfejsow
class SqlUserRepository(UserReader, UserWriter, UserRemover):
    def find_by_id(self, id):
        return self.db.query("SELECT ... WHERE id=%s", id)
    def find_by_email(self, email):
        return self.db.query("SELECT ... WHERE email=%s", email)
    def find_all(self):
        return self.db.query("SELECT * FROM users")
    def save(self, user):
        self.db.execute("INSERT INTO users ...", user)
    def update(self, user):
        self.db.execute("UPDATE users ...", user)
    def delete(self, id):
        self.db.execute("DELETE FROM users WHERE id=%s", id)

# Serwisy deklaruja dokladnie swoje potrzeby
class UserReportService:
    def __init__(self, reader: UserReader):  # tylko odczyt!
        self.reader = reader

class UserProfileService:
    def __init__(self, reader: UserReader, writer: UserWriter):
        self.reader = reader
        self.writer = writer

Przyklad 3: Adapter dla zewnetrznych API

// Zewnetrzna biblioteka — szerokie API
interface CloudProvider {
    createVM(config: VMConfig): VM;
    deleteVM(id: string): void;
    listVMs(): VM[];
    createStorage(config: StorageConfig): Storage;
    deleteStorage(id: string): void;
    createNetwork(config: NetworkConfig): Network;
    // ... 50 kolejnych metod
}

// Twoj kod potrzebuje tylko VM
interface VMManager {
    create(config: VMConfig): VM;
    delete(id: string): void;
    list(): VM[];
}

// Adapter zaweza interfejs
class CloudVMManager implements VMManager {
    constructor(private cloud: CloudProvider) {}

    create(config: VMConfig): VM {
        return this.cloud.createVM(config);
    }
    delete(id: string): void {
        this.cloud.deleteVM(id);
    }
    list(): VM[] {
        return this.cloud.listVMs();
    }
}

// Serwis zalezy od waskiego interfejsu
class DeploymentService {
    constructor(private vms: VMManager) {}  // nie CloudProvider!

    deploy(app: Application): void {
        const vm = this.vms.create({
            cpu: app.requiredCPU,
            memory: app.requiredMemory
        });
    }
}

Przyklad 4: ISP w konfiguracji

// ZLE — jeden obiekt konfiguracji, wszyscy widza wszystko
interface AppConfig {
    dbHost: string; dbPort: number;
    dbUser: string; dbPassword: string;
    smtpHost: string; smtpPort: number;
    jwtSecret: string; logLevel: string;
}

// EmailService widzi haslo do bazy danych!
class EmailService {
    constructor(private config: AppConfig) {}
}

// DOBRZE — wydzielone konfiguracje
interface SmtpConfig {
    host: string; port: number; user: string;
}

interface DatabaseConfig {
    host: string; port: number;
    user: string; password: string;
}

class EmailService {
    constructor(private smtp: SmtpConfig) {}
    // widzi TYLKO konfiguracje SMTP
}

Sygnaly naruszenia ISP

  • Implementacje zawieraja metody rzucajace NotImplementedError
  • Interfejs ma wiecej niz 5-7 metod
  • Klasy implementujace interfejs maja wiele pustych metod
  • Testy wymagaja mockowania metod, ktorych testowany kod nie wywoluje
  • Zmiana w interfejsie wymusza zmiany w klasach niekorzystajacych ze zmienionej metody
