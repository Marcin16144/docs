Code splitting — React.lazy

Dzielisz bundle na mniejsze chunki ladowane na zadanie.

import { lazy, Suspense } from 'react';
import { Routes, Route } from 'react-router';

// Zamiast import na gorze
const Dashboard = lazy(() => import('./pages/Dashboard'));
const Settings = lazy(() => import('./pages/Settings'));
const AdminPanel = lazy(() => import('./pages/AdminPanel'));

export default function App() {
  return (
    <Suspense fallback={<Spinner />}>
      <Routes>
        <Route path="/dashboard" element={<Dashboard />} />
        <Route path="/settings" element={<Settings />} />
        <Route path="/admin" element={<AdminPanel />} />
      </Routes>
    </Suspense>
  );
}

Lazy ladowanie ciezkich komponentow (np. edytor, wykresy):

const Chart = lazy(() => import('./Chart'));

function Page() {
  const [showChart, setShowChart] = useState(false);
  return (
    <>
      <button onClick={() => setShowChart(true)}>Pokaz wykres</button>
      {showChart && (
        <Suspense fallback={<p>Ladowanie wykresu...</p>}>
          <Chart />
        </Suspense>
      )}
    </>
  );
}

Suspense streaming

W React 18+ Suspense streamuje HTML z serwera. Klient widzi skeleton, dane wpadaja gdy gotowe. Dziala automatycznie z RSC.

// app/dashboard/page.tsx (Next.js Server Component)
import { Suspense } from 'react';

export default function Dashboard() {
  return (
    <>
      <Header /> {/* renderuje sie od razu */}

      <Suspense fallback={<StatsSkeleton />}>
        <UserStats /> {/* czeka na dane, klient dostaje placeholder */}
      </Suspense>

      <Suspense fallback={<ChartSkeleton />}>
        <Charts /> {/* niezaleznie od UserStats */}
      </Suspense>
    </>
  );
}

Virtualizacja — dlugie listy

Renderuj tylko widoczne wiersze. Lista 10 000 elementow — widzisz 30, render 30. Tanstack Virtual to standard.

npm i @tanstack/react-virtual

import { useVirtualizer } from '@tanstack/react-virtual';
import { useRef } from 'react';

function VirtualList({ items }) {
  const parentRef = useRef(null);

  const v = useVirtualizer({
    count: items.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 48,
    overscan: 5,
  });

  return (
    <div ref={parentRef} style={{ height: 600, overflow: 'auto' }}>
      <div style={{ height: v.getTotalSize(), position: 'relative' }}>
        {v.getVirtualItems().map(row => (
          <div
            key={row.key}
            style={{
              position: 'absolute',
              top: 0,
              left: 0,
              width: '100%',
              height: row.size,
              transform: `translateY(${row.start}px)`,
            }}
          >
            {items[row.index].name}
          </div>
        ))}
      </div>
    </div>
  );
}

Prefetching

Pobieraj dane/kod ZANIM uzytkownik kliknie.

// Next.js Link — auto prefetch w viewport
<Link href="/dashboard" prefetch>Dashboard</Link>

// React Router 7 — prefetch na hover/intent
<Link to="/dashboard" prefetch="intent">Dashboard</Link>

// Tanstack Query — prefetch dane przed nawigacja
const qc = useQueryClient();
const prefetchPost = (id) => qc.prefetchQuery({
  queryKey: ['post', id],
  queryFn: () => fetchPost(id),
});

<Link onMouseEnter={() => prefetchPost(123)} to="/posts/123">Czytaj</Link>

// React.memo — nie rerenduj jesli props identyczne (shallow)
// W React 19 z Compilerem — AUTOMATYCZNE
const Heavy = React.memo(function Heavy({ data }) {
  // expensive render...
  return <div>{data.title}</div>;
});

// Custom comparator
const Heavy = React.memo(
  function Heavy(props) { ... },
  (prev, next) => prev.id === next.id // tylko id liczy sie
);

// Hooks — useMemo/useCallback (znow, Compiler robi automat)
const filtered = useMemo(
  () => items.filter(i => i.active),
