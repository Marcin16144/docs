1. Python — FastAPI

# pip install fastapi uvicorn sqlalchemy asyncpg pydantic
from fastapi import FastAPI, HTTPException, Depends, Query
from pydantic import BaseModel, Field
from uuid import UUID, uuid4
from decimal import Decimal

app = FastAPI(title="Products API")

class ProductIn(BaseModel):
    name: str = Field(min_length=1, max_length=200)
    price: Decimal = Field(gt=0)
    stock: int = Field(ge=0, default=0)

class ProductOut(ProductIn):
    id: UUID

db: dict[UUID, ProductOut] = {}  # w prod: PostgreSQL

@app.get("/products", response_model=list[ProductOut])
async def list_products(page: int = Query(1, ge=1), size: int = Query(20, le=100)):
    items = list(db.values())[(page-1)*size : page*size]
    return items

@app.post("/products", response_model=ProductOut, status_code=201)
async def create_product(p: ProductIn):
    out = ProductOut(id=uuid4(), **p.model_dump())
    db[out.id] = out
    return out

@app.get("/products/{pid}", response_model=ProductOut)
async def get_product(pid: UUID):
    if pid not in db: raise HTTPException(404, "Not found")
    return db[pid]

@app.delete("/products/{pid}", status_code=204)
async def delete_product(pid: UUID):
    db.pop(pid, None)

# Uruchom: uvicorn main:app --reload
# Docs: http://localhost:8000/docs

2. Node.js — Express + Zod

// npm i express zod @types/express tsx
import express from 'express';
import { z } from 'zod';
import { randomUUID } from 'crypto';

const app = express();
app.use(express.json());

const ProductSchema = z.object({
    name: z.string().min(1).max(200),
    price: z.number().positive(),
    stock: z.number().int().nonnegative().default(0)
});
type Product = z.infer<typeof ProductSchema> & { id: string };

const db = new Map<string, Product>();

app.get('/products', (req, res) => {
    const page = +(req.query.page ?? 1);
    const size = Math.min(+(req.query.size ?? 20), 100);
    const all = [...db.values()];
    res.json(all.slice((page-1)*size, page*size));
});

app.post('/products', (req, res) => {
    const parsed = ProductSchema.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ errors: parsed.error.errors });
    const product = { id: randomUUID(), ...parsed.data };
    db.set(product.id, product);
    res.status(201).json(product);
});

app.get('/products/:id', (req, res) => {
    const p = db.get(req.params.id);
    if (!p) return res.status(404).json({ error: 'Not found' });
    res.json(p);
});

app.listen(3000, () => console.log('http://localhost:3000'));

3. C# / .NET 9 — Minimal API

// dotnet new web -n Api && cd Api && dotnet add package FluentValidation.AspNetCore
using FluentValidation;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddValidatorsFromAssemblyContaining<Program>();
var app = builder.Build();

record Product(Guid Id, string Name, decimal Price, int Stock);
record ProductIn(string Name, decimal Price, int Stock);

class ProductInValidator : AbstractValidator<ProductIn>
{
    public ProductInValidator()
    {
        RuleFor(x => x.Name).NotEmpty().MaximumLength(200);
        RuleFor(x => x.Price).GreaterThan(0);
        RuleFor(x => x.Stock).GreaterThanOrEqualTo(0);
    }
}

var db = new Dictionary<Guid, Product>();

app.MapGet("/products", (int page = 1, int size = 20) =>
    db.Values.Skip((page-1)*size).Take(Math.Min(size, 100)));

app.MapPost("/products", async (ProductIn p, IValidator<ProductIn> v) =>
{
    var r = await v.ValidateAsync(p);
    if (!r.IsValid) return Results.ValidationProblem(r.ToDictionary());
    var product = new Product(Guid.NewGuid(), p.Name, p.Price, p.Stock);
    db[product.Id] = product;
    return Results.Created($"/products/{product.Id}", product);
});

app.MapGet("/products/{id}", (Guid id) =>
    db.TryGetValue(id, out var p) ? Results.Ok(p) : Results.NotFound());

app.Run();

4. Rust — Axum + SQLx

// Cargo.toml: axum = "0.7", tokio = { version = "1", features = ["full"] }
// sqlx = { version = "0.8", features = ["postgres", "uuid", "runtime-tokio-rustls"] }
// serde = { version = "1", features = ["derive"] }, uuid = { version = "1", features = ["v4", "serde"] }

use axum::{
    extract::{Path, Query, State},
    http::StatusCode, response::IntoResponse, routing::{get, post}, Json, Router,
};
use serde::{Deserialize, Serialize};
use sqlx::PgPool;
use uuid::Uuid;

#[derive(Serialize, sqlx::FromRow)]
struct Product { id: Uuid, name: String, price: f64, stock: i32 }

#[derive(Deserialize)]
struct ProductIn { name: String, price: f64, stock: Option<i32> }

#[derive(Deserialize)]
struct Page { page: Option<i64>, size: Option<i64> }

async fn list(State(db): State<PgPool>, Query(p): Query<Page>) -> impl IntoResponse {
    let size = p.size.unwrap_or(20).min(100);
    let page = p.page.unwrap_or(1);
    let rows = sqlx::query_as::<_, Product>(
        "SELECT * FROM products ORDER BY name LIMIT $1 OFFSET $2")
        .bind(size).bind((page-1)*size).fetch_all(&db).await.unwrap();
    Json(rows)
}

async fn create(State(db): State<PgPool>, Json(p): Json<ProductIn>) -> impl IntoResponse {
    if p.price <= 0.0 return StatusCode::BAD_REQUEST.into_response();
    let row = sqlx::query_as::<_, Product>(
        "INSERT INTO products (id, name, price, stock) VALUES ($1,$2,$3,$4) RETURNING *")
        .bind(Uuid::new_v4()).bind(&p.name).bind(p.price).bind(p.stock.unwrap_or(0))
        .fetch_one(&db).await.unwrap();
    (StatusCode::CREATED, Json(row)).into_response()
}

#[tokio::main]
async fn main() {
    let db = PgPool::connect("postgres://localhost/shop").await.unwrap();
    let app = Router::new()
        .route("/products", get(list).post(create))
        .with_state(db);
    let listener = tokio::net::TcpListener::bind("0.0.0.0:3000").await.unwrap();
    axum::serve(listener, app).await.unwrap();
}

5. Go — Fiber (najszybsze w benchmarkach 2026)

// go mod init products && go get github.com/gofiber/fiber/v3 github.com/google/uuid
package main

import (
    "github.com/gofiber/fiber/v3"
    "github.com/google/uuid"
    "sync"
)

type Product struct {
    ID    string  `json:"id"`
    Name  string  `json:"name" validate:"required,max=200"`
    Price float64 `json:"price" validate:"gt=0"`
    Stock int     `json:"stock" validate:"gte=0"`
}

var (
    db = map[string]Product{}
    mu sync.RWMutex
)

func main() {
    app := fiber.New()
    app.Get("/products", func(c fiber.Ctx) error {
        mu.RLock(); defer mu.RUnlock()
        out := make([]Product, 0, len(db))
        for _, p := range db { out = append(out, p) }
        return c.JSON(out)
    })
    app.Post("/products", func(c fiber.Ctx) error {
        var p Product
        if err := c.Bind().JSON(&p); err != nil {
            return c.Status(400).JSON(fiber.Map{"error": err.Error()})
        }
        p.ID = uuid.NewString()
        mu.Lock(); db[p.ID] = p; mu.Unlock()
        return c.Status(201).JSON(p)
    })
    app.Listen(":3000")
}

6. Wzorzec wspólny: walidacja, błędy, ETag

Niezależnie od języka, dobre REST API ma:

AspektPraktyka
Status codes200 OK / 201 Created / 204 No Content / 400 Bad / 404 / 409 Conflict / 422 Validation
WalidacjaZod (TS), Pydantic (Py), FluentValidation (.NET), validator (Go) — fail-fast
BłędyRFC 9457 (Problem Details for HTTP APIs)
Paginacja?page=1&size=20 + nagłówki X-Total-Count / Link
WersjonowanieURL /v1/... lub Accept: application/vnd.api+json;version=1
CacheETag + If-None-Match → 304
OpenAPIAuto-generowany z kodu (FastAPI, NSwag, oapi-codegen)

Problem Details (RFC 9457) — uniwersalny format błędu

{
  "type": "https://example.com/errors/validation",
  "title": "Validation failed",
  "status": 422,
  "detail": "price must be greater than 0",
  "instance": "/products",
  "errors": {
    "price": ["must be greater than 0"]
  }
}
