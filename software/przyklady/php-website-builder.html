<?php
// private/src/Router.php
namespace App;

class Router
{
    private array $routes = [];
    private string $prefix = '';

    public function get(string $path, callable $handler): void   { $this->add('GET', $path, $handler); }
    public function post(string $path, callable $handler): void  { $this->add('POST', $path, $handler); }
    public function put(string $path, callable $handler): void   { $this->add('PUT', $path, $handler); }
    public function delete(string $path, callable $handler): void{ $this->add('DELETE', $path, $handler); }

    public function group(string $prefix, callable $fn): void
    {
        $old = $this->prefix;
        $this->prefix .= $prefix;
        $fn($this);
        $this->prefix = $old;
    }

    private function add(string $method, string $path, callable $handler): void
    {
        $full = rtrim($this->prefix . $path, '/') ?: '/';
        // {param} → (?P<param>[^/]+)
        $pattern = '#^' . preg_replace('/\{(\w+)\}/', '(?P<$1>[^/]+)', $full) . '$#';
        $this->routes[] = compact('method', 'pattern', 'handler');
    }

    public function dispatch(string $method, string $path): void
    {
        // PUT/DELETE w PHP - dla form: _method override
        if ($method === 'POST' && isset($_POST['_method'])) {
            $method = strtoupper($_POST['_method']);
        }

        foreach ($this->routes as $r) {
            if ($r['method'] !== $method) continue;
            if (preg_match($r['pattern'], $path, $m)) {
                $params = array_filter($m, fn($k) => !is_int($k), ARRAY_FILTER_USE_KEY);
                $result = ($r['handler'])(...array_values($params));
                if (is_array($result)) {
                    header('Content-Type: application/json');
                    echo json_encode($result, JSON_UNESCAPED_UNICODE | JSON_THROW_ON_ERROR);
                } elseif (is_string($result)) {
                    echo $result;
                }
                return;
            }
        }
        http_response_code(404);
        echo '404 Not Found';
    }
}

// private/src/Db.php
namespace App;

use PDO;

class Db
{
    public PDO $pdo;

    public function __construct(array $config)
    {
        $dsn = "mysql:host={$config['host']};dbname={$config['name']};charset=utf8mb4";
        $this->pdo = new PDO($dsn, $config['user'], $config['pass'], [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,            // prawdziwy binding
            PDO::MYSQL_ATTR_INIT_COMMAND => "SET NAMES utf8mb4 COLLATE utf8mb4_unicode_ci"
        ]);
    }

    public function one(string $sql, array $params = []): ?array
    {
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        $r = $stmt->fetch();
        return $r ?: null;
    }

    public function all(string $sql, array $params = []): array
    {
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        return $stmt->fetchAll();
    }

    public function insert(string $table, array $data): int
    {
        $cols = array_keys($data);
        $placeholders = array_map(fn($c) => ":$c", $cols);
        $sql = "INSERT INTO `$table` (`" . implode('`,`', $cols) . "`) VALUES (" . implode(',', $placeholders) . ")";
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($data);
        return (int)$this->pdo->lastInsertId();
    }

    public function update(string $table, array $data, array $where): int
    {
        $set = implode(', ', array_map(fn($c) => "`$c` = :$c", array_keys($data)));
        $cond = implode(' AND ', array_map(fn($c) => "`$c` = :w_$c", array_keys($where)));
        $sql = "UPDATE `$table` SET $set WHERE $cond";

        $params = $data;
        foreach ($where as $k => $v) $params["w_$k"] = $v;

        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        return $stmt->rowCount();
    }

    public function delete(string $table, array $where): int
    {
        $cond = implode(' AND ', array_map(fn($c) => "`$c` = :$c", array_keys($where)));
        $stmt = $this->pdo->prepare("DELETE FROM `$table` WHERE $cond");
        $stmt->execute($where);
        return $stmt->rowCount();
    }
}

        return $props;

        }
        return $props;

class BlockRegistry
{
    /** @var array<string, class-string<BlockInterface>> */
    private array $blocks = [];

    public function register(string $class): void
    {
        $this->blocks[$class::type()] = $class;
    }

    public function getClass(string $type): ?string
    {
        return $this->blocks[$type] ?? null;
    }

    public function all(): array
    {
        return array_map(fn($c) => [
            'type'  => $c::type(),
            'label' => $c::label(),
            'defaults' => $c::defaultProps(),
            'schema' => $c::schema()
        ], array_values($this->blocks));
    }

    public static function createDefault(): self
    {
        $r = new self();
        $r->register(Hero::class);
        $r->register(TextImage::class);
        $r->register(Gallery::class);
        return $r;
    }
}
<?php
// private/src/Renderer/PageRenderer.php
namespace App\Renderer;

use App\Db;
use Twig\Environment;

class PageRenderer
{
    public function __construct(
        private Db $db,
        private BlockRegistry $registry,
        private Environment $twig,
        private FileCache $cache
    ) {}

    public function renderBySlug(string $slug): string
    {
        $cacheKey = "page_slug_{$slug}";
        if ($html = $this->cache->get($cacheKey)) {
            header('X-Cache: HIT');
            return $html;
        }

        $page = $this->db->one(
            "SELECT * FROM pages WHERE slug = ? AND status = 'published'",
            [$slug]
        );
        if (!$page) {
            http_response_code(404);
            return $this->twig->render('404.twig');
        }

        $blocks = $this->db->all(
            "SELECT * FROM blocks WHERE page_id = ? AND visible = 1 ORDER BY position",
            [$page['id']]
        );

        $context = [
            'mediaUrl' => fn(int $id, string $size = 'large') => $this->mediaUrl($id, $size),
            'mediaAlt' => fn(int $id) => $this->mediaAlt($id)
        ];

        $renderedBlocks = '';
        foreach ($blocks as $blockRow) {
            $class = $this->registry->getClass($blockRow['type']);
            if (!$class) continue;

            $instance = new $class();
            $props = json_decode($blockRow['props'], true);
            $prepared = $instance->prepare($props, $context);
            $renderedBlocks .= $this->twig->render($class::template(), $prepared);
        }

        $html = $this->twig->render('layout.twig', [
            'title'      => $page['title'],
            'meta_desc'  => $page['meta_desc'],
            'content'    => $renderedBlocks
        ]);

        $this->cache->set($cacheKey, $html, 300);    // 5 min
        header('X-Cache: MISS');
        return $html;
    }

    private function mediaUrl(int $id, string $size = 'large'): string
    {
        static $cache = [];
        if (!isset($cache[$id])) {
            $row = $this->db->one("SELECT filename FROM media WHERE id = ?", [$id]);
            $cache[$id] = $row['filename'] ?? '';
        }
        if (!$cache[$id]) return '/assets/placeholder.svg';

        $base = pathinfo($cache[$id], PATHINFO_FILENAME);
        $ext  = pathinfo($cache[$id], PATHINFO_EXTENSION);
        return "/uploads/{$base}-{$size}.{$ext}";
    }

    private function mediaAlt(int $id): ?string
    {
        $row = $this->db->one("SELECT alt_text FROM media WHERE id = ?", [$id]);
        return $row['alt_text'] ?? null;
    }
}
<?php
// private/src/Renderer/FileCache.php
namespace App\Renderer;

class FileCache
{
    public function __construct(private string $dir)
    {
        if (!is_dir($dir)) mkdir($dir, 0755, true);
    }

    public function get(string $key): ?string
    {
        $f = $this->path($key);
        if (!is_file($f)) return null;
        $payload = @unserialize(file_get_contents($f));
        if (!$payload || $payload['exp'] < time()) {
            @unlink($f);
            return null;
        }
        return $payload['val'];
    }

    public function set(string $key, string $val, int $ttl): void
    {
        $tmp = $this->path($key) . '.tmp';
        file_put_contents($tmp, serialize(['exp' => time() + $ttl, 'val' => $val]), LOCK_EX);
        rename($tmp, $this->path($key));
    }

    public function flush(): void
    {
        foreach (glob($this->dir . '/*.cache') as $f) @unlink($f);
    }

    private function path(string $key): string
    {
        return $this->dir . '/' . hash('xxh3', $key) . '.cache';
    }
}

<?php
// private/src/Api/PagesController.php
namespace App\Api;

use App\Db;

class PagesController
{
    public function __construct(private Db $db) {}

    public function index(): array
    {
        return ['pages' => $this->db->all("SELECT * FROM pages ORDER BY updated_at DESC")];
    }

    public function show(string $id): array
    {
        $page = $this->db->one("SELECT * FROM pages WHERE id = ?", [(int)$id]);
        if (!$page) { http_response_code(404); return ['error' => 'Not found']; }
        $page['blocks'] = $this->db->all(
            "SELECT id, type, props, position, visible FROM blocks WHERE page_id = ? ORDER BY position",
            [(int)$id]
        );
        // Decode JSON props
        foreach ($page['blocks'] as &$b) $b['props'] = json_decode($b['props'], true);
        return $page;
    }

    public function create(): array
    {
        $input = json_decode(file_get_contents('php://input'), true);
        if (empty($input['slug']) || empty($input['title'])) {
            http_response_code(422);
            return ['error' => 'slug and title required'];
        }

        $id = $this->db->insert('pages', [
            'slug'      => preg_replace('/[^a-z0-9-]/', '', strtolower($input['slug'])),
            'title'     => mb_substr($input['title'], 0, 255),
            'meta_desc' => mb_substr($input['meta_desc'] ?? '', 0, 160),
            'status'    => 'draft'
        ]);
        http_response_code(201);

namespace App\Api;

use App\Db;
use App\Renderer\BlockRegistry;

class BlocksController
{
    public function __construct(private Db $db, private BlockRegistry $registry) {}

    public function create(): array
    {
        $input = json_decode(file_get_contents('php://input'), true);
        $type = $input['type'] ?? '';
        $pageId = (int)($input['page_id'] ?? 0);

        $class = $this->registry->getClass($type);
        if (!$class || !$pageId) {
            http_response_code(422);
            return ['error' => 'Invalid type or page_id'];
        }

        $nextPos = (int)$this->db->pdo->query("SELECT COALESCE(MAX(position), 0) + 1 FROM blocks WHERE page_id = $pageId")->fetchColumn();

        $id = $this->db->insert('blocks', [
            'page_id'  => $pageId,
            'type'     => $type,
            'props'    => json_encode($class::defaultProps(), JSON_UNESCAPED_UNICODE),
            'position' => $nextPos
        ]);

        return [
            'id'       => $id,
            'type'     => $type,
            'props'    => $class::defaultProps(),
            'position' => $nextPos,
            'visible'  => 1
        ];
    }

    public function update(string $id): array
    {
        $input = json_decode(file_get_contents('php://input'), true);
        $block = $this->db->one("SELECT * FROM blocks WHERE id = ?", [(int)$id]);
        if (!$block) { http_response_code(404); return ['error' => 'Not found']; }

        $update = [];
        if (isset($input['props']) && is_array($input['props'])) {
            $class = $this->registry->getClass($block['type']);
            // Walidacja propsów wg schema (uproszczona)
            $sanitized = $this->sanitizeProps($input['props'], $class::schema());
            $update['props'] = json_encode($sanitized, JSON_UNESCAPED_UNICODE);
        }
        if (isset($input['visible'])) $update['visible'] = $input['visible'] ? 1 : 0;

        if ($update) $this->db->update('blocks', $update, ['id' => (int)$id]);
        return ['ok' => true];
    }

    public function destroy(string $id): array
    {
        $this->db->delete('blocks', ['id' => (int)$id]);
        return ['ok' => true];
    }

    public function reorder(): array
    {
        $input = json_decode(file_get_contents('php://input'), true);
        $ids = $input['ids'] ?? [];

        $this->db->pdo->beginTransaction();
        try {
            foreach ($ids as $pos => $id) {
                $this->db->update('blocks', ['position' => $pos + 1], ['id' => (int)$id]);
            }
            $this->db->pdo->commit();
        } catch (\Throwable $e) {
            $this->db->pdo->rollBack();
            http_response_code(500);
            return ['error' => 'Reorder failed'];
        }
        return ['ok' => true];
    }

    private function sanitizeProps(array $input, array $schema): array
    {
        $out = [];
        foreach ($schema as $key => $rules) {
            if (!array_key_exists($key, $input)) continue;
            $val = $input[$key];

            switch ($rules['type']) {
                case 'string':
                case 'html':
                    if (!is_string($val)) continue 2;
                    if (isset($rules['max'])) $val = mb_substr($val, 0, $rules['max']);
                    break;
                case 'integer':
                    $val = (int)$val;
                    break;
                case 'enum':
                    if (!in_array($val, $rules['values'], true)) continue 2;
                    break;
                case 'array':
                    if (!is_array($val)) continue 2;
                    if (($rules['of'] ?? null) === 'integer') $val = array_map('intval', $val);
                    break;
            }
            $out[$key] = $val;
        }
        return $out;
    }
}

namespace App;

class Auth
{
    public function __construct(private Db $db) {}

    public function login(array $input): array
    {
        $email = $input['email'] ?? '';
        $password = $input['password'] ?? '';
        $csrf = $input['_csrf'] ?? '';

        if (!hash_equals($_SESSION['csrf'] ?? '', $csrf)) {
            http_response_code(419);
            return ['error' => 'CSRF mismatch'];
        }

        // Rate limit per IP (uproszczone — w prod: tabela)
        $key = 'login_' . $_SERVER['REMOTE_ADDR'];
        $_SESSION[$key] = ($_SESSION[$key] ?? 0) + 1;
        if ($_SESSION[$key] > 5) {
            http_response_code(429);
            return ['error' => 'Too many attempts'];
        }

        $user = $this->db->one("SELECT * FROM users WHERE email = ?", [$email]);
        if (!$user || !password_verify($password, $user['password_hash'])) {
            return ['error' => 'Invalid credentials'];
        }

        // Re-hash jeśli parametry się zmieniły
        if (password_needs_rehash($user['password_hash'], PASSWORD_ARGON2ID)) {
            $this->db->update('users',
                ['password_hash' => password_hash($password, PASSWORD_ARGON2ID)],
                ['id' => $user['id']]);
        }

        session_regenerate_id(true);
        $_SESSION['user_id'] = $user['id'];
        $_SESSION['user_role'] = $user['role'];
        unset($_SESSION[$key]);

        header('Location: /admin');
        return ['ok' => true];
    }

    public function logout(): array
    {
        $_SESSION = [];
        session_destroy();
        header('Location: /admin/login');
        return ['ok' => true];
    }

    public static function require(Db $db): void
    {
        if (empty($_SESSION['user_id'])) {
            header('Location: /admin/login');
            exit;
        }
        // Refresh ostatnia aktywność (anty-hijack)
        if (!isset($_SESSION['last_activity'])) {
            $_SESSION['last_activity'] = time();
        } elseif (time() - $_SESSION['last_activity'] > 1800) {     // 30 min idle
            $_SESSION = [];
            session_destroy();
            header('Location: /admin/login');
            exit;
        }
        $_SESSION['last_activity'] = time();
    }

<?php
// public_html/install.php — UŻYWAĆ TYLKO RAZ, POTEM USUNĄĆ!
declare(strict_types=1);

if (file_exists(__DIR__ . '/../private/config/config.php')) {
    die('Już zainstalowany. Usuń install.php.');
}

$step = $_POST['step'] ?? '1';

if ($step === '1') {
    // Formularz konfiguracji
    ?>
    <!DOCTYPE html>
    <html lang="pl"><head><meta charset="UTF-8"><title>Instalator</title></head>
    <body style="font-family:sans-serif;max-width:500px;margin:50px auto;padding:20px">
        <h1>Instalator Buildera</h1>
        <form method="post">
            <input type="hidden" name="step" value="2">
            <h3>Baza danych</h3>
            <p><label>Host:<br><input name="db_host" value="localhost" required></label></p>
            <p><label>Nazwa bazy:<br><input name="db_name" required></label></p>
            <p><label>Użytkownik:<br><input name="db_user" required></label></p>
            <p><label>Hasło:<br><input type="password" name="db_pass"></label></p>
            <h3>Konto admina</h3>
            <p><label>Email:<br><input type="email" name="admin_email" required></label></p>
            <p><label>Hasło (min. 12 znaków):<br><input type="password" name="admin_pass" minlength="12" required></label></p>
            <button type="submit">Zainstaluj</button>
        </form>
    </body></html>
    <?php
    exit;
}

// Step 2: instalacja
try {
    $pdo = new PDO(
        "mysql:host={$_POST['db_host']};dbname={$_POST['db_name']};charset=utf8mb4",
        $_POST['db_user'], $_POST['db_pass'],
        [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
    );

    // Wykonaj migrację
    $sql = file_get_contents(__DIR__ . '/../private/migrations/001_init.sql');
    $pdo->exec($sql);

    // Stwórz admina
    $stmt = $pdo->prepare("INSERT INTO users (email, password_hash, role) VALUES (?, ?, 'admin')");
    $stmt->execute([
        $_POST['admin_email'],
        password_hash($_POST['admin_pass'], PASSWORD_ARGON2ID)
    ]);

    // Zapisz config
    $configDir = __DIR__ . '/../private/config';
    if (!is_dir($configDir)) mkdir($configDir, 0700, true);

    $config = "<?php\nreturn " . var_export([
        'debug' => false,
        'db' => [
            'host' => $_POST['db_host'],
            'name' => $_POST['db_name'],
            'user' => $_POST['db_user'],
            'pass' => $_POST['db_pass']
        ],
        'session_lifetime' => 1800
    ], true) . ";\n";

    file_put_contents("$configDir/config.php", $config);
    chmod("$configDir/config.php", 0600);

    // Stwórz cache & uploads
    @mkdir(__DIR__ . '/../private/cache', 0755, true);
    @mkdir(__DIR__ . '/uploads', 0755, true);

    // USUŃ INSTALATORA
    @unlink(__FILE__);

    echo "<h1>✅ Zainstalowano!</h1>";
    echo "<p><a href='/admin/login'>Zaloguj się</a></p>";

} catch (Throwable $e) {
    echo "<h1>❌ Błąd</h1><pre>" . htmlspecialchars($e->getMessage()) . "</pre>";
}
⚠️ WAŻNE: Instalator SAM SIĘ USUWA po sukcesie (linia @unlink(__FILE__)). Jeśli to nie zadziała (np. brak uprawnień) — USUŃ go ręcznie przez FTP. Inaczej ktoś może go uruchomić i nadpisać konfigurację.
