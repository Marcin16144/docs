🌍 Sieci — debugging i diagnostyka

tcpdump, ss, dig, openssl, curl — komendy, które uratują życie w incydencie produkcyjnym.

1. Czy port otwarty? Czy działa?

# Lokalnie — co nasłuchuje
ss -tulnp                             # wszystkie sockety + proces
ss -tnp state established             # aktywne TCP
sudo lsof -i :443                       # co używa portu 443

# Z innej maszyny — test TCP
nc -zv example.com 443                  # otwarty / zamknięty
nmap -p 80,443,8080 example.com         # skan kilku portów

# Test latencji
ping -c 5 example.com
mtr --report --report-cycles 20 example.com    # ping + traceroute combo

# Traceroute
traceroute -T -p 443 example.com         # TCP traceroute (omija ICMP block)

2. tcpdump — co przelatuje w sieci

# HTTP traffic na porcie 80
sudo tcpdump -i any -A -s 0 'tcp port 80 and (((ip[2:2] - ((ip[0]&0xf)<<2)) - ((tcp[12]&0xf0)>>2)) != 0)'

# DNS queries (do/od konkretnego IP)
sudo tcpdump -i any -n 'udp port 53'
sudo tcpdump -i any -n host 8.8.8.8

# Save do pliku + analiza w Wireshark
sudo tcpdump -i any -w capture.pcap port 443
# Otwórz w Wireshark: File → Open

# TLS handshake (client hello, certificate)
sudo tcpdump -i any -n -X port 443 and 'tcp[((tcp[12:1] & 0xf0) >> 2):1] = 0x16'

3. DNS — dig i diagnostyka

# Wszystkie rekordy
dig example.com ANY +noall +answer

# Konkretny typ
dig example.com A           # IPv4
dig example.com AAAA        # IPv6
dig example.com MX          # mail
dig example.com TXT         # SPF, DKIM, DMARC
dig example.com NS          # nameservery
dig example.com CAA         # kto może wystawiać certy

# Z konkretnego resolvera (omija lokalny cache)
dig @8.8.8.8 example.com
dig @1.1.1.1 example.com +trace          # pełny łańcuch DNS

# DNSSEC validation
dig example.com +dnssec +multi

# Reverse DNS
dig -x 8.8.8.8
nslookup 8.8.8.8

4. TLS — debugowanie certyfikatów

# Co serwer wystawia (cały łańcuch)
openssl s_client -connect example.com:443 -servername example.com -showcerts < /dev/null

# Tylko dni do wygaśnięcia
echo | openssl s_client -connect example.com:443 -servername example.com 2>/dev/null \
    | openssl x509 -noout -dates

# Wszystkie domeny z SAN (multi-domain cert)
echo | openssl s_client -connect example.com:443 -servername example.com 2>/dev/null \
    | openssl x509 -noout -ext subjectAltName

# Test konkretnego protokołu
openssl s_client -connect example.com:443 -tls1_3
openssl s_client -connect example.com:443 -tls1_2

# SSL Labs CLI — pełny scan jak w przeglądarce
docker run --rm jumanjiman/ssllabs-scan -grade example.com

# Generuj self-signed cert do testów
openssl req -x509 -newkey rsa:4096 -nodes -keyout key.pem -out cert.pem \
    -days 365 -subj "/CN=localhost" \
    -addext "subjectAltName=DNS:localhost,IP:127.0.0.1"

5. curl — full debug

# Pokaż wszystko (request + response headers)
curl -vL https://example.com

# Czas każdej fazy: DNS, connect, TLS, transfer
curl -w "@-" -o /dev/null -s https://example.com <<'EOF'
    dns:        %{time_namelookup}s
    connect:    %{time_connect}s
    tls:        %{time_appconnect}s
    pretransfer:%{time_pretransfer}s
    starttransfer: %{time_starttransfer}s
    ---
    total:      %{time_total}s
    size:       %{size_download} bytes
    http_code:  %{http_code}
EOF

# Wymuś IPv4 / IPv6
curl -4 https://example.com
curl -6 https://example.com

# Custom DNS (omiń DNS, sprawdź konkretny serwer)
curl --resolve example.com:443:203.0.113.42 https://example.com

# POST z JSON
curl -X POST https://api.example.com/users \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer $TOKEN" \
    -d '{"name":"Anna","email":"a@example.com"}'

# Upload pliku
curl -F "file=@photo.jpg" -F "caption=Vacation" https://api.example.com/upload

7. Routing — sprawdź jak ruch idzie

# Tablica routingu
ip route show
ip route get 8.8.8.8                  # którędy ruch do tego IP?

# Interfejsy
ip addr show
ip link show

# Dodaj statyczną trasę
sudo ip route add 10.0.0.0/24 via 192.168.1.1 dev eth0

# Co jest na tym samym L2 (ARP table)
ip neigh show

# Test MTU (jumbo frames)
ping -M do -s 1472 8.8.8.8             # 1472 + 28 ICMP = 1500 MTU
