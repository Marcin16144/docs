🔄 CI/CD — GitHub Actions workflow

Build, test, security scan, deploy. Cache, matrix, secrets, environments — produkcyjny pipeline.

1. Pełny pipeline Node.js / TypeScript

# .github/workflows/ci.yml
name: CI/CD

on:
  push:
    branches: [main]
  pull_request:

concurrency:    # anuluj poprzednie runs tej samej gałęzi
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  test:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        node: [20, 22]
    services:
      postgres:
        image: postgres:17-alpine
        env: { POSTGRES_PASSWORD: test }
        ports: ["5432:5432"]
        options: >-
          --health-cmd "pg_isready -U postgres"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5

    steps:
      - uses: actions/checkout@v4

      - name: Setup Node ${{ matrix.node }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node }}
          cache: 'npm'

      - run: npm ci
      - run: npm run lint
      - run: npm run typecheck
      - name: Test
        env:
          DATABASE_URL: postgres://postgres:test@localhost:5432/postgres
        run: npm test -- --coverage

      - name: Upload coverage
        uses: codecov/codecov-action@v4
        with:
          token: ${{ secrets.CODECOV_TOKEN }}

  security:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: NPM audit
        run: npm audit --audit-level=high

      - name: Trivy filesystem scan
        uses: aquasecurity/trivy-action@master
        with:
          scan-type: 'fs'
          severity: 'CRITICAL,HIGH'
          exit-code: '1'

      - name: Detect secrets
        uses: gitleaks/gitleaks-action@v2
        env: { GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }} }

  build-and-push:
    needs: [test, security]
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    permissions:
      contents: read
      packages: write
      id-token: write   # dla OIDC z cloud
    steps:
      - uses: actions/checkout@v4

      - name: Setup buildx
        uses: docker/setup-buildx-action@v3

      - name: Login to GHCR
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Metadata (tags, labels)
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ghcr.io/${{ github.repository }}
          tags: |
            type=ref,event=branch
            type=sha,prefix=,format=short
            type=raw,value=latest,enable={{is_default_branch}}

      - name: Build & push
        uses: docker/build-push-action@v5
        with:
          context: .
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          cache-from: type=gha
          cache-to: type=gha,mode=max
          provenance: true
          sbom: true

  deploy:
    needs: build-and-push
    runs-on: ubuntu-latest
    environment:
      name: production
      url: https://api.example.com
    steps:
      - name: Configure AWS via OIDC (bez secrets!)
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::123456789012:role/github-deploy
          aws-region: eu-central-1

      - name: Deploy to ECS
        run: |
          aws ecs update-service --cluster prod \
            --service api --force-new-deployment

2. Cache — przyspieszanie buildów

# Node.js
- uses: actions/setup-node@v4
  with: { node-version: 22, cache: 'npm' }

# Python (uv)
- uses: actions/setup-python@v5
  with: { python-version: '3.13', cache: 'pip' }
- run: pip install uv && uv sync --frozen

# Rust
- uses: Swatinem/rust-cache@v2

# .NET
- uses: actions/setup-dotnet@v4
  with: { dotnet-version: '9.0.x' }
- uses: actions/cache@v4
  with:
    path: ~/.nuget/packages
    key: ${{ runner.os }}-nuget-${{ hashFiles('**/packages.lock.json') }}

# Docker layers (BuildKit cache)
- uses: docker/build-push-action@v5
  with:
    cache-from: type=gha
    cache-to: type=gha,mode=max

3. Environments + manual approval

# GitHub repo → Settings → Environments → "production"
# - Required reviewers: [twoj-team]
# - Wait timer: 5 min
# - Branch restriction: main

jobs:
  deploy-prod:
    environment:
      name: production
      url: ${{ steps.deploy.outputs.url }}
    steps:
      - id: deploy
        run: |
          echo "url=https://api.example.com" >> $GITHUB_OUTPUT
          ./deploy.sh prod

4. Reusable workflow (call from many repos)

# .github/workflows/reusable-test.yml
name: Reusable Test Workflow
on:
  workflow_call:
    inputs:
      node-version:
        type: string
        default: "22"
    secrets:
      CODECOV_TOKEN: { required: true }

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: ${{ inputs.node-version }} }
      - run: npm ci && npm test

# .github/workflows/ci.yml — w innym repo
jobs:
  call-tests:
    uses: my-org/shared-workflows/.github/workflows/reusable-test.yml@v1
    with: { node-version: "20" }
    secrets: inherit

5. Quality gates — blokuj merge przy regresji

# Coverage musi nie spaść poniżej 80%
- name: Coverage gate
  run: |
    coverage=$(jq '.total.lines.pct' coverage/summary.json)
    if (( $(echo "$coverage < 80" | bc -l) )); then
      echo "::error::Coverage $coverage% below 80%"
      exit 1
    fi

# Performance regression (Lighthouse CI)
- uses: treosh/lighthouse-ci-action@v11
  with:
    urls: |
      https://staging.example.com
    configPath: './.lighthouserc.json'
    temporaryPublicStorage: true
    budgetPath: './.lh-budget.json'

# Bundle size check
- uses: andresz1/size-limit-action@v1
  with:
    github_token: ${{ secrets.GITHUB_TOKEN }}

6. Secrets — bezpiecznie

# ❌ ŹLE: hardcoded
env:
  API_KEY: "sk_live_abc123"

# ✅ DOBRZE: secret z UI / org-level
env:
  API_KEY: ${{ secrets.API_KEY }}

# ✅ JESZCZE LEPIEJ: OIDC do AWS/GCP/Azure (bez secrets w ogóle)
- uses: aws-actions/configure-aws-credentials@v4
  with:
    role-to-assume: arn:aws:iam::ACCOUNT:role/github-deploy
    aws-region: eu-central-1

# ✅ Multi-environment secrets
jobs:
  deploy:
    environment: ${{ inputs.env }}  # staging | production
    steps:
      - env: { DB_URL: ${{ secrets.DB_URL }} }  # secret per env
        run: ./deploy.sh

7. Komendy debug, gdy CI się sypie

# Re-run failed jobs tylko
gh run rerun <run-id> --failed

# Watch live
gh run watch

# Pobierz logi
gh run view <run-id> --log

# SSH na runnera (action-tmate)
- name: Debug via SSH (5 min)
  if: failure()
  uses: mxschmitt/action-tmate@v3
  with: { limit-access-to-actor: true }
# Logi wyświetlą SSH command — wklej w terminal

# Lokalna replikacja workflow (act)
brew install act    # lub: choco install act-cli
act -j test         # uruchom job "test" lokalnie

8. Najczęstsze błędy

SymptomFix
"Permission denied" przy push do GHCRDodaj permissions: { packages: write }
Test passe lokalnie, fail w CISprawdź timezone, locale, file paths (case sensitive!)
Cache nie działaKlucz cache musi być deterministyczny (hash lockfile)
Build trwa 30 minWłącz BuildKit cache, paraleluj jobs, użyj larger runner
Random flaky failuresConcurrency in jobs, race conditions w testach, brak wait
"Out of disk space"sudo rm -rf /usr/share/dotnet /opt/ghc na początku (GHA runner)
