1. systemd unit dla Node/Python/Go aplikacji

# /etc/systemd/system/myapi.service
[Unit]
Description=My API
After=network.target postgresql.service
Requires=postgresql.service

[Service]
Type=simple
User=app
Group=app
WorkingDirectory=/opt/myapi
EnvironmentFile=/etc/myapi.env
ExecStart=/usr/bin/node /opt/myapi/dist/server.js
Restart=on-failure
RestartSec=5s
StandardOutput=journal
StandardError=journal
SyslogIdentifier=myapi

# Hardening
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=true
ReadWritePaths=/opt/myapi/data
ProtectKernelTunables=true
ProtectKernelModules=true
ProtectControlGroups=true
RestrictNamespaces=true
LockPersonality=true
MemoryDenyWriteExecute=true
RestrictSUIDSGID=true
SystemCallArchitectures=native

[Install]
WantedBy=multi-user.target
# Aktywacja
sudo systemctl daemon-reload
sudo systemctl enable --now myapi.service

# Logi
journalctl -u myapi -f --since "10 minutes ago"
journalctl -u myapi -p err --since today    # tylko błędy

# Status
systemctl status myapi
systemctl show myapi --property=MemoryCurrent,CPUUsageNSec

2. Bash z error handling (production-grade)

#!/usr/bin/env bash
# backup-db.sh — backup PostgreSQL z rotacją
set -euo pipefail      # exit on error, undefined var, pipe failure
IFS=$'\n\t'             # bezpieczny IFS (nie wyłapuje spacji)

readonly BACKUP_DIR="/var/backups/postgres"
readonly RETENTION_DAYS=14
readonly DB_NAME="production"
readonly S3_BUCKET="s3://my-backups/db"

log() { echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] $*" >&2; }
die() { log "ERROR: $*"; exit 1; }

cleanup() {
    local rc=$?
    if [[ -n "${TMPFILE:-}" && -f "$TMPFILE" ]]; then
        rm -f "$TMPFILE"
    fi
    return $rc
}
trap cleanup EXIT

main() {
    mkdir -p "$BACKUP_DIR"
    local timestamp; timestamp=$(date -u +%Y%m%d-%H%M%S)
    local file="$BACKUP_DIR/${DB_NAME}-${timestamp}.sql.gz"
    TMPFILE=$(mktemp)

    log "Backing up $DB_NAME → $file"
    pg_dump "$DB_NAME" | gzip -9 > "$TMPFILE" || die "pg_dump failed"

    # Atomowo przenieś (rename = atomic na tym samym fs)
    mv "$TMPFILE" "$file"
    unset TMPFILE

    # Wgraj do S3
    aws s3 cp "$file" "$S3_BUCKET/" --storage-class STANDARD_IA \
        || log "WARN: S3 upload failed (local backup OK)"

    # Rotacja lokalnie
    find "$BACKUP_DIR" -name "${DB_NAME}-*.sql.gz" -mtime +$RETENTION_DAYS -delete

    log "Done. Size: $(du -h "$file" | cut -f1)"
}

main "$@"

4. nftables (zastępca iptables w 2026)

# /etc/nftables.conf
flush ruleset

table inet filter {
    chain input {
        type filter hook input priority filter; policy drop;

        # Pozwól loopback, established
        iif lo accept
        ct state established,related accept
        ct state invalid drop

        # ICMP (ping) — limit
        ip protocol icmp icmp type echo-request limit rate 5/second accept
        ip6 nexthdr icmpv6 accept

        # SSH na port 2222 + brute force protection
        tcp dport 2222 ct state new limit rate 3/minute accept
        tcp dport 2222 ct state new drop

        # HTTP/HTTPS
        tcp dport { 80, 443 } accept

        # Drop reszta + log
        log prefix "nft drop: " limit rate 5/minute
        counter
    }
    chain forward { type filter hook forward priority filter; policy drop; }
    chain output  { type filter hook output priority filter; policy accept; }
}
# Włącz
sudo systemctl enable --now nftables
sudo nft -f /etc/nftables.conf
sudo nft list ruleset

# Tymczasowo blokuj IP
sudo nft add rule inet filter input ip saddr 203.0.113.42 drop

5. Monitoring CPU/RAM/disk z bash

# Top procesy po RAM
ps aux --sort=-%mem | head -10

# Top procesy po CPU
ps aux --sort=-%cpu | head -10

# Wykorzystanie disk z procentami
df -hT --exclude-type=tmpfs --exclude-type=devtmpfs

# I/O per proces (apt install iotop)
sudo iotop -oPa

# Co zajmuje najwięcej miejsca w katalogu (uniwersalny one-liner)
du -h --max-depth=1 /var | sort -hr | head -10

# Network connections per process
sudo ss -tunlp

# Live monitoring (najlepszy w 2026: btop)

6. Logrotate

# /etc/logrotate.d/myapi
/var/log/myapi/*.log {
    daily
    rotate 14
    compress
    delaycompress
    missingok
    notifempty
    create 640 app app
    sharedscripts
    postrotate
        systemctl reload myapi.service > /dev/null 2>&1 || true
    endscript
}
