1. Vitest — unit + integration (TypeScript)

// npm i -D vitest @vitest/coverage-v8
// vitest.config.ts
import { defineConfig } from 'vitest/config';
export default defineConfig({
    test: {
        environment: 'node',
        coverage: { reporter: ['text', 'html', 'lcov'], lines: 80 }
    }
});

// src/cart.ts
export class Cart {
    items: { id: string, qty: number, price: number }[] = [];
    add(id: string, price: number) {
        const existing = this.items.find(i => i.id === id);
        existing ? existing.qty++ : this.items.push({ id, qty: 1, price });
    }
    get total() { return this.items.reduce((s, i) => s + i.qty * i.price, 0); }
}

// src/cart.test.ts
import { describe, it, expect, beforeEach, vi } from 'vitest';
import { Cart } from './cart';

describe('Cart', () => {
    let cart: Cart;
    beforeEach(() => { cart = new Cart(); });

    it('dodaje pierwszy produkt', () => {
        cart.add('a', 10);
        expect(cart.items).toHaveLength(1);
        expect(cart.items[0]).toMatchObject({ id: 'a', qty: 1, price: 10 });
    });

    it('zwiększa ilość przy ponownym dodaniu', () => {
        cart.add('a', 10);
        cart.add('a', 10);
        expect(cart.items[0].qty).toBe(2);
    });

    it('liczy total', () => {
        cart.add('a', 10);
        cart.add('b', 5);
        cart.add('a', 10);
        expect(cart.total).toBe(25);
    });
});

// Mockowanie modułu

    saveOrder: vi.fn().mockResolvedValue({ id: 'mock-id' })
}));

async def test_create_user(client, sample_user):

3. xUnit (.NET 9)

// dotnet new xunit -n MyApi.Tests
// dotnet add reference ../MyApi/MyApi.csproj
// dotnet add package FluentAssertions
// dotnet add package Moq

using FluentAssertions;
using Moq;
using Xunit;

public class CartServiceTests
{
    [Fact]
    public void Add_NewItem_IncreasesItemCount()
    {
        var cart = new Cart();
        cart.Add("a", 10);
        cart.Items.Should().HaveCount(1);
        cart.Items[0].Qty.Should().Be(1);
    }

    [Theory]
    [InlineData(10, 5, 50)]
    [InlineData(0, 100, 0)]
    [InlineData(1.5, 2, 3)]
    public void Total_CalculatesCorrectly(decimal price, int qty, decimal expected)
    {
        var cart = new Cart();
        for (var i = 0; i < qty; i++) cart.Add("a", price);
        cart.Total.Should().Be(expected);
    }

    [Fact]
    public async Task PlaceOrder_CallsRepository()
    {
        var repoMock = new Mock<IOrderRepository>();
        repoMock.Setup(r => r.SaveAsync(It.IsAny<Order>()))
            .Returns(Task.CompletedTask);

        var service = new OrderService(repoMock.Object);
        await service.PlaceOrderAsync(new Order());

        repoMock.Verify(r => r.SaveAsync(It.IsAny<Order>()), Times.Once);
    }
}

4. Testcontainers — prawdziwa baza w testach

// npm i -D testcontainers @testcontainers/postgresql
import { describe, it, beforeAll, afterAll, expect } from 'vitest';
import { PostgreSqlContainer, StartedPostgreSqlContainer } from '@testcontainers/postgresql';
import { Client } from 'pg';

describe('UserRepository integration', () => {
    let container: StartedPostgreSqlContainer;
    let db: Client;

    beforeAll(async () => {
        container = await new PostgreSqlContainer('postgres:17-alpine')
            .withDatabase('testdb')
            .withUsername('test')
            .withPassword('test')
            .start();

        db = new Client({ connectionString: container.getConnectionUri() });
        await db.connect();
        await db.query('CREATE TABLE users (id SERIAL, email TEXT UNIQUE)');
    }, 60_000);

    afterAll(async () => {
        await db.end();
        await container.stop();
    });

    it('wstawia użytkownika', async () => {
        await db.query('INSERT INTO users (email) VALUES ($1)', ['a@b.com']);
        const { rows } = await db.query('SELECT * FROM users');
        expect(rows).toHaveLength(1);
    });
});
💡 Testcontainers daje prawdziwą bazę, Redis, Kafka, Mailhog w <5s. Wymaga Dockera. Mocki SQL kłamią — Testcontainers nie.
