🌐 GraphQL — schema + resolvery w 3 językach

Apollo Server (Node), Hot Chocolate (.NET 9), Strawberry (Python) — to samo schema, różne implementacje.

Wspólne schema (SDL)

type Product {
  id: ID!
  name: String!
  price: Float!
  stock: Int!
  category: Category!
}

type Category {
  id: ID!
  name: String!
  products: [Product!]!
}

type Query {
  products(first: Int = 20, after: String): ProductConnection!
  product(id: ID!): Product
}

type Mutation {
  createProduct(input: ProductInput!): Product!
  updateStock(id: ID!, delta: Int!): Product!
}

input ProductInput {
  name: String!
  price: Float!
  stock: Int = 0
  categoryId: ID!
}

type ProductConnection {
  edges: [ProductEdge!]!
  pageInfo: PageInfo!
}
type ProductEdge { cursor: String!, node: Product! }
type PageInfo { hasNextPage: Boolean!, endCursor: String }

1. Apollo Server (Node.js + TypeScript)

// npm i @apollo/server graphql dataloader
import { ApolloServer } from '@apollo/server';
import { startStandaloneServer } from '@apollo/server/standalone';
import DataLoader from 'dataloader';

const typeDefs = `#graphql
  type Product { id: ID!, name: String!, price: Float!, stock: Int!, category: Category! }
  type Category { id: ID!, name: String!, products: [Product!]! }
  type Query { products: [Product!]!, product(id: ID!): Product }
  type Mutation { createProduct(name: String!, price: Float!, categoryId: ID!): Product! }
`;

// DataLoader rozwiązuje N+1 problem przez batching
const categoryLoader = new DataLoader<string, any>(async (ids) => {
    const rows = await db.query('SELECT * FROM categories WHERE id = ANY($1)', [ids]);
    const byId = new Map(rows.map(r => [r.id, r]));
    return ids.map(id => byId.get(id));
});

const resolvers = {
    Query: {
        products: () => db.query('SELECT * FROM products'),
        product: (_: any, { id }: { id: string }) =>
            db.queryOne('SELECT * FROM products WHERE id = $1', [id])
    },
    Product: {
        category: (p: { categoryId: string }) => categoryLoader.load(p.categoryId)
    },
    Mutation: {
        createProduct: async (_: any, args: any) => {
            if (args.price <= 0) throw new Error('Invalid price');
            return db.queryOne(
                'INSERT INTO products (name, price, category_id) VALUES ($1, $2, $3) RETURNING *',
                [args.name, args.price, args.categoryId]
            );
        }
    }
};

const server = new ApolloServer({ typeDefs, resolvers });
const { url } = await startStandaloneServer(server, { listen: { port: 4000 } });
console.log(`🚀 Ready at ${url}`);
⚠️ Bez DataLoader pobranie 100 produktów ze ich kategoriami = 101 zapytań SQL. Z DataLoader = 2.

2. Hot Chocolate (.NET 9)

// dotnet add package HotChocolate.AspNetCore
var builder = WebApplication.CreateBuilder(args);
builder.Services
    .AddGraphQLServer()
    .AddQueryType<Query>()
    .AddMutationType<Mutation>()
    .AddProjections()
    .AddFiltering()
    .AddSorting();

var app = builder.Build();
app.MapGraphQL();
app.Run();

public class Query
{
    [UseProjection, UseFiltering, UseSorting]
    public IQueryable<Product> GetProducts([Service] AppDbContext db) => db.Products;

    public async Task<Product?> GetProductAsync(Guid id, [Service] AppDbContext db) =>
        await db.Products.FindAsync(id);
}

public class Mutation
{
    public async Task<Product> CreateProductAsync(
        ProductInput input, [Service] AppDbContext db)
    {
        if (input.Price <= 0) throw new GraphQLException("Invalid price");
        var product = new Product { Id = Guid.NewGuid(),
                                   Name = input.Name, Price = input.Price };
        db.Products.Add(product);
        await db.SaveChangesAsync();
        return product;
    }
}

public record ProductInput(string Name, decimal Price, Guid CategoryId);
💡 [UseProjection] automatycznie generuje SELECT tylko żądanych pól — żadnego N+1.

3. Strawberry (Python + FastAPI)

# pip install strawberry-graphql[fastapi] sqlalchemy
import strawberry
from strawberry.fastapi import GraphQLRouter
from fastapi import FastAPI
from uuid import UUID

@strawberry.type
class Category:
    id: UUID
    name: str

@strawberry.type
class Product:
    id: UUID
    name: str
    price: float
    stock: int

    @strawberry.field
    async def category(self) -> Category:
        return await category_loader.load(self.category_id)

@strawberry.type
class Query:
    @strawberry.field
    async def products(self, first: int = 20) -> list[Product]:
        return await db.fetch_all("SELECT * FROM products LIMIT $1", first)

    @strawberry.field
    async def product(self, id: UUID) -> Product | None:
        return await db.fetch_one("SELECT * FROM products WHERE id = $1", id)

@strawberry.input
class ProductInput:
    name: str
    price: float
    stock: int = 0
    category_id: UUID

@strawberry.type
class Mutation:
    @strawberry.mutation
    async def create_product(self, input: ProductInput) -> Product:
        if input.price <= 0:
            raise Exception("Invalid price")
        row = await db.fetch_one(
            "INSERT INTO products (name, price, stock, category_id) "
            "VALUES ($1, $2, $3, $4) RETURNING *",
            input.name, input.price, input.stock, input.category_id)
        return Product(**row)

schema = strawberry.Schema(query=Query, mutation=Mutation)
graphql_app = GraphQLRouter(schema)
app = FastAPI()
app.include_router(graphql_app, prefix="/graphql")

5. Bezpieczeństwo: query depth + complexity

// Apollo: blokuj malicious queries
import depthLimit from 'graphql-depth-limit';
import { createComplexityRule } from 'graphql-validation-complexity';

const server = new ApolloServer({
    typeDefs, resolvers,
    validationRules: [
        depthLimit(7),                                    // max zagłębienie
        createComplexityRule({ maximumComplexity: 1000 })   // max "punkty"
    ]
});
