1. CTE — schemat zapytania krokami

-- Top 3 produkty w każdej kategorii po sprzedaży w 2026 Q1
WITH sales_q1 AS (
    SELECT
        p.id, p.name, p.category_id,
        SUM(oi.quantity * oi.unit_price) AS revenue
    FROM order_items oi
    JOIN orders o    ON o.id = oi.order_id
    JOIN products p  ON p.id = oi.product_id
    WHERE o.placed_at >= '2026-01-01'
      AND o.placed_at < '2026-04-01'
    GROUP BY p.id, p.name, p.category_id
),
ranked AS (
    SELECT *,
           ROW_NUMBER() OVER (PARTITION BY category_id ORDER BY revenue DESC) AS rn
    FROM sales_q1
)
SELECT c.name AS category, r.name, r.revenue, r.rn
FROM ranked r
JOIN categories c ON c.id = r.category_id
WHERE r.rn <= 3
ORDER BY c.name, r.rn;

2. Window functions — analytics bez self-join

-- Running total + porównanie z poprzednim dniem
SELECT
    date,
    revenue,
    SUM(revenue) OVER (ORDER BY date) AS cumulative,
    LAG(revenue, 1) OVER (ORDER BY date) AS prev_day,
    revenue - LAG(revenue, 1) OVER (ORDER BY date) AS day_change,
    AVG(revenue) OVER (ORDER BY date ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS ma_7d
FROM daily_sales
ORDER BY date DESC
LIMIT 30;

-- Top 5 największych klientów w każdym kraju (DENSE_RANK obsługuje remisy)
SELECT country, customer_name, total_spent
FROM (
    SELECT c.country, c.name AS customer_name,
           SUM(o.total) AS total_spent,
           DENSE_RANK() OVER (PARTITION BY c.country ORDER BY SUM(o.total) DESC) AS rk
    FROM customers c JOIN orders o ON o.customer_id = c.id
    GROUP BY c.id
) t
WHERE rk <= 5;

3. Indeksy — kiedy które

-- B-tree: domyślny, dla =, <, >, ORDER BY, range scan
CREATE INDEX idx_orders_customer_date ON orders(customer_id, placed_at DESC);

-- Hash: tylko równość (rzadko potrzebny, B-tree jest prawie zawsze lepszy)
CREATE INDEX idx_sessions_hash ON sessions USING HASH(token);

-- Partial: indeksuj tylko podzbiór wierszy — mniejszy, szybszy
CREATE INDEX idx_orders_pending ON orders(placed_at)
WHERE status = 'pending';

-- Covering / Include: pokrywający — bez sięgania do tabeli
CREATE INDEX idx_users_email_inc ON users(email) INCLUDE (id, name, role);

-- GIN dla full-text i jsonb
CREATE INDEX idx_products_meta ON products USING GIN(metadata jsonb_path_ops);
CREATE INDEX idx_products_fts ON products USING GIN(to_tsvector('simple', name));

-- BRIN dla bardzo dużych tabel time-series (mały rozmiar)
CREATE INDEX idx_logs_brin ON logs USING BRIN(created_at);

-- Sprawdź czy indeks jest używany
EXPLAIN (ANALYZE, BUFFERS) SELECT * FROM orders WHERE customer_id = 42;
-- Index Scan = OK. Seq Scan = sprawdź WHERE/JOIN.

4. UPSERT (PostgreSQL)

-- Dodaj lub zaktualizuj
INSERT INTO products (sku, name, price, stock)
VALUES ('ABC-123', 'Widget', 99.99, 10)
ON CONFLICT (sku) DO UPDATE SET
    name = EXCLUDED.name,
    price = EXCLUDED.price,
    stock = products.stock + EXCLUDED.stock,
    updated_at = NOW()
RETURNING *;

5. Prisma — TypeScript

// schema.prisma:
// model Order {
//     id          String      @id @default(uuid())
//     customerId  String
//     customer    Customer    @relation(fields: [customerId], references: [id])
//     items       OrderItem[]
//     total       Decimal     @db.Decimal(12, 2)
//     placedAt    DateTime    @default(now())
//     @@index([customerId, placedAt(sort: Desc)])
// }

import { PrismaClient } from '@prisma/client';
const prisma = new PrismaClient();

// Eager load — ZAWSZE wybieraj select/include explicit, nie pobieraj * z relacji
const orders = await prisma.order.findMany({
    where: { customerId: 'abc', placedAt: { gte: new Date('2026-01-01') } },
    include: { items: { include: { product: { select: { name: true, price: true } } } } },
    orderBy: { placedAt: 'desc' },
    take: 20
});

// Transakcja
await prisma.$transaction(async (tx) => {
    const order = await tx.order.create({ data: { customerId, total } });
    await tx.orderItem.createMany({ data: items.map(i => ({ ...i, orderId: order.id })) });
    await tx.customer.update({ where: { id: customerId },
        data: { totalSpent: { increment: total } } });
});

// Raw SQL gdy ORM nie wystarcza
const topProducts = await prisma.$queryRaw<{ id: string, name: string, rev: number }[]>`
    SELECT p.id, p.name, SUM(oi.quantity * oi.unit_price) AS rev
    FROM products p
    JOIN order_items oi ON oi.product_id = p.id
    WHERE oi.created_at > NOW() - INTERVAL '30 days'
    GROUP BY p.id
    ORDER BY rev DESC LIMIT 10
`;

6. SQLAlchemy 2.0 — async + typed

from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, selectinload, relationship

class Base(DeclarativeBase): pass

class Order(Base):
    __tablename__ = "orders"
    id: Mapped[str] = mapped_column(primary_key=True)
    customer_id: Mapped[str]
    total: Mapped[float]
    items: Mapped[list["OrderItem"]] = relationship(back_populates="order")

# Async query
async with AsyncSession(engine) as session:
    stmt = (
        select(Order)
        .where(Order.customer_id == customer_id)
        .options(selectinload(Order.items))  # N+1 fix
        .order_by(Order.placed_at.desc())
        .limit(20)
    )
    orders = (await session.execute(stmt)).scalars().all()

# Agregacja
revenue_per_day = await session.execute(
    select(func.date(Order.placed_at).label("day"),
           func.sum(Order.total).label("revenue"))
    .group_by("day")
    .order_by("day")
)
