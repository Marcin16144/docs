public interface IOrderValidator { void Validate(Order o); }
public interface IOrderRepository { Task SaveAsync(Order o); }
public interface IOrderNotifier { Task NotifyAsync(Order o); }

public class PlaceOrderHandler(
    IOrderValidator validator,
    IOrderRepository repo,
    IOrderNotifier notifier)
{
    public async Task HandleAsync(Order order)
    {
        validator.Validate(order);
        await repo.SaveAsync(order);
        await notifier.NotifyAsync(order);
    }
}

3. Repository + Unit of Work (TypeScript + Prisma)

import { PrismaClient } from '@prisma/client';

export interface UserRepository {
    findById(id: string): Promise<User | null>;
    save(user: User): Promise<void>;
}

export class PrismaUserRepository implements UserRepository {
    constructor(private tx: PrismaClient) {}
    findById(id: string) { return this.tx.user.findUnique({ where: { id } }); }
    async save(user: User) { await this.tx.user.upsert({
        where: { id: user.id }, update: user, create: user
    }); }
}

// Unit of Work przez Prisma transaction
export async function transferMoney(prisma: PrismaClient, from: string, to: string, amount: number) {
    return prisma.$transaction(async (tx) => {
        const repo = new PrismaUserRepository(tx as PrismaClient);
        const sender = await repo.findById(from);
        const receiver = await repo.findById(to);
        if (!sender || !receiver) throw new Error('User not found');
        if (sender.balance < amount) throw new Error('Insufficient funds');
        sender.balance -= amount;
        receiver.balance += amount;
        await repo.save(sender);
        await repo.save(receiver);
    });
}

4. DDD — Aggregate Root z inwariantami

Python Domena nie zależy od bazy/HTTP — czysta logika biznesowa.

from dataclasses import dataclass, field
from decimal import Decimal
from uuid import UUID, uuid4

@dataclass(frozen=True)
class Money:  # Value Object
    amount: Decimal
    currency: str

    def __add__(self, other: "Money") -> "Money":
        if self.currency != other.currency:
            raise ValueError("Currency mismatch")
        return Money(self.amount + other.amount, self.currency)

@dataclass
class OrderLine:
    product_id: UUID
    quantity: int
    price: Money

class Order:  # Aggregate Root
    def __init__(self, customer_id: UUID):
        self.id = uuid4()
        self.customer_id = customer_id
        self._lines: list[OrderLine] = []
        self.status = "draft"
        self._events: list = []

    def add_line(self, line: OrderLine):
        if self.status != "draft":
            raise ValueError("Cannot modify confirmed order")
        if line.quantity <= 0:
            raise ValueError("Quantity must be positive")
        self._lines.append(line)

    def confirm(self):
        if not self._lines:
            raise ValueError("Empty order")
        self.status = "confirmed"
        self._events.append({"type": "OrderConfirmed", "id": str(self.id)})

    @property
    def total(self) -> Money:
        return sum((Money(l.price.amount * l.quantity, l.price.currency) for l in self._lines),
                   start=Money(Decimal("0"), "PLN"))
📝 Aggregate Root chroni inwarianty — nikt z zewnątrz nie modyfikuje _lines bezpośrednio.

5. CQRS — split Command i Query

// COMMAND (write side) — używa bogatej domeny
public record CreateProductCommand(string Name, decimal Price);

public class CreateProductHandler(IProductRepository repo)
{
    public async Task<Guid> HandleAsync(CreateProductCommand cmd)
    {
        var product = Product.Create(cmd.Name, cmd.Price); // validates invariants
        await repo.SaveAsync(product);
        return product.Id;
    }
}

// QUERY (read side) — czysty SQL, projekcja, bez ORM
public record ProductListItem(Guid Id, string Name, decimal Price, int Stock);

public class ProductListQuery(string connStr)
{
    public async Task<IReadOnlyList<ProductListItem>> ExecuteAsync(int page, int size)
    {
        using var conn = new NpgsqlConnection(connStr);
        return (await conn.QueryAsync<ProductListItem>(
            @"SELECT id, name, price, stock FROM products
              ORDER BY name LIMIT @size OFFSET @offset",
            new { size, offset = (page - 1) * size })).ToList();
    }
}

6. Saga (orchestration) — TypeScript + Temporal

import { proxyActivities } from '@temporalio/workflow';
import type * as activities from './activities';

const { reserveInventory, chargePayment, shipOrder,
        releaseInventory, refundPayment } = proxyActivities<typeof activities>({
    startToCloseTimeout: '1 minute',
    retry: { maximumAttempts: 3 }
});

export async function orderSaga(orderId: string): Promise<void> {
    let inventoryReserved = false;
    let paymentCharged = false;
    try {
        await reserveInventory(orderId); inventoryReserved = true;
        await chargePayment(orderId);     paymentCharged = true;
        await shipOrder(orderId);
    } catch (err) {
        // Compensating transactions — w odwrotnej kolejności
        if (paymentCharged) await refundPayment(orderId);
        if (inventoryReserved) await releaseInventory(orderId);
        throw err;
    }
}

7. Circuit Breaker (Polly w .NET 9)

using Polly;
using Polly.CircuitBreaker;

var pipeline = new ResiliencePipelineBuilder<HttpResponseMessage>()
    .AddCircuitBreaker(new CircuitBreakerStrategyOptions<HttpResponseMessage>
    {
        FailureRatio = 0.5,                       // 50% błędów otwiera obwód
        MinimumThroughput = 10,                  // po min. 10 requestach
        SamplingDuration = TimeSpan.FromSeconds(30),
        BreakDuration = TimeSpan.FromSeconds(15),
        ShouldHandle = new PredicateBuilder<HttpResponseMessage>()
            .Handle<HttpRequestException>()
            .HandleResult(r => (int)r.StatusCode >= 500)
    })
    .Build();

var response = await pipeline.ExecuteAsync(async ct =>
    await httpClient.GetAsync("https://api.partner.com/data", ct));

8. Hexagonal — Ports & Adapters

// === DOMAIN (core) ===
export interface PaymentPort {
    charge(amount: number, currency: string): Promise<string>;
}
export class CheckoutService {
    constructor(private payment: PaymentPort) {}
    async checkout(cartTotal: number): Promise<string> {
        return this.payment.charge(cartTotal, 'PLN');
    }
}

// === ADAPTER (infrastructure) ===
export class StripeAdapter implements PaymentPort {
    constructor(private stripe: Stripe) {}
    async charge(amount: number, currency: string): Promise<string> {
        const intent = await this.stripe.paymentIntents.create({
            amount: Math.round(amount * 100), currency
        });
        return intent.id;
    }
}

// === TEST (in-memory adapter) ===
export class InMemoryPayment implements PaymentPort {
    charges: { amount: number }[] = [];
    async charge(amount: number) { this.charges.push({ amount }); return 'mock-id'; }
}
