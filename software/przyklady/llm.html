2. Claude API — basic + tool use

from anthropic import Anthropic
client = Anthropic()  # czyta ANTHROPIC_API_KEY z env

# --- Basic ---
msg = client.messages.create(
    model="claude-opus-4-7",
    max_tokens=1024,
    messages=[{"role": "user", "content": "Krótko: czym jest MoE?"}]
)
print(msg.content[0].text)

# --- Tool use ---
tools = [{
    "name": "get_weather",
    "description": "Pobiera bieżącą pogodę dla miasta",
    "input_schema": {
        "type": "object",
        "properties": {"city": {"type": "string"}},
        "required": ["city"]
    }
}]

response = client.messages.create(
    model="claude-opus-4-7",
    max_tokens=1024,
    tools=tools,
    messages=[{"role": "user", "content": "Jaka pogoda w Krakowie?"}]
)

if response.stop_reason == "tool_use":
    tool_use = next(b for b in response.content if b.type == "tool_use")
    result = fetch_weather(tool_use.input["city"])  # Twoja funkcja
    final = client.messages.create(
        model="claude-opus-4-7", max_tokens=1024, tools=tools,
        messages=[
            {"role": "user", "content": "Jaka pogoda w Krakowie?"},
            {"role": "assistant", "content": response.content},
            {"role": "user", "content": [{
                "type": "tool_result",
                "tool_use_id": tool_use.id,
                "content": result
            }]}
        ]
    )

3. Prompt caching — duże system prompty

# Cache system prompt (długi! min. 1024 tokeny dla Opus, 2048 dla Sonnet/Haiku)
# TTL: 5 minut domyślnie, można rozszerzyć do 1h
response = client.messages.create(
    model="claude-opus-4-7",
    max_tokens=1024,
    system=[
        {
            "type": "text",
            "text": open("long_codebase_context.md").read(),  # np. 50k tokens
            "cache_control": {"type": "ephemeral"}
        }
    ],
    messages=[{"role": "user", "content": "Wyjaśnij funkcję X"}]
)
# Pierwsze wywołanie: pełna cena. Kolejne w 5 min: tylko ~10% kosztu cache hit.
print(response.usage.cache_creation_input_tokens)  # pierwsze: ile zapisano
print(response.usage.cache_read_input_tokens)      # kolejne: ile odczytano (90% taniej)
💡 Cache działa dopiero gdy zawartość ma min. 1024 tokeny (Opus) lub 2048 (Sonnet/Haiku). Krótsze prompty są ignorowane.

4. Minimalny RAG (Chroma + sentence-transformers)

# pip install chromadb sentence-transformers anthropic
import chromadb
from sentence_transformers import SentenceTransformer
from anthropic import Anthropic

embedder = SentenceTransformer("intfloat/multilingual-e5-large")
client = chromadb.PersistentClient(path="./chroma_db")
coll = client.get_or_create_collection("docs")

# 1. Indeksuj
docs = [
    "Python jest interpretowanym językiem wysokiego poziomu.",
    "Rust to język systemowy z gwarancjami bezpieczeństwa pamięci.",
    "FastAPI to nowoczesny framework Python do API.",
]
coll.add(
    ids=[str(i) for i in range(len(docs))],
    embeddings=embedder.encode(docs).tolist(),
    documents=docs
)

# 2. Zapytaj
query = "Który język jest bezpieczny pamięciowo?"
results = coll.query(query_embeddings=[embedder.encode(query).tolist()], n_results=2)
context = "\n".join(results["documents"][0])

# 3. Wygeneruj odpowiedź z kontekstem
anthropic = Anthropic()
msg = anthropic.messages.create(
    model="claude-opus-4-7", max_tokens=512,
    messages=[{"role": "user", "content": f"Kontekst:\n{context}\n\nPytanie: {query}"}]
)
print(msg.content[0].text)

5. RAG produkcyjny — pgvector + hybrid search + reranking

-- Schema
CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pg_trgm;

CREATE TABLE chunks (
    id BIGSERIAL PRIMARY KEY,
    doc_id UUID NOT NULL,
    content TEXT NOT NULL,
    embedding vector(1024),  -- multilingual-e5-large
    fts tsvector GENERATED ALWAYS AS (to_tsvector('simple', content)) STORED
);

CREATE INDEX chunks_embedding_idx ON chunks
    USING hnsw (embedding vector_cosine_ops) WITH (m = 16, ef_construction = 64);
CREATE INDEX chunks_fts_idx ON chunks USING GIN (fts);
# Hybrid search: kombinacja vector + BM25/FTS, potem rerank
from sentence_transformers import CrossEncoder
reranker = CrossEncoder("BAAI/bge-reranker-v2-m3")

def retrieve(query: str, k: int = 5) -> list[str]:
    q_emb = embedder.encode([f"query: {query}"])[0].tolist()

    # 1. Pobierz top 20 z vector i top 20 z FTS
    rows = db.execute("""
        WITH vec AS (
            SELECT id, content, 1 - (embedding <=> %s::vector) AS score
            FROM chunks ORDER BY embedding <=> %s::vector LIMIT 20
        ),
        fts AS (
            SELECT id, content, ts_rank(fts, plainto_tsquery('simple', %s)) AS score
            FROM chunks WHERE fts @@ plainto_tsquery('simple', %s)
            ORDER BY score DESC LIMIT 20
        )
        SELECT DISTINCT id, content FROM (SELECT * FROM vec UNION SELECT * FROM fts) u
    """, [q_emb, q_emb, query, query]).fetchall()

    # 2. Rerank cross-encoderem (precyzyjniejszy niż samo cosine)
    pairs = [(query, r[1]) for r in rows]
    scores = reranker.predict(pairs)
    ranked = sorted(zip(rows, scores), key=lambda x: x[1], reverse=True)
    return [r[0][1] for r in ranked[:k]]

6. Fine-tuning LoRA (Unsloth, ~10 minut na T4)

# pip install unsloth
from unsloth import FastLanguageModel
from trl import SFTTrainer
from datasets import load_dataset
from transformers import TrainingArguments

model, tokenizer = FastLanguageModel.from_pretrained(
    model_name="unsloth/Llama-3.2-3B-Instruct",
    max_seq_length=2048,
    load_in_4bit=True,
)

model = FastLanguageModel.get_peft_model(
    model, r=16, lora_alpha=16, lora_dropout=0,
    target_modules=["q_proj", "k_proj", "v_proj", "o_proj",
                    "gate_proj", "up_proj", "down_proj"],
    bias="none", use_gradient_checkpointing="unsloth",
)

dataset = load_dataset("json", data_files="train.jsonl", split="train")

trainer = SFTTrainer(
    model=model, tokenizer=tokenizer, train_dataset=dataset,
    dataset_text_field="text", max_seq_length=2048,
    args=TrainingArguments(
        per_device_train_batch_size=2, gradient_accumulation_steps=4,
        warmup_steps=5, num_train_epochs=1, learning_rate=2e-4,
        fp16=False, bf16=True, logging_steps=1,
        output_dir="./out", optim="adamw_8bit"
    )
)
trainer.train()
model.save_pretrained_merged("./merged", tokenizer, save_method="merged_16bit")

7. Agent z tool calling (Pydantic AI)

# pip install pydantic-ai
from pydantic_ai import Agent, RunContext
from pydantic import BaseModel
import httpx

class Deps(BaseModel):
    http: httpx.AsyncClient

agent = Agent(
    "anthropic:claude-opus-4-7",
    deps_type=Deps,
    system_prompt="You are a helpful research assistant. Use search when needed.",
)

@agent.tool
async def search_web(ctx: RunContext[Deps], query: str) -> str:
    """Wyszukuje w internecie i zwraca podsumowanie."""
    r = await ctx.deps.http.get("https://api.search.brave.com/res/v1/web/search",
                              params={"q": query},
                              headers={"X-Subscription-Token": SEARCH_KEY})
    results = r.json()["web"]["results"][:3]
    return "\n".join(f"{r['title']}: {r['description']}" for r in results)

@agent.tool
async def fetch_page(ctx: RunContext[Deps], url: str) -> str:
    """Pobiera tekst ze strony WWW."""
    r = await ctx.deps.http.get(url, follow_redirects=True)
    return r.text[:3000]

async def main():
    async with httpx.AsyncClient() as http:
        result = await agent.run("Co nowego w Pythonie 3.14?", deps=Deps(http=http))
        print(result.output)
