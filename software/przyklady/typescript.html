📘 TypeScript — zaawansowane wzorce

Branded types, discriminated unions, mapped types, Zod walidacja, narrowing. TypeScript 5.7+.

1. Branded types — zapobiegnij pomieszaniu ID

// ❌ Bez brand — łatwo wstawić user id zamiast order id
function getOrder(orderId: string) { ... }
const userId: string = "user-123";
getOrder(userId);  // ❌ kompilator nie zauważy!

// ✅ Z brand
type Brand<K, T> = K & { __brand: T };
type UserId = Brand<string, 'UserId'>;
type OrderId = Brand<string, 'OrderId'>;

function userId(s: string): UserId { return s as UserId; }
function orderId(s: string): OrderId { return s as OrderId; }

function getOrder(id: OrderId) { ... }

const u = userId("user-123");
getOrder(u);   // ❌ Type 'UserId' is not assignable to 'OrderId'
getOrder(orderId("ord-456"));  // ✅

2. Discriminated unions — type-safe state

type AsyncState<T> =
    | { status: 'idle' }
    | { status: 'loading' }
    | { status: 'success'; data: T }
    | { status: 'error'; error: Error };

function render<T>(state: AsyncState<T>): string {
    switch (state.status) {
        case 'idle':    return 'Ready';
        case 'loading': return 'Loading...';
        case 'success': return `Got: ${JSON.stringify(state.data)}`;
        case 'error':   return `Error: ${state.error.message}`;
        default: {
            const _exhaustive: never = state;  // kompilator pilnuje exhaustywności!
            throw new Error(_exhaustive);
        }
    }
}

// Jeśli dodasz nowy variant 'cancelled' i nie obsłużysz w switch,
// TypeScript natychmiast zaprotestuje przy linii _exhaustive.

3. Mapped types — przekształcanie kształtu

type User = { id: string; name: string; email: string; age: number };

// Wszystkie pola optional
type PartialUser = Partial<User>;

// Tylko wybrane
type UserPreview = Pick<User, 'id' | 'name'>;

// Bez wybranych
type UserPublic = Omit<User, 'email'>;

// Wszystkie readonly
type Frozen = Readonly<User>;

// Własny mapped type — wszystkie pola jako string
type AllStrings<T> = { [K in keyof T]: string };
type UserForm = AllStrings<User>;
// = { id: string; name: string; email: string; age: string }

// Nullable wszystko
type Nullable<T> = { [K in keyof T]: T[K] | null };

// Tylko właściwości danego typu
type KeysOfType<T, V> = { [K in keyof T]: T[K] extends V ? K : never }[keyof T];
type StringKeys = KeysOfType<User, string>;   // 'id' | 'name' | 'email'
type NumberKeys = KeysOfType<User, number>;   // 'age'

4. Template literal types

type Method = 'GET' | 'POST' | 'PUT' | 'DELETE';
type Path = '/users' | '/orders' | '/products';
type Route = `${Method} ${Path}`;
// = "GET /users" | "GET /orders" | ... 12 kombinacji

// Event handler names
type EventName = 'click' | 'focus' | 'blur';
type Handler<T extends string> = `on${Capitalize<T>}`;
type Handlers = Handler<EventName>;
// = "onClick" | "onFocus" | "onBlur"

// Path parameter extraction
type ExtractParams<T extends string> =
    T extends `${infer _}/:${infer Param}/${infer Rest}`
        ? Param | ExtractParams<`/${Rest}`>
        : T extends `${infer _}/:${infer Param}`
            ? Param
            : never;

type P = ExtractParams<'/users/:userId/posts/:postId'>;
// = "userId" | "postId"

5. Zod — walidacja runtime + typy compile-time

import { z } from 'zod';

const UserSchema = z.object({
    id: z.string().uuid(),
    name: z.string().min(1).max(100),
    email: z.string().email(),
    role: z.enum(['admin', 'user', 'guest']),
    age: z.number().int().min(0).max(150),
    createdAt: z.string().datetime().transform(s => new Date(s)),
    tags: z.array(z.string()).default([]),
    address: z.object({
        city: z.string(),
        zip: z.string().regex(/^\d{2}-\d{3}$/, 'Format: NN-NNN')
    }).optional()
});

type User = z.infer<typeof UserSchema>;
// Type wnioskowany automatycznie z schematu!

// Walidacja
const result = UserSchema.safeParse(rawData);
if (!result.success) {
    console.log(result.error.flatten());
    // { fieldErrors: { email: ['Invalid email'], age: ['Number must be >= 0'] } }
} else {
    const user: User = result.data;  // pełna pewność typu i wartości
}

6. Type guards i narrowing

// User-defined type guard
function isError(value: unknown): value is Error {
    return value instanceof Error;
}

try {
    someOperation();
} catch (err) {
    if (isError(err)) {
        console.log(err.message, err.stack);  // TS wie że to Error
    } else if (typeof err === 'string') {
        console.log(err);
    }
}

// Assertion function — rzuca jeśli warunek false
function assertDefined<T>(value: T | undefined, name: string): asserts value is T {
    if (value === undefined) throw new Error(`${name} is undefined`);
}

function process(maybeUser: User | undefined) {
    assertDefined(maybeUser, 'user');
    console.log(maybeUser.name);  // TS wie że nie undefined
}

7. const assertions + as const

// Bez as const
const colors = ['red', 'green', 'blue'];  // string[]

// Z as const
const colors = ['red', 'green', 'blue'] as const;
// readonly ['red', 'green', 'blue']
type Color = (typeof colors)[number];  // 'red' | 'green' | 'blue'

// Konfiguracja z preserved literals
const config = {
    apiUrl: 'https://api.example.com',
    timeout: 5000,
    methods: ['GET', 'POST']
} as const;
// Wszystkie pola readonly, literały zachowane

type Method = (typeof config.methods)[number];  // 'GET' | 'POST'

8. Utility types — własna biblioteka

// Deep partial — wszystkie zagnieżdżone pola optional
type DeepPartial<T> = T extends object
    ? { [P in keyof T]?: DeepPartial<T[P]> }
    : T;

// Deep readonly
type DeepReadonly<T> = T extends object
    ? { readonly [P in keyof T]: DeepReadonly<T[P]> }
    : T;

// Promise unwrap
type Awaited<T> = T extends Promise<infer U> ? Awaited<U> : T;
// (już built-in od TS 4.5)

// Function arguments & return
type Params = Parameters<typeof fetch>;
type Result = ReturnType<typeof fetch>;
type Resolved = Awaited<ReturnType<typeof fetch>>;  // Response

// Distributed union — nieparami
type NonNull<T> = T extends null | undefined ? never : T;
// = built-in NonNullable<T>
