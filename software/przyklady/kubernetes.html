☸️ Kubernetes — gotowe manifesty

Deployment + Service + Ingress + ConfigMap + Secret + HPA — produkcyjny zestaw na 1 stronę.

1. Deployment + Service

apiVersion: apps/v1
kind: Deployment
metadata:
  name: api
  labels: { app: api }
spec:
  replicas: 3
  strategy:
    type: RollingUpdate
    rollingUpdate: { maxSurge: 1, maxUnavailable: 0 }
  selector:
    matchLabels: { app: api }
  template:
    metadata:
      labels: { app: api }
    spec:
      serviceAccountName: api
      securityContext:
        runAsNonRoot: true
        runAsUser: 1000
        fsGroup: 1000
      containers:
        - name: api
          image: ghcr.io/myorg/api:1.4.2
          ports: [{ containerPort: 3000 }]
          env:
            - { name: DATABASE_URL, valueFrom: { secretKeyRef: { name: api-secrets, key: db_url } } }
            - { name: LOG_LEVEL, valueFrom: { configMapKeyRef: { name: api-config, key: log_level } } }
          resources:
            requests: { cpu: "100m", memory: "128Mi" }
            limits:   { cpu: "500m", memory: "512Mi" }
          livenessProbe:
            httpGet: { path: /health, port: 3000 }
            initialDelaySeconds: 10
            periodSeconds: 30
          readinessProbe:
            httpGet: { path: /ready, port: 3000 }
            periodSeconds: 5
          securityContext:
            allowPrivilegeEscalation: false
            readOnlyRootFilesystem: true
            capabilities: { drop: ["ALL"] }
---
apiVersion: v1
kind: Service
metadata: { name: api }
spec:
  selector: { app: api }
  ports: [{ port: 80, targetPort: 3000 }]
  type: ClusterIP

2. ConfigMap + Secret

apiVersion: v1
kind: ConfigMap
metadata: { name: api-config }
data:
  log_level: "info"
  feature_flags: |
    new_checkout: true
    ai_recommendations: false
---
apiVersion: v1
kind: Secret
metadata: { name: api-secrets }
type: Opaque
stringData:    # automatycznie zakodowane base64
  db_url: postgres://app:secret@db:5432/app
  jwt_key: "twoj-bardzo-dlugi-klucz-min-32-znaki"
⚠️ Nie commituj Secret do gita. Użyj Sealed Secrets, External Secrets Operator lub SOPS.

3. Ingress z TLS (cert-manager + Let's Encrypt)

apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: api
  annotations:
    cert-manager.io/cluster-issuer: "letsencrypt-prod"
    nginx.ingress.kubernetes.io/proxy-body-size: "10m"
    nginx.ingress.kubernetes.io/rate-limit-rpm: "60"
spec:
  ingressClassName: nginx
  tls:
    - { hosts: [api.example.com], secretName: api-tls }
  rules:
    - host: api.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service: { name: api, port: { number: 80 } }

4. HPA — autoskalowanie

apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata: { name: api }
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: api
  minReplicas: 3
  maxReplicas: 20
  metrics:
    - type: Resource
      resource: { name: cpu, target: { type: Utilization, averageUtilization: 70 } }
    - type: Resource
      resource: { name: memory, target: { type: Utilization, averageUtilization: 80 } }
  behavior:
    scaleDown:
      stabilizationWindowSeconds: 300  # 5 min cooldown
    scaleUp:
      policies: [{ type: Percent, value: 100, periodSeconds: 60 }]

5. NetworkPolicy — domyślnie zero-trust

# Default deny — nic do/ze świata, dopóki nie zezwolisz
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata: { name: default-deny }
spec:
  podSelector: {}
  policyTypes: ["Ingress", "Egress"]
---
# API może odbierać tylko od ingress-nginx i komunikować się z db
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata: { name: api-policy }
spec:
  podSelector: { matchLabels: { app: api } }
  policyTypes: ["Ingress", "Egress"]
  ingress:
    - from:
        - { namespaceSelector: { matchLabels: { name: ingress-nginx } } }
      ports: [{ port: 3000 }]
  egress:
    - to: [{ podSelector: { matchLabels: { app: db } } }]
      ports: [{ port: 5432 }]
    - to: []  # DNS
      ports: [{ port: 53, protocol: UDP }]

6. Komendy kubectl, których używasz codziennie

# Apply / delete
kubectl apply -f k8s/
kubectl delete -f k8s/

# Inspect
kubectl get pods -A -o wide
kubectl describe pod api-7d9f8c5d7-x2p9l
kubectl logs -f deploy/api --tail=100
kubectl logs -l app=api --since=10m --all-containers

# Debugowanie
kubectl exec -it deploy/api -- sh
kubectl port-forward svc/api 3000:80
kubectl run debug --rm -it --image=nicolaka/netshoot -- bash
kubectl debug -it deploy/api --image=busybox --target=api

# Skalowanie / rollout
kubectl scale deploy/api --replicas=10
kubectl rollout status deploy/api
kubectl rollout history deploy/api
kubectl rollout undo deploy/api --to-revision=3

# Resources
kubectl top pods
kubectl top nodes
