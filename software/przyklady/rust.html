1. Ownership — przykłady i pułapki

fn main() {
    let s = String::from("hello");
    let s2 = s;          // move — s NIE ma już dostępu
    // println!("{s}");  // ❌ błąd kompilacji: borrow of moved value
    println!("{s2}");    // ✅

    let n = 42;             // Copy types (i32, bool, char) — kopiują się
    let n2 = n;
    println!("{n} {n2}");  // ✅ oba działają

    let s = String::from("hi");
    print_borrow(&s);     // pożyczka — s dalej istnieje
    println!("{s}");
}

fn print_borrow(s: &String) { println!("borrowed: {s}"); }

2. Result + ? operator (error handling bez wyjątków)

use std::fs;
use std::io;
use std::num::ParseIntError;

// thiserror crate ułatwia tworzenie custom errors
#[derive(thiserror::Error, Debug)]
enum ConfigError {
    #[error("Cannot read config file: {0}")]
    Io(#[from] io::Error),

    #[error("Invalid port number: {0}")]
    InvalidPort(#[from] ParseIntError),

    #[error("Missing field: {0}")]
    MissingField(String),
}

fn load_port(path: &str) -> Result<u16, ConfigError> {
    let content = fs::read_to_string(path)?;       // ? = unwrap or return Err
    let line = content.lines().next()
        .ok_or(ConfigError::MissingField("port".into()))?;
    let port: u16 = line.trim().parse()?;
    Ok(port)
}

fn main() {
    match load_port("config.txt") {
        Ok(p) => println!("Port: {p}"),
        Err(e) => eprintln!("Error: {e}"),
    }
}

4. Async + tokio (web scraper)

// Cargo.toml:
// tokio = { version = "1", features = ["full"] }
// reqwest = { version = "0.12", features = ["json"] }
// serde = { version = "1", features = ["derive"] }

use serde::Deserialize;
use tokio::task::JoinSet;

#[derive(Deserialize, Debug)]
struct Repo { name: String, stargazers_count: u32 }

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let users = vec!["rust-lang", "tokio-rs", "serde-rs"];
    let client = reqwest::Client::builder()
        .user_agent("rust-example")
        .build()?;

    let mut set = JoinSet::new();
    for user in users {
        let client = client.clone();
        let user = user.to_string();
        set.spawn(async move {
            let url = format!("https://api.github.com/users/{user}/repos");
            let repos: Vec<Repo> = client.get(&url).send().await?.json().await?;
            Ok::<_, reqwest::Error>((user, repos))
        });
    }

    while let Some(result) = set.join_next().await {
        match result? {
            Ok((user, repos)) => {
                let top = repos.iter()
                    .max_by_key(|r| r.stargazers_count)
                    .map(|r| (&r.name, r.stargazers_count));
                println!("{user}: top repo = {top:?}");
            }
            Err(e) => eprintln!("Error: {e}"),
        }
    }
    Ok(())
}

5. Generics + traits

use std::fmt::Display;
use std::ops::Add;

// Generic function with multiple trait bounds
fn sum_and_print<T>(items: &[T]) -> T
where
    T: Add<Output = T> + Copy + Default + Display,
{
    let total = items.iter().copied().fold(T::default(), |acc, x| acc + x);
    println!("Sum: {total}");
    total
}

// Custom trait
trait Greet {
    fn greeting(&self) -> String;
    fn greet(&self) { println!("{}", self.greeting()); }  // default impl
}

struct User { name: String }
impl Greet for User {
    fn greeting(&self) -> String { format!("Hello, {}", self.name) }
}

fn main() {
    sum_and_print(&[1, 2, 3, 4]);
    sum_and_print(&[1.5, 2.5, 3.5]);
    User { name: "Anna".into() }.greet();
}

8. Najczęstsze błędy borrow checker

// ❌ "cannot borrow as mutable more than once"
let mut v = vec![1, 2, 3];
let first = &v[0];
v.push(4);              // błąd: borrow as mutable while immutably borrowed
println!("{first}");

// ✅ Zamknij scope pożyczki
let mut v = vec![1, 2, 3];
{
    let first = &v[0];
    println!("{first}");
}
v.push(4);              // OK — first już nie żyje

// ❌ "use of moved value"
let s = String::from("hi");
let s2 = s;
println!("{s}");         // błąd: s zostało moved

// ✅ Klonuj jeśli musisz mieć kopię
let s = String::from("hi");
let s2 = s.clone();
println!("{s} {s2}");
