Doctrine ORM

Encje z atrybutami PHP 8, EntityManager, DQL, QueryBuilder, asocjacje i migracje

Doctrine vs Active Record

Doctrine implementuje wzorzec Data Mapper — separuje logike domenowa (encje) od logiki persystencji (EntityManager). Encje to czyste obiekty PHP bez dziedziczenia z klas ORM. Doctrine zarzadza stanem encji w Unit of Work i wysyla zmiany do bazy dopiero przy flush(). To podejscie lepiej nadaje sie do skomplikowanych modeli domenowych i architektur DDD.

Definicja encji z atrybutami PHP 8

Doctrine 2.9+ obsługuje atrybuty PHP 8 jako alternatywe dla adnotacji XML i YAML. Atrybuty sa sprawdzane w czasie wykonania i oferuja pelna kontrole nad mapowaniem obiektowo-relacyjnym bezposrednio w klasie encji.

<?php

namespace App\Entity;

use Doctrine\ORM\Mapping as ORM;
use Doctrine\Common\Collections\ArrayCollection;
use Doctrine\Common\Collections\Collection;
use Symfony\Component\Validator\Constraints as Assert;

#[ORM\Entity(repositoryClass: ProductRepository::class)]
#[ORM\Table(name: 'products')]
#[ORM\HasLifecycleCallbacks]
class Product
{
    #[ORM\Id]
    #[ORM\GeneratedValue(strategy: 'IDENTITY')]
    #[ORM\Column(type: 'integer')]
    private ?int $id = null;

    #[ORM\Column(type: 'string', length: 255, unique: true)]
    #[Assert\NotBlank]
    #[Assert\Length(min: 2, max: 255)]
    private string $name;

    #[ORM\Column(type: 'string', length: 255, unique: true)]
    private string $slug;

    #[ORM\Column(type: 'decimal', precision: 10, scale: 2)]
    #[Assert\Positive]
    private string $price;

    #[ORM\Column(type: 'text', nullable: true)]
    private ?string $description = null;

    #[ORM\Column(type: 'integer', options: ['default' => 0])]
    private int $stock = 0;

    #[ORM\Column(type: 'datetime_immutable')]
    private \DateTimeImmutable $createdAt;

    #[ORM\Column(type: 'datetime_immutable', nullable: true)]
    private ?\DateTimeImmutable $updatedAt = null;

    // Relacja Many-to-One: wiele produktow nalezy do jednej kategorii
    #[ORM\ManyToOne(targetEntity: Category::class, inversedBy: 'products')]
    #[ORM\JoinColumn(nullable: false, onDelete: 'RESTRICT')]
    private Category $category;

    // Relacja One-to-Many: jeden produkt ma wiele recenzji
    #[ORM\OneToMany(
        mappedBy: 'product',
        targetEntity: Review::class,
        cascade: ['persist', 'remove'],
        orphanRemoval: true,
        fetch: 'LAZY'
    )]
    #[ORM\OrderBy(['createdAt' => 'DESC'])]
    private Collection $reviews;

    // Relacja Many-to-Many: produkt moze miec wiele tagow
    #[ORM\ManyToMany(targetEntity: Tag::class, cascade: ['persist'])]
    #[ORM\JoinTable(name: 'products_tags')]
    private Collection $tags;

    public function __construct(string $name, string $price, Category $category)
    {
        $this->name      = $name;
        $this->price     = $price;
        $this->category  = $category;
        $this->createdAt = new \DateTimeImmutable();
        $this->reviews   = new ArrayCollection();
        $this->tags      = new ArrayCollection();
    }

    // Lifecycle callback — wywolywany przed Update
    #[ORM\PreUpdate]
    public function onPreUpdate(): void
    {
        $this->updatedAt = new \DateTimeImmutable();
    }

    public function getId(): ?int { return $this->id; }
    public function getName(): string { return $this->name; }

    public function addReview(Review $review): void
    {
        if (!$this->reviews->contains($review)) {
            $this->reviews->add($review);
            $review->setProduct($this);
        }
    }

    public function removeReview(Review $review): void
    {
        $this->reviews->removeElement($review);
    }

    public function getReviews(): Collection { return $this->reviews; }
}

EntityManager — Unit of Work

EntityManager jest centralnym komponentem Doctrine. Sluzy do pobierania, zapisywania i usuwania encji. Implementuje wzorzec Unit of Work — sledzi wszystkie zmiany w encjach i wysyla je do bazy w jednej transakcji przy wywolaniu flush().

<?php

namespace App\Controller;

use App\Entity\Product;
use App\Repository\ProductRepository;
use Doctrine\ORM\EntityManagerInterface;
use Symfony\Bundle\FrameworkBundle\Controller\AbstractController;
use Symfony\Component\HttpFoundation\JsonResponse;
use Symfony\Component\Routing\Attribute\Route;

#[Route('/api/products', name: 'api_products_')]
class ProductController extends AbstractController
{
    public function __construct(
        private readonly EntityManagerInterface $em,
        private readonly ProductRepository $productRepository,
    ) {}

    #[Route('', name: 'index', methods: ['GET'])]
    public function index(): JsonResponse
    {
        $products = $this->productRepository->findAllInStock();
        return $this->json($products, context: ['groups' => ['product:list']]);
    }

    #[Route('', name: 'create', methods: ['POST'])]
    public function create(Request $request): JsonResponse
    {
        $data     = json_decode($request->getContent(), true, flags: JSON_THROW_ON_ERROR);
        $category = $this->em->find(Category::class, $data['category_id']);

        $product = new Product($data['name'], $data['price'], $category);

        // persist() — rejestruje encje w Unit of Work
        $this->em->persist($product);
        // flush() — wysyla INSERT/UPDATE/DELETE do bazy (transakcja)
        $this->em->flush();

        return $this->json($product, 201, context: ['groups' => ['product:detail']]);
    }

    #[Route('/{id}', name: 'delete', methods: ['DELETE'])]
    public function delete(Product $product): JsonResponse
    {
        $this->em->remove($product);
        $this->em->flush();
        return $this->json(null, 204);
    }
}

Repository Pattern i DQL

Repository hermetyzuje logike zapytan do bazy danych. Doctrine oferuje bazowa klase EntityRepository z metodami jak find(), findBy(), findOneBy(). Wlasne metody mozna pisac w DQL lub QueryBuilder.

Wlasne repozytorium z DQL i QueryBuilder 
<?php

namespace App\Repository;

use App\Entity\Product;
use Doctrine\Bundle\DoctrineBundle\Repository\ServiceEntityRepository;
use Doctrine\Persistence\ManagerRegistry;
use Doctrine\ORM\QueryBuilder;

class ProductRepository extends ServiceEntityRepository
{
    public function __construct(ManagerRegistry $registry)
    {
        parent::__construct($registry, Product::class);
    }

    // DQL — Doctrine Query Language (obiektowe SQL)
    public function findExpensive(float $minPrice): array
    {
        return $this->getEntityManager()
            ->createQuery(
                'SELECT p, c FROM App\Entity\Product p
                 JOIN p.category c
                 WHERE p.price >= :price
                 AND p.stock > 0
                 ORDER BY p.price DESC'
            )
            ->setParameter('price', $minPrice)
            ->setMaxResults(50)
            ->getResult();
    }

    // QueryBuilder — programowe budowanie zapytan
    public function findAllInStock(): array
    {
        return $this->createQueryBuilder('p')
            ->select('p', 'c', 't')
            ->leftJoin('p.category', 'c')
            ->leftJoin('p.tags', 't')
            ->where('p.stock > :stock')
            ->setParameter('stock', 0)
            ->orderBy('p.createdAt', 'DESC')
            ->setMaxResults(100)
            ->getQuery()
            ->getResult();
    }

    // Natywne SQL z mapowaniem na encje
    public function findByFullText(string $query): array
    {
        $sql = 'SELECT p.* FROM products p
                WHERE MATCH(p.name, p.description) AGAINST(:q IN BOOLEAN MODE)
                AND p.stock > 0
                LIMIT 20';

        $rsm = new \Doctrine\ORM\Query\ResultSetMappingBuilder($this->getEntityManager());
        $rsm->addRootEntityFromClassMetadata(Product::class, 'p');

        return $this->getEntityManager()
            ->createNativeQuery($sql, $rsm)
            ->setParameter('q', $query . '*')
            ->getResult();
    }
}
