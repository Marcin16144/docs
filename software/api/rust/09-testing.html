Testowanie Rust API

Testy jednostkowe async, testy integracyjne Axum, testcontainers, wiremock i cargo-nextest — kompletny zestaw narzedzi do testowania API w Rust.

#[tokio::test] — asynchroniczne testy jednostkowe

Makro #[tokio::test] dziala analogicznie do #[tokio::main] — uruchamia runtime Tokio dla kazdego testu. Testy async sa pelnoprawnymi testami Rust i dzialaja z cargo test.

// src/services/user_service.rs
#[cfg(test)]
mod tests {
    use super::*;
    use tokio::test;

    #[tokio::test]
    async fn test_hash_and_verify_password() {
        let password = "super_secret_123!";
        let hash = hash_password(password).await.unwrap();

        assert_ne!(hash, password);
        assert!(hash.starts_with("$argon2"));

        let is_valid = verify_password(password, &hash).await.unwrap();
        assert!(is_valid);

        let is_invalid = verify_password("wrong_password", &hash).await.unwrap();
        assert!(!is_invalid);
    }

    #[tokio::test]
    async fn test_jwt_sign_and_verify() {
        let service = JwtService::new_hs256(b"test_secret_key_minimum_32_bytes!");
        let token = service.sign("user_42", "test@example.com", "user").unwrap();

        let claims = service.verify(&token).unwrap();
        assert_eq!(claims.sub, "user_42");
        assert_eq!(claims.email, "test@example.com");
        assert_eq!(claims.role, "user");
    }

    #[tokio::test]
    async fn test_expired_jwt_is_rejected() {
        let service = JwtService::new_hs256(b"test_secret_key_minimum_32_bytes!");
        let expired_token = "eyJ0eXAiOiJKV1QiLCJhbGciOiJIUzI1NiJ9.fake";
        assert!(service.verify(expired_token).is_err());
    }
}

axum::test — TestClient dla handlerów

Biblioteka axum-test (lub wbudowany tower::ServiceExt) pozwala testowac handlery Axum bez uruchamiania prawdziwego serwera HTTP. Requesty sa wysylane bezposrednio przez trait Service.

// Cargo.toml [dev-dependencies]
// axum-test = "15"

use axum_test::TestServer;

async fn build_test_app() -> TestServer {
    let db = create_test_db().await;
    let state = Arc::new(AppState { db });
    let app = build_router(state);
    TestServer::new(app).unwrap()
}

#[tokio::test]
async fn test_create_user_returns_201() {
    let server = build_test_app().await;

    let response = server
        .post("/api/v1/users")
        .json(&serde_json::json!({
            "name": "Jan Kowalski",
            "email": "jan@example.com"
        }))
        .await;

    response.assert_status_created();
    let body: serde_json::Value = response.json();
    assert_eq!(body["name"], "Jan Kowalski");
    assert!(body["id"].is_number());
}

#[tokio::test]
async fn test_get_user_not_found_returns_404() {
    let server = build_test_app().await;
    let response = server.get("/api/v1/users/99999").await;
    response.assert_status_not_found();
}

#[tokio::test]
async fn test_protected_endpoint_requires_auth() {
    let server = build_test_app().await;
    let response = server.get("/api/v1/profile").await;
    response.assert_status_unauthorized();
}

#[tokio::test]
async fn test_protected_endpoint_with_valid_token() {
    let server = build_test_app().await;
    let token = generate_test_token("user_1");

    let response = server
        .get("/api/v1/profile")
        .authorization_bearer(token)
        .await;

    response.assert_status_ok();
}

reqwest — HTTP client w testach integracyjnych

Do testow integracyjnych uruchamiajacych prawdziwy serwer uzyj reqwest jako client HTTP. Serwer startuje na losowym porcie (:0), klient odpytuje go przez http://127.0.0.1:PORT.

// tests/integration_test.rs
use tokio::net::TcpListener;

async fn spawn_test_server() -> (String, tokio::task::JoinHandle<()>) {
    let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
    let addr = listener.local_addr().unwrap();
    let app = build_app_for_tests().await;

    let handle = tokio::spawn(async move {
        axum::serve(listener, app).await.unwrap();
    });

    (format!("http://{addr}"), handle)
}

#[tokio::test]
async fn integration_test_full_user_lifecycle() {
    let (base_url, _server) = spawn_test_server().await;
    let client = reqwest::Client::new();

    // Tworzenie uzytkownika
    let create_resp = client
        .post(format!("{base_url}/api/v1/users"))
        .json(&serde_json::json!({"name": "Anna", "email": "anna@test.com"}))
        .send().await.unwrap();
    assert_eq!(create_resp.status(), 201);
    let user: serde_json::Value = create_resp.json().await.unwrap();
    let user_id = user["id"].as_i64().unwrap();

    // Pobieranie uzytkownika
    let get_resp = client
        .get(format!("{base_url}/api/v1/users/{user_id}"))
        .send().await.unwrap();
    assert_eq!(get_resp.status(), 200);
}

testcontainers-rs — PostgreSQL w Docker

Crate testcontainers uruchamia kontenery Docker na czas testu i automatycznie je zatrzymuje po zakonczeniu. Idealny do testow z prawdziwa baza danych.

// [dev-dependencies]
// testcontainers = "0.21"
// testcontainers-modules = { version = "0.10", features = ["postgres"] }

use testcontainers::{runners::AsyncRunner, ImageExt};
use testcontainers_modules::postgres::Postgres;

#[tokio::test]
async fn test_with_real_postgres() {
    let container = Postgres::default()
        .with_db_name("testdb")
        .with_user("test")
        .with_password("test")
        .start()
        .await
        .unwrap();

    let port = container.get_host_port_ipv4(5432).await.unwrap();
    let db_url = format!("postgres://test:test@127.0.0.1:{port}/testdb");

    let pool = PgPoolOptions::new()
        .max_connections(5)
        .connect(&db_url).await.unwrap();

    // Uruchomienie migracji
    sqlx::migrate!("./migrations").run(&pool).await.unwrap();

    // Testy z prawdziwa baza
    let user_id = create_user(&pool, "Test User", "test@db.com").await.unwrap();
    let user = find_user_by_id(&pool, user_id).await.unwrap();
    assert!(user.is_some());
    assert_eq!(user.unwrap().email, "test@db.com");
    // kontener zatrzymuje sie automatycznie po wyjsciu ze scope
}

wiremock-rs — mockowanie zewnetrznych HTTP

Gdy twoje API wywoluje zewnetrzne serwisy (payment gateway, SMS, email), uzyj wiremock do mockowania tych polaczen w testach.

use wiremock::{MockServer, Mock, ResponseTemplate};
use wiremock::matchers::{method, path, header};

#[tokio::test]
async fn test_send_notification_calls_external_api() {
    let mock_server = MockServer::start().await;

    Mock::given(method("POST"))
        .and(path("/v1/messages"))
        .and(header("Authorization", "Bearer test_key"))
        .respond_with(
            ResponseTemplate::new(200)
                .set_body_json(serde_json::json!({"id": "msg_123", "status": "sent"}))
        )
        .expect(1)  // oczekujemy dokladnie 1 wywolania
        .mount(&mock_server)
        .await;

    let service = NotificationService::new(mock_server.uri(), "test_key");
    let result = service.send_notification("user@example.com", "Witaj!").await;

    assert!(result.is_ok());
    mock_server.verify().await;  // sprawdza czy expect(1) zostal spelniony
}

cargo nextest run

cargo nextest run --test '*'

cargo nextest run test_create_user

# Struktura projektu z testami integracyjnymi:
# my-api/
# ├── src/
# │   ├── main.rs
# │   ├── lib.rs          <-- eksportuj pub fn build_app()
# │   ├── handlers/
# │   └── services/
# │       └── user.rs     <-- #[cfg(test)] mod tests { ... }
# └── tests/
#     ├── common/
#     │   └── mod.rs      <-- wspolne helpery testowe
#     ├── users_test.rs
#     └── auth_test.rs
Dobra praktyka: Testy jednostkowe (#[cfg(test)] w module) testuja logike funkcji izolowanie. Testy w folderze tests/ sa testami integracyjnymi — testuja aplikacje jako calosc przez HTTP. Unikaj mockowania w testach integracyjnych; zamiast tego uzyj testcontainers z prawdziwa baza danych.
