jsonwebtoken — JWT encode/decode

Crate jsonwebtoken implementuje standard RFC 7519. Wspiera algorytmy symetryczne (HS256, HS512) i asymetryczne (RS256, ES256). Dla produkcji zalecany jest RS256 — klucz prywatny sluzy do podpisywania, publiczny do weryfikacji (mozna go udostepnic publicznie).

use jsonwebtoken::{encode, decode, Header, Algorithm, Validation, EncodingKey, DecodingKey};
use serde::{Serialize, Deserialize};
use std::time::{SystemTime, UNIX_EPOCH};

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct Claims {
    pub sub: String,          // subject (user id)
    pub email: String,
    pub role: String,
    pub exp: usize,           // expiration (Unix timestamp)
    pub iat: usize,           // issued at
    pub jti: String,          // JWT ID (do revokacji)
}

pub struct JwtService {
    encoding_key: EncodingKey,
    decoding_key: DecodingKey,
    validation: Validation,
}

impl JwtService {
    pub fn new_hs256(secret: &[u8]) -> Self {
        let mut validation = Validation::new(Algorithm::HS256);
        validation.leeway = 5; // 5 sekund tolerancji
        Self {
            encoding_key: EncodingKey::from_secret(secret),
            decoding_key: DecodingKey::from_secret(secret),
            validation,
        }
    }

    pub fn new_rs256(private_pem: &[u8], public_pem: &[u8]) -> anyhow::Result<Self> {
        Ok(Self {
            encoding_key: EncodingKey::from_rsa_pem(private_pem)?,
            decoding_key: DecodingKey::from_rsa_pem(public_pem)?,
            validation: Validation::new(Algorithm::RS256),
        })
    }

    pub fn sign(&self, user_id: &str, email: &str, role: &str) -> anyhow::Result<String> {
        let now = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs() as usize;
        let claims = Claims {
            sub: user_id.to_string(),
            email: email.to_string(),
            role: role.to_string(),
            iat: now,
            exp: now + 3600,  // 1 godzina
            jti: uuid::Uuid::new_v4().to_string(),
        };
        Ok(encode(&Header::new(Algorithm::HS256), &claims, &self.encoding_key)?)
    }

    pub fn verify(&self, token: &str) -> anyhow::Result<Claims> {
        let data = decode::<Claims>(token, &self.decoding_key, &self.validation)?;
        Ok(data.claims)
    }
}

JWT middleware dla Axum

Najczystszy sposob to implementacja FromRequestParts — Axum wywola go automatycznie gdy handler ma odpowiedni argument.

use axum::{extract::FromRequestParts, http::{request::Parts, StatusCode}};
use axum::async_trait;

pub struct AuthUser(pub Claims);

#[async_trait]
impl<S> FromRequestParts<S> for AuthUser
where
    S: Send + Sync,
{
    type Rejection = (StatusCode, axum::Json<serde_json::Value>);

    async fn from_request_parts(parts: &mut Parts, _state: &S) -> Result<Self, Self::Rejection> {
        let token = parts.headers
            .get("Authorization")
            .and_then(|v| v.to_str().ok())
            .and_then(|v| v.strip_prefix("Bearer "))
            .ok_or_else(|| (
                StatusCode::UNAUTHORIZED,
                axum::Json(serde_json::json!({"error": "Brak tokena"}))
            ))?;

        // JwtService pobrany z extensions (dodany przez middleware)
        let jwt = parts.extensions.get::<Arc<JwtService>>()
            .ok_or_else(|| (StatusCode::INTERNAL_SERVER_ERROR, axum::Json(serde_json::json!({}))))?;

        jwt.verify(token)
            .map(AuthUser)
            .map_err(|_| (
                StatusCode::UNAUTHORIZED,
                axum::Json(serde_json::json!({"error": "Nieprawidlowy token"}))
            ))
    }
}

// Uzycie w handlerze — Axum automatycznie weryfikuje JWT
async fn protected_handler(auth: AuthUser) -> impl IntoResponse {
    axum::Json(serde_json::json!({"user_id": auth.0.sub, "role": auth.0.role}))
}

Haszowanie hasel — argon2 i bcrypt

Nigdy nie przechowuj hasel w postaci plaintext ani jako MD5/SHA. Uzyj dedykowanych funkcji do haszowania hasel z wbudowanym soleniem.

// argon2 — ZALECANE (OWASP 2024)
use argon2::{Argon2, PasswordHash, PasswordHasher, PasswordVerifier};
use argon2::password_hash::{rand_core::OsRng, SaltString};

pub fn hash_password(password: &str) -> anyhow::Result<String> {
    let salt = SaltString::generate(&mut OsRng);
    let argon2 = Argon2::default();
    let hash = argon2
        .hash_password(password.as_bytes(), &salt)?
        .to_string();
    Ok(hash)
}

pub fn verify_password(password: &str, hash: &str) -> anyhow::Result<bool> {
    let parsed_hash = PasswordHash::new(hash)?;
    Ok(Argon2::default().verify_password(password.as_bytes(), &parsed_hash).is_ok())
}

// bcrypt — alternatywa
use bcrypt::{hash, verify, DEFAULT_COST};

pub fn hash_bcrypt(password: &str) -> anyhow::Result<String> {
    Ok(hash(password, DEFAULT_COST + 2)?)  // cost 14 dla lepszego bezpieczenstwa
}

pub fn verify_bcrypt(password: &str, hash: &str) -> anyhow::Result<bool> {
    Ok(verify(password, hash)?)
}

OAuth2 — PKCE flow z crate oauth2

Crate oauth2 implementuje RFC 6749 i RFC 7636 (PKCE). PKCE (Proof Key for Code Exchange) jest obowiazkowy dla aplikacji publicznych (SPA, mobile).

use oauth2::{
    AuthorizationCode, AuthUrl, ClientId, ClientSecret, CsrfToken,
    PkceCodeChallenge, RedirectUrl, Scope, TokenUrl,
    basic::BasicClient,
};

pub fn build_oauth_client() -> BasicClient {
    BasicClient::new(
        ClientId::new(std::env::var("OAUTH_CLIENT_ID").unwrap()),
        Some(ClientSecret::new(std::env::var("OAUTH_CLIENT_SECRET").unwrap())),
        AuthUrl::new("https://accounts.google.com/o/oauth2/v2/auth".to_string()).unwrap(),
        Some(TokenUrl::new("https://oauth2.googleapis.com/token".to_string()).unwrap()),
    )
    .set_redirect_uri(RedirectUrl::new(std::env::var("OAUTH_REDIRECT_URI").unwrap()).unwrap())
}

// Krok 1: Generowanie URL autoryzacji
pub async fn oauth_start(State(client): State<Arc<BasicClient>>) -> impl IntoResponse {
    let (pkce_challenge, pkce_verifier) = PkceCodeChallenge::new_random_sha256();
    let (auth_url, csrf_token) = client
        .authorize_url(CsrfToken::new_random)
        .add_scope(Scope::new("openid".to_string()))
        .add_scope(Scope::new("email".to_string()))
        .set_pkce_challenge(pkce_challenge)
        .url();

    // Zapisz csrf_token i pkce_verifier w sesji/redis
    axum::Json(serde_json::json!({"url": auth_url.to_string()}))
}

// Krok 2: Wymiana kodu na token
pub async fn oauth_callback(
    Query(params): Query<CallbackParams>,
    State(client): State<Arc<BasicClient>>,
) -> impl IntoResponse {
    // Weryfikacja csrf_token z sesji
    // Pobranie pkce_verifier z sesji
    let token = client
        .exchange_code(AuthorizationCode::new(params.code))
        .set_pkce_verifier(/* pkce_verifier z sesji */)
        .request_async(oauth2::reqwest::async_http_client)
        .await
        .unwrap();
    // Pobranie profilu uzytkownika i wystawienie wlasnego JWT
}

API Key extractor — FromRequestParts

API key to prostszy mechanizm dla serwis-do-serwis komunikacji. Mozna go odczytywac z naglowka X-API-Key lub query param ?api_key=.

pub struct ApiKey(pub String);

#[async_trait]
impl<S: Send + Sync> FromRequestParts<S> for ApiKey {
    type Rejection = (StatusCode, axum::Json<serde_json::Value>);

    async fn from_request_parts(parts: &mut Parts, _: &S) -> Result<Self, Self::Rejection> {
        // Proba z naglowka X-API-Key
        let key = parts.headers
            .get("X-API-Key")
            .and_then(|v| v.to_str().ok())
            .map(str::to_string);

        match key {
            Some(k) if !k.is_empty() => Ok(ApiKey(k)),
            _ => Err((
                StatusCode::UNAUTHORIZED,
                axum::Json(serde_json::json!({"error": "Wymagany naglowek X-API-Key"}))
            ))
        }
    }
}
Refresh tokeny: Przechowuj refresh tokeny w bazie danych (tabela refresh_tokens z polem revoked_at). Przy kazdym uzyciu sprawdzaj czy nie zostal uniewazniony. Access token powinien miec krotki czas zycia (15 min), refresh token dlugi (30 dni).
