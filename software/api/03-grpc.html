gRPC

Wysokowydajne RPC oparte na Protocol Buffers i HTTP/2 — typy serwisow, interceptory i obserwowalnosc.

Protocol Buffers (proto3)

Protocol Buffers (protobuf) to jezyk opisu interfejsu (IDL) i format serializacji danych opracowany przez Google. Jest znacznie szybszy i bardziej zwarty niz JSON — binarny format, silne typowanie, wsteczna kompatybilnosc przez numerowanie pol.

// user_service.proto
syntax = "proto3";

package user.v1;

option go_package = "github.com/example/api/user/v1;userv1";
option java_package = "com.example.user.v1";

import "google/protobuf/timestamp.proto";
import "google/protobuf/empty.proto";

// Enum
enum UserRole {
  USER_ROLE_UNSPECIFIED = 0;   // zawsze zarezerwuj 0 jako UNSPECIFIED
  USER_ROLE_USER        = 1;
  USER_ROLE_ADMIN       = 2;
  USER_ROLE_MODERATOR   = 3;
}

// Wiadomosc — odpowiednik struktury
message User {
  string                    id         = 1;
  string                    name       = 2;
  string                    email      = 3;
  UserRole                  role       = 4;
  google.protobuf.Timestamp created_at = 5;
  repeated string           tags       = 6;  // tablica stringow
  optional string           bio        = 7;  // pole opcjonalne (proto3)
}

message CreateUserRequest {
  string   name     = 1;
  string   email    = 2;
  string   password = 3;
  UserRole role     = 4;
}

message CreateUserResponse {
  User user = 1;
}

message GetUserRequest {
  string id = 1;
}

message ListUsersRequest {
  int32  page_size   = 1;
  string page_token  = 2;
  string filter      = 3;  // np. "role=ADMIN"
}

message ListUsersResponse {
  repeated User users         = 1;
  string        next_page_token = 2;
  int32         total_size    = 3;
}

message UpdateUserRequest {
  string id   = 1;
  User   user = 2;
  // FieldMask — aktualizuj tylko wybrane pola
  google.protobuf.FieldMask update_mask = 3;
}

// Definicja serwisu
service UserService {
  // Unary RPC — jeden request, jedna odpowiedz
  rpc GetUser    (GetUserRequest)    returns (User);
  rpc CreateUser (CreateUserRequest) returns (CreateUserResponse);
  rpc UpdateUser (UpdateUserRequest) returns (User);
  rpc DeleteUser (GetUserRequest)    returns (google.protobuf.Empty);

  // Server-streaming — jeden request, strumien odpowiedzi
  rpc ListUsers  (ListUsersRequest)  returns (stream User);

  // Client-streaming — strumien requestow, jedna odpowiedz
  rpc BulkCreateUsers (stream CreateUserRequest) returns (CreateUserResponse);

  // Bidirectional streaming
  rpc Chat (stream ChatMessage) returns (stream ChatMessage);
}

Typy serwisow

gRPC definiuje cztery typy komunikacji, w zaleznosci od tego, czy request i response sa pojedynczymi wiadomosciami czy strumieniami.

Typ Request Response Przyklad uzycia
Unary 1 wiadomosc 1 wiadomosc GetUser, CreateOrder — klasyczny call RPC
Server-side streaming 1 wiadomosc Strumien ListUsers, eksport danych, watch na zasob
Client-side streaming Strumien 1 wiadomosc BulkUpload, agregacja danych z klienta
Bidirectional streaming Strumien Strumien Chat, real-time updates, telemetria

Implementacja serwisu (Go)

// server/user_service.go
package server

import (
    "context"
    "time"

    "google.golang.org/grpc/codes"
    "google.golang.org/grpc/status"
    pb "github.com/example/api/user/v1"
)

type UserServiceServer struct {
    pb.UnimplementedUserServiceServer  // embed dla forward compatibility
    db UserRepository
}

// Unary RPC
func (s *UserServiceServer) GetUser(
    ctx context.Context,
    req *pb.GetUserRequest,
) (*pb.User, error) {
    if req.Id == "" {
        return nil, status.Error(codes.InvalidArgument, "id is required")
    }

    user, err := s.db.FindByID(ctx, req.Id)
    if err != nil {
        return nil, status.Errorf(codes.Internal, "database error: %v", err)
    }
    if user == nil {
        return nil, status.Errorf(codes.NotFound, "user %q not found", req.Id)
    }

    return userToProto(user), nil
}

// Server-side streaming
func (s *UserServiceServer) ListUsers(
    req *pb.ListUsersRequest,
    stream pb.UserService_ListUsersServer,
) error {
    ctx := stream.Context()

    users, err := s.db.List(ctx, req.PageSize, req.Filter)
    if err != nil {
        return status.Errorf(codes.Internal, "list failed: %v", err)
    }

    for _, u := range users {
        // Sprawdz czy klient nie anulowal polaczenia
        if err := ctx.Err(); err != nil {
            return status.FromContextError(err).Err()
        }
        if err := stream.Send(userToProto(u)); err != nil {
            return err
        }
    }
    return nil
}

// Client-side streaming
func (s *UserServiceServer) BulkCreateUsers(
    stream pb.UserService_BulkCreateUsersServer,
) error {
    var created int32
    for {
        req, err := stream.Recv()
        if err == io.EOF {
            return stream.SendAndClose(&pb.CreateUserResponse{
                // summary response
            })
        }
        if err != nil {
            return status.Errorf(codes.Internal, "recv error: %v", err)
        }
        if err := s.db.Create(stream.Context(), reqToUser(req)); err != nil {
            return status.Errorf(codes.Internal, "create failed: %v", err)
        }
        created++
    }
}

Deadlines i cancellation

Kazde wywolanie gRPC powinno miec deadline — maksymalny czas oczekiwania na odpowiedz. Bez deadline, zablokowane goroutines moga gromadzic sie i doprowadzic do wyczerpania zasobow. Cancellation propaguje sie przez caly lancuch wywolan.

Zawsze ustawiaj deadline

Brak deadline to jeden z najczestszych bledow w gRPC. Klient moze czekac w nieskonczonosc na odpowiedz niedostepnego serwera. Stosuj context.WithTimeout lub context.WithDeadline dla kazdego call.

// Klient Go z deadline i retry
conn, err := grpc.Dial(
    "user-service:50051",
    grpc.WithTransportCredentials(insecure.NewCredentials()),
    grpc.WithDefaultServiceConfig(`{
      "methodConfig": [{
        "name": [{"service": "user.v1.UserService"}],
        "retryPolicy": {
          "maxAttempts": 3,
          "initialBackoff": "0.1s",
          "maxBackoff": "1s",
          "backoffMultiplier": 2,
          "retryableStatusCodes": ["UNAVAILABLE", "RESOURCE_EXHAUSTED"]
        },
        "timeout": "5s"
      }]
    }`),
)

client := pb.NewUserServiceClient(conn)

// Deadline per call
ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
defer cancel()

user, err := client.GetUser(ctx, &pb.GetUserRequest{Id: "123"})
if err != nil {
    st, _ := status.FromError(err)
    switch st.Code() {
    case codes.NotFound:
        log.Println("user not found")
    case codes.DeadlineExceeded:
        log.Println("request timed out")
    case codes.Unauthenticated:
        log.Println("not authenticated")
    default:
        log.Printf("unexpected error: %v", err)
    }
}

gRPC-Web (przeglarka)

Przegladarki nie wspieraja HTTP/2 gRPC bezposrednio. gRPC-Web jest warstawa translacji — proxy (np. Envoy lub grpcwebproxy) konwertuje zapytania z przegladarki na natywne gRPC. Alternatywnie, ConnectRPC oferuje protokol kompatybilny jednoczesnie z gRPC i REST/JSON.

Interceptory

Interceptory to middleware dla gRPC — mozna je dodac po stronie serwera i klienta. Sluza do logowania, autentykacji, rate limitingu, tracingu i obslugi bledow.

// Interceptor logowania (Go)
func LoggingInterceptor(
    ctx context.Context,
    req interface{},
    info *grpc.UnaryServerInfo,
    handler grpc.UnaryHandler,
) (interface{}, error) {
    start := time.Now()

    // Wyciagnij metadane
    md, _ := metadata.FromIncomingContext(ctx)
    requestID := md.Get("x-request-id")

    resp, err := handler(ctx, req)

    duration := time.Since(start)
    st, _ := status.FromError(err)

    log.Printf(
        "method=%s duration=%s code=%s request_id=%v",
        info.FullMethod, duration, st.Code(), requestID,
    )

    return resp, err
}

// Interceptor autentykacji
func AuthInterceptor(
    ctx context.Context,
    req interface{},
    info *grpc.UnaryServerInfo,
    handler grpc.UnaryHandler,
) (interface{}, error) {
    md, ok := metadata.FromIncomingContext(ctx)
    if !ok {
        return nil, status.Error(codes.Unauthenticated, "missing metadata")
    }

    tokens := md.Get("authorization")
    if len(tokens) == 0 {
        return nil, status.Error(codes.Unauthenticated, "missing token")
    }

    claims, err := validateJWT(tokens[0])
    if err != nil {
        return nil, status.Errorf(codes.Unauthenticated, "invalid token: %v", err)
    }

    ctx = context.WithValue(ctx, userClaimsKey, claims)
    return handler(ctx, req)
}

// Rejestracja interceptorow
server := grpc.NewServer(
    grpc.ChainUnaryInterceptor(
        LoggingInterceptor,
        AuthInterceptor,
        RecoveryInterceptor,
    ),
    grpc.ChainStreamInterceptor(
        StreamLoggingInterceptor,
        StreamAuthInterceptor,
    ),
)

Obserwowalnosc — OpenTelemetry

gRPC integruje sie z OpenTelemetry przez interceptory. Automatycznie propaguje trace context przez metadane gRPC, dodaje spany dla kazdego call i eksportuje metryki (latency, error rate, throughput).

Kody statusow gRPC

gRPC ma wlasny zestaw kodow statusow (codes.OK, codes.NotFound, codes.InvalidArgument, codes.Internal itd.) niezalezny od HTTP. Mapowanie na HTTP odbywa sie w warstwie gRPC-Web lub transcoding.
