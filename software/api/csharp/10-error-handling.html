Error handling i ProblemDetails

IExceptionHandler, UseExceptionHandler, RFC 7807 ProblemDetails, niestandardowe typy bledow i logowanie

Strategia obslugi wyjatkow

Dobra strategia obslugi bledow w API powinna: zwracac spójne, ustandaryzowane odpowiedzi (RFC 7807 ProblemDetails), logowac wystarczajaco duzo kontekstu do diagnozy, ale nie ujawniac szczegolów implementacyjnych klientom zewnetrznym.

Niestandardowe typy wyjatkow domenowych

// Bazowy wyjatek domenowy
public abstract class DomainException : Exception
{
    public string ErrorCode { get; }
    public int StatusCode { get; }

    protected DomainException(
        string message,
        string errorCode,
        int statusCode = 400)
        : base(message)
    {
        ErrorCode = errorCode;
        StatusCode = statusCode;
    }
}

// Konkretne wyjatki
public class NotFoundException : DomainException
{
    public NotFoundException(string resource, object id)
        : base($"{resource} o Id={id} nie znaleziono",
               "NOT_FOUND", 404) { }
}

public class ConflictException : DomainException
{
    public ConflictException(string message)
        : base(message, "CONFLICT", 409) { }
}

public class BusinessRuleException : DomainException
{
    public BusinessRuleException(string rule, string detail)
        : base(detail, rule, 422) { }
}

public class ForbiddenException : DomainException
{
    public ForbiddenException(string message = "Brak uprawnien")
        : base(message, "FORBIDDEN", 403) { }
}

// Uzycie w serwisie
public async Task<Product> GetByIdAsync(int id, CancellationToken ct)
{
    var product = await _repo.GetByIdAsync(id, ct);
    if (product is null)
        throw new NotFoundException("Produkt", id);
    return product;
}

public async Task<Order> PlaceOrderAsync(
    PlaceOrderRequest req, CancellationToken ct)
{
    var customer = await _customerRepo.GetByIdAsync(req.CustomerId, ct)
        ?? throw new NotFoundException("Klient", req.CustomerId);

    if (customer.IsBlocked)
        throw new BusinessRuleException(
            "CUSTOMER_BLOCKED",
            "Konto klienta jest zablokowane i nie moze skladac zamowien");

    var product = await _productRepo.GetByIdAsync(req.ProductId, ct)
        ?? throw new NotFoundException("Produkt", req.ProductId);

    if (product.Stock < req.Quantity)
        throw new BusinessRuleException(
            "INSUFFICIENT_STOCK",
            $"Dostepny stan: {product.Stock}, zamowiono: {req.Quantity}");

    return await _orderRepo.CreateAsync(req, ct);
}

IExceptionHandler — .NET 8+

IExceptionHandler to interfejs dostepny od .NET 8, ktory zastepuje starsze podejscia. Mozna laczyc wiele handlerow w lancuch.

// Handler dla wyjatkow domenowych
public class DomainExceptionHandler : IExceptionHandler
{
    private readonly ILogger<DomainExceptionHandler> _logger;
    private readonly IProblemDetailsService _problemDetails;

    public DomainExceptionHandler(
        ILogger<DomainExceptionHandler> logger,
        IProblemDetailsService problemDetails)
    {
        _logger = logger;
        _problemDetails = problemDetails;
    }

    public async ValueTask<bool> TryHandleAsync(
        HttpContext httpContext,
        Exception exception,
        CancellationToken cancellationToken)
    {
        if (exception is not DomainException domainEx)
            return false;  // Przekaz do nastepnego handlera

        _logger.LogWarning(
            "Wyjatek domenowy {ErrorCode}: {Message}",
            domainEx.ErrorCode, domainEx.Message);

        httpContext.Response.StatusCode = domainEx.StatusCode;

        await _problemDetails.WriteAsync(new ProblemDetailsContext
        {
            HttpContext = httpContext,
            ProblemDetails =
            {
                Status = domainEx.StatusCode,
                Title = GetTitleForStatus(domainEx.StatusCode),
                Detail = domainEx.Message,
                Type = $"https://myapi.example.com/errors/{domainEx.ErrorCode.ToLower()}",
                Extensions =
                {
                    ["errorCode"] = domainEx.ErrorCode,
                    ["traceId"] = httpContext.TraceIdentifier
                }
            }
        });

        return true;  // Obsluzono
    }

    private static string GetTitleForStatus(int status) => status switch
    {
        400 => "Nieprawidlowe zadanie",
        403 => "Dostep zabroniony",
        404 => "Nie znaleziono zasobu",
        409 => "Konflikt danych",
        422 => "Naruszenie reguly biznesowej",
        _   => "Blad"
    };
}

// Handler dla nieobsluzonych wyjatkow (fallback)
public class GlobalExceptionHandler : IExceptionHandler
{
    private readonly ILogger<GlobalExceptionHandler> _logger;
    private readonly IProblemDetailsService _problemDetails;

    public GlobalExceptionHandler(
        ILogger<GlobalExceptionHandler> logger,
        IProblemDetailsService problemDetails)
    {
        _logger = logger;
        _problemDetails = problemDetails;
    }

    public async ValueTask<bool> TryHandleAsync(
        HttpContext httpContext,
        Exception exception,
        CancellationToken cancellationToken)
    {
        _logger.LogError(exception,
            "Nieobsluzony wyjatek dla {Method} {Path}",
            httpContext.Request.Method,
            httpContext.Request.Path);

        httpContext.Response.StatusCode =
            StatusCodes.Status500InternalServerError;

        await _problemDetails.WriteAsync(new ProblemDetailsContext
        {
            HttpContext = httpContext,
            Exception = exception,
            ProblemDetails =
            {
                Status = 500,
                Title = "Wewnetrzny blad serwera",
                Type = "https://tools.ietf.org/html/rfc9110#section-15.6.1",
                Extensions =
                {
                    ["traceId"] = httpContext.TraceIdentifier
                }
            }
        });

        return true;
    }
}

// Rejestracja w Program.cs
builder.Services.AddProblemDetails();
builder.Services.AddExceptionHandler<DomainExceptionHandler>();
builder.Services.AddExceptionHandler<GlobalExceptionHandler>();

app.UseExceptionHandler();  // aktywuje pipeline IExceptionHandler

ProblemDetails Factory — niestandardowe formatowanie

// Niestandardowa fabryka ProblemDetails
public class CustomProblemDetailsFactory : ProblemDetailsFactory
{
    private readonly ApiBehaviorOptions _options;

    public CustomProblemDetailsFactory(
        IOptions<ApiBehaviorOptions> options)
    {
        _options = options.Value;
    }

    public override ProblemDetails CreateProblemDetails(
        HttpContext httpContext,
        int? statusCode = null,
        string? title = null,
        string? type = null,
        string? detail = null,
        string? instance = null)
    {
        var problem = new ProblemDetails
        {
            Status = statusCode ?? 500,
            Title = title,
            Type = type,
            Detail = detail,
            Instance = instance ?? httpContext.Request.Path
        };

        problem.Extensions["traceId"] = httpContext.TraceIdentifier;
        problem.Extensions["timestamp"] =
            DateTimeOffset.UtcNow.ToString("O");

        _options.ClientErrorMapping.TryGetValue(
            problem.Status.Value, out var mapping);

        problem.Title ??= mapping?.Title;
        problem.Type ??= mapping?.Link;

        return problem;
    }

    public override ValidationProblemDetails CreateValidationProblemDetails(
        HttpContext httpContext,
        ModelStateDictionary modelStateDictionary,
        int? statusCode = null,
        string? title = null,
        string? type = null,
        string? detail = null,
        string? instance = null)
    {
        var problem = new ValidationProblemDetails(modelStateDictionary)
        {
            Status = statusCode ?? 422,
            Title = title ?? "Blad walidacji danych",
            Type = type ?? "https://myapi.example.com/errors/validation",
            Detail = detail,
            Instance = instance ?? httpContext.Request.Path
        };

        problem.Extensions["traceId"] = httpContext.TraceIdentifier;
        return problem;
    }
}

// Rejestracja
builder.Services.AddSingleton<ProblemDetailsFactory,
    CustomProblemDetailsFactory>();

Rozroznienie srodowisk — dev vs prod

// Program.cs — rozna konfiguracja per srodowisko
if (app.Environment.IsDevelopment())
{
    // Szczegolowe informacje o wyjatku (stack trace, inner exception)
    app.UseDeveloperExceptionPage();
}
else
{
    // Produkcja — tylko bezpieczne ProblemDetails bez stack trace
    app.UseExceptionHandler();
    app.UseHsts();
}

// Warunkowe logowanie — wiecej szczegolow w dev
builder.Logging.AddFilter("Microsoft.AspNetCore", level =>
    app.Environment.IsDevelopment()
        ? level >= LogLevel.Debug
        : level >= LogLevel.Warning);

// Structured logging z enricherem srodowiska
Log.Logger = new LoggerConfiguration()
    .Enrich.WithProperty("Environment",
        builder.Environment.EnvironmentName)
    .Enrich.WithProperty("Application", "MyApi")
    .WriteTo.Console(new JsonFormatter())
    .WriteTo.Conditional(
        e => !builder.Environment.IsDevelopment(),
        s => s.Seq("http://seq:5341"))
    .CreateLogger();

Zasada bezpieczenstwa: Nigdy nie ujawniaj stack trace, nazw wewnetrznych klas ani szczegolów bazy danych w odpowiedziach produkcyjnych. Uzywaj traceId — klient zglosic moze incydent podajac ten identyfikator, a Ty znajdziesz pelne szczegoly w logach.
