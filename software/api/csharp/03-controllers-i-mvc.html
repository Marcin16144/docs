[ApiController] — co robi ten atrybut?

Atrybut [ApiController] wlacza zestaw konwencji API, ktore zmniejszaja ilosc kodu plytki. Bez niego musisz recznie sprawdzac ModelState, bindowac cialo zadania atrybutem i inne.

Funkcja wlaczona przez [ApiController]Opis
Automatyczna walidacja ModelStateZwraca 400 ValidationProblem jesli ModelState jest nieprawidlowy
Inferowanie [FromBody]Parametry zlozonych typow sa bindowane z ciala JSON automatycznie
Inferowanie [FromRoute]Parametry zgodne z segmentem trasy sa bindowane z URL
Multipart/form-data bindingIFormFile bindowane automatycznie
ProblemDetails dla bledow klientaOdpowiedzi 4xx zwracaja RFC 7807 ProblemDetails

Attribute routing

[ApiController]
[Route("api/v{version:apiVersion}/[controller]")]
[Produces("application/json")]
public class ProductsController : ControllerBase
{
    private readonly IProductService _service;
    private readonly ILogger<ProductsController> _logger;

    public ProductsController(
        IProductService service,
        ILogger<ProductsController> logger)
    {
        _service = service;
        _logger = logger;
    }

    // GET api/v1/products
    [HttpGet]
    [ProducesResponseType(typeof(PagedResult<ProductDto>), 200)]
    public async Task<ActionResult<PagedResult<ProductDto>>> GetAll(
        [FromQuery] ProductQueryParams queryParams,
        CancellationToken ct)
    {
        var result = await _service.GetPagedAsync(queryParams, ct);
        return Ok(result);
    }

    // GET api/v1/products/42
    [HttpGet("{id:int}", Name = "GetProductById")]
    [ProducesResponseType(typeof(ProductDto), 200)]
    [ProducesResponseType(typeof(ProblemDetails), 404)]
    public async Task<ActionResult<ProductDto>> GetById(int id, CancellationToken ct)
    {
        var product = await _service.GetByIdAsync(id, ct);
        if (product is null)
            return NotFound(new ProblemDetails
            {
                Title = "Produkt nie znaleziony",
                Detail = $"Produkt o Id={id} nie istnieje.",
                Status = 404
            });
        return Ok(product);
    }

    // POST api/v1/products
    [HttpPost]
    [Authorize]
    [ProducesResponseType(typeof(ProductDto), 201)]
    [ProducesResponseType(typeof(ValidationProblemDetails), 400)]
    public async Task<ActionResult<ProductDto>> Create(
        [FromBody] CreateProductRequest request,
        CancellationToken ct)
    {
        var product = await _service.CreateAsync(request, ct);
        return CreatedAtRoute("GetProductById",
            new { id = product.Id }, product);
    }

    // PUT api/v1/products/42
    [HttpPut("{id:int}")]
    [Authorize]
    public async Task<ActionResult<ProductDto>> Update(
        int id,
        [FromBody] UpdateProductRequest request,
        CancellationToken ct)
    {
        var product = await _service.UpdateAsync(id, request, ct);
        return product is null ? NotFound() : Ok(product);
    }

    // DELETE api/v1/products/42
    [HttpDelete("{id:int}")]
    [Authorize(Roles = "Admin")]
    [ProducesResponseType(204)]
    [ProducesResponseType(404)]
    public async Task<IActionResult> Delete(int id, CancellationToken ct)
    {
        var deleted = await _service.DeleteAsync(id, ct);
        return deleted ? NoContent() : NotFound();
    }
}

Model binding — zlozzone typy

Model binding w kontrolerach automatycznie mapuje dane z roznych zrodel HTTP na parametry akcji. Dla zlozonych obiektow uzywane jest cialo zadania (JSON), dla prostych typow — query string lub route.

// Zbiorczy parametr query z walidacja
public class ProductQueryParams
{
    [Range(1, int.MaxValue, ErrorMessage = "Strona musi byc >= 1")]
    public int Page { get; set; } = 1;

    [Range(1, 100, ErrorMessage = "Rozmiar strony musi byc od 1 do 100")]
    public int PageSize { get; set; } = 20;

    [MaxLength(100)]
    public string? Search { get; set; }

    public ProductSortField SortBy { get; set; } = ProductSortField.Name;
    public bool SortDescending { get; set; } = false;
}

// Cialo zadania (automatycznie [FromBody] dzieki [ApiController])
public record CreateProductRequest(
    [Required][StringLength(200)] string Name,
    [Required][Range(0.01, 999999)] decimal Price,
    [Required][Range(0, 100000)] int Stock,
    [StringLength(1000)] string? Description
);

// Mieszane zrodla
[HttpPost("{categoryId:int}/products")]
public async Task<IActionResult> CreateInCategory(
    [FromRoute] int categoryId,          // z URL
    [FromBody] CreateProductRequest req, // z JSON body
    [FromHeader(Name = "X-Trace-Id")] string? traceId,  // z naglowka
    [FromServices] ILogger<ProductsController> logger)  // z DI
{
    logger.LogInformation("Trace: {TraceId}", traceId);
    var product = await _service.CreateInCategoryAsync(categoryId, req);
    return CreatedAtAction(nameof(GetById), new { id = product.Id }, product);
}

ActionResult<T> vs IActionResult

ActionResult<T> jest preferowany — pozwala OpenAPI automatycznie wykryc typ odpowiedzi sukcesu. IActionResult uzywaj gdy akcja moze zwracac wiele roznych typow sukcesu (rzadkie). 
// Preferowany sposob — ActionResult<T>
[HttpGet("{id:int}")]
public async Task<ActionResult<ProductDto>> GetById(int id)
{
    var p = await _service.GetByIdAsync(id);
    // Niejawna konwersja: return p; == return Ok(p);
    return p is null ? NotFound() : p;
}

// Niestandardowy ActionResult
public class CsvResult : IActionResult
{
    private readonly IEnumerable<object> _data;
    private readonly string _filename;

    public CsvResult(IEnumerable<object> data, string filename)
    {
        _data = data;
        _filename = filename;
    }

    public async Task ExecuteResultAsync(ActionContext context)
    {
        var response = context.HttpContext.Response;
        response.ContentType = "text/csv";
        response.Headers.ContentDisposition = $"attachment; filename={_filename}";
        await using var writer = new StreamWriter(response.Body);
        foreach (var row in _data)
            await writer.WriteLineAsync(row.ToString());
    }
}

// Uzycie
[HttpGet("export")]
public IActionResult ExportCsv()
{
    var data = _service.GetAllForExport();
    return new CsvResult(data, "products.csv");
}

ProblemDetails — RFC 7807

ProblemDetails to ustandaryzowany format odpowiedzi bledow (RFC 7807). Od .NET 7 ASP.NET Core uzywa go domyslnie dla wszystkich odpowiedzi 4xx/5xx.

// Wlaczenie ProblemDetails (Program.cs)
builder.Services.AddProblemDetails(options =>
{
    options.CustomizeProblemDetails = ctx =>
    {
        ctx.ProblemDetails.Extensions["traceId"] =
            ctx.HttpContext.TraceIdentifier;
        ctx.ProblemDetails.Extensions["instance"] =
            $"{ctx.HttpContext.Request.Method} {ctx.HttpContext.Request.Path}";
    };
});

// Przyklad odpowiedzi ProblemDetails
// HTTP 404
{
  "type": "https://tools.ietf.org/html/rfc9110#section-15.5.5",
  "title": "Not Found",
  "status": 404,
  "detail": "Produkt o Id=99 nie istnieje.",
  "instance": "GET /api/v1/products/99",
  "traceId": "00-4f12ab...3c-00"
}

// Rzucanie ProblemDetails z dowolnego miejsca
throw new HttpRequestException("Zewnetrzny serwis niedostepny",
    inner: null, statusCode: HttpStatusCode.BadGateway);

Action Filters

Filtry akcji dzialaja podobnie do middleware, ale na poziomie kontrolera lub akcji. Umozliwiaja przechwytywanie przed i po wykonaniu akcji.

// Synchroniczny filter akcji
public class LogActionFilter : IActionFilter
{
    private readonly ILogger<LogActionFilter> _logger;

    public LogActionFilter(ILogger<LogActionFilter> logger)
    {
        _logger = logger;
    }

    public void OnActionExecuting(ActionExecutingContext context)
    {
        _logger.LogInformation(
            "Wywolanie akcji {Action} z parametrami {@Params}",
            context.ActionDescriptor.DisplayName,
            context.ActionArguments);
    }

    public void OnActionExecuted(ActionExecutedContext context)
    {
        if (context.Exception is not null)
            _logger.LogError(context.Exception,
                "Akcja {Action} rzucila wyjatek",
                context.ActionDescriptor.DisplayName);
    }
}

// Asynchroniczny filter — preferowany dla operacji async
public class PerformanceFilter : IAsyncActionFilter
{
    private readonly ILogger<PerformanceFilter> _logger;

    public PerformanceFilter(ILogger<PerformanceFilter> logger)
    {
        _logger = logger;
    }

    public async Task OnActionExecutionAsync(
        ActionExecutingContext context,
        ActionExecutionDelegate next)
    {
        var sw = Stopwatch.StartNew();
        var executed = await next();
        sw.Stop();

        if (sw.ElapsedMilliseconds > 500)
            _logger.LogWarning(
                "Wolna akcja {Action}: {Ms}ms",
                context.ActionDescriptor.DisplayName,
                sw.ElapsedMilliseconds);
    }
}

// Rejestracja globalna
builder.Services.AddControllers(options =>
{
    options.Filters.Add<LogActionFilter>();
    options.Filters.Add<PerformanceFilter>();
});

// Rejestracja na poziomie kontrolera lub akcji
[ServiceFilter(typeof(LogActionFilter))]
public class ProductsController : ControllerBase { }

[HttpGet]
[ServiceFilter(typeof(PerformanceFilter))]
public async Task<IActionResult> GetAll() { ... }

Response Compression i Output Caching

// Response compression
builder.Services.AddResponseCompression(options =>
{
    options.EnableForHttps = true;
    options.Providers.Add<BrotliCompressionProvider>();
    options.Providers.Add<GzipCompressionProvider>();
    options.MimeTypes = ResponseCompressionDefaults.MimeTypes
        .Concat(new[] { "application/json" });
});
builder.Services.Configure<BrotliCompressionProviderOptions>(o =>
    o.Level = CompressionLevel.Optimal);

app.UseResponseCompression();

// Output Caching (.NET 7+)
builder.Services.AddOutputCache(options =>
{
    options.AddBasePolicy(b => b.Cache());
    options.AddPolicy("Products", b => b
        .Cache()
        .Expire(TimeSpan.FromMinutes(5))
        .SetVaryByQuery("page", "size", "search")
        .Tag("products"));
});

app.UseOutputCache();

// Uzywanie na akcji
[HttpGet]
[OutputCache(PolicyName = "Products")]
public async Task<ActionResult<PagedResult<ProductDto>>> GetAll(...)
{ ... }

// Inwaliacja cache po zmianie
[HttpPost]
public async Task<ActionResult<ProductDto>> Create(
    CreateProductRequest req,
    IOutputCacheStore cache,
    CancellationToken ct)
{
    var product = await _service.CreateAsync(req, ct);
    await cache.EvictByTagAsync("products", ct);
    return CreatedAtRoute("GetProductById", new { id = product.Id }, product);
}
Wskazowka: Output Cache w .NET 9 wspiera distributed cache (Redis). Uzyj AddOutputCache() z AddStackExchangeRedisOutputCache() dla srodowisk wieloinstancyjnych.
