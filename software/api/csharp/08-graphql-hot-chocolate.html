GraphQL z Hot Chocolate v15

Setup, queries, mutations, subscriptions, DataLoader, filtering/sorting/paging i Federation

Czym jest Hot Chocolate?

Hot Chocolate to wiodaca implementacja serwera GraphQL dla .NET, rozwijana przez ChilliCream. Wersja 15 (2026) wprowadza ulepszone Federation v2, Fusion (distributed GraphQL) i usprawniony model konfiguracji.

GraphQL vs REST — kiedy co wybrac:
  • GraphQL: Elastyczne pobieranie danych, wiele klientow o roznych potrzebach, agregacja wielu zrodel, real-time subscriptions
  • REST: Proste CRUD API, caching HTTP, publiczne API z ustabilizowanym kontraktem

Setup — konfiguracja Hot Chocolate v15

// dotnet add package HotChocolate.AspNetCore
// dotnet add package HotChocolate.Data
// dotnet add package HotChocolate.Data.EntityFramework

// Program.cs
builder.Services
    .AddGraphQLServer()
    .AddQueryType<Query>()
    .AddMutationType<Mutation>()
    .AddSubscriptionType<Subscription>()
    .AddTypeExtension<ProductTypeExtension>()
    .AddFiltering()
    .AddSorting()
    .AddProjections()
    .AddPagingArguments()
    .RegisterDbContext<AppDbContext>()
    .AddInMemorySubscriptions()       // lub .AddRedisSubscriptions()
    .AddAuthorization();

var app = builder.Build();
app.UseWebSockets();  // wymagane dla subscriptions
app.MapGraphQL("/graphql");
app.MapBananaCakePop("/graphql-ui");  // Banana Cake Pop UI
app.Run();

Query — pobieranie danych

// Typ Query — annotation-based (code-first)
[QueryType]
public class Query
{
    // Proste zapytanie z filtrowaniem, sortowaniem i stronicowaniem
    [UsePaging(IncludeTotalCount = true)]
    [UseProjection]
    [UseFiltering]
    [UseSorting]
    public IQueryable<Product> GetProducts(AppDbContext db)
        => db.Products.AsNoTracking();

    // Zapytanie po Id — zwraca null jesli nie istnieje
    [UseProjection]
    public Task<Product?> GetProductAsync(
        int id,
        AppDbContext db,
        CancellationToken ct)
        => db.Products.FirstOrDefaultAsync(p => p.Id == id, ct);

    // Zapytanie z DataLoader
    public Task<Product?> GetProductByIdAsync(
        int id,
        ProductByIdDataLoader loader,
        CancellationToken ct)
        => loader.LoadAsync(id, ct);

    // Zapytanie z paginacja bazujaca na kursorze
    [UsePaging(MaxPageSize = 50, DefaultPageSize = 20)]
    [UseFiltering]
    [UseSorting]
    public IQueryable<Order> GetOrders(
        [Service] AppDbContext db,
        [HotChocolate.Data.Filters.Filter] Expression<Func<Order, bool>>? where)
        => db.Orders.AsNoTracking();
}

DataLoader — rozwiazanie problemu N+1

DataLoader grupuje (batch) zapytania do bazy danych, eliminujac problem N+1 zapytan. Zamiast N osobnych zapytan dla N obiektow — jedno zapytanie WHERE Id IN (...).

// DataLoader dla produktow
public class ProductByIdDataLoader
    : BatchDataLoader<int, Product>
{
    private readonly IDbContextFactory<AppDbContext> _factory;

    public ProductByIdDataLoader(
        IDbContextFactory<AppDbContext> factory,
        IBatchScheduler scheduler,
        DataLoaderOptions options)
        : base(scheduler, options)
    {
        _factory = factory;
    }

    protected override async Task<IReadOnlyDictionary<int, Product>>
        LoadBatchAsync(
            IReadOnlyList<int> keys,
            CancellationToken ct)
    {
        await using var db = await _factory.CreateDbContextAsync(ct);
        return await db.Products
            .AsNoTracking()
            .Where(p => keys.Contains(p.Id))
            .ToDictionaryAsync(p => p.Id, ct);
    }
}

// DataLoader dla kolekcji (produkty danej kategorii)
public class ProductsByCategoryDataLoader
    : GroupedDataLoader<int, Product>
{
    private readonly IDbContextFactory<AppDbContext> _factory;

    public ProductsByCategoryDataLoader(
        IDbContextFactory<AppDbContext> factory,
        IBatchScheduler scheduler,
        DataLoaderOptions options)
        : base(scheduler, options)
    {
        _factory = factory;
    }

    protected override async Task<ILookup<int, Product>>
        LoadGroupedBatchAsync(
            IReadOnlyList<int> categoryIds,
            CancellationToken ct)
    {
        await using var db = await _factory.CreateDbContextAsync(ct);
        var products = await db.Products
            .AsNoTracking()
            .Where(p => categoryIds.Contains(p.CategoryId))
            .ToListAsync(ct);
        return products.ToLookup(p => p.CategoryId);
    }
}

Mutation — modyfikacja danych

// Input types dla mutacji
public record CreateProductInput(
    string Name,
    decimal Price,
    int Stock,
    int CategoryId,
    string? Description);

public record UpdateProductInput(
    int Id,
    Optional<string> Name,
    Optional<decimal> Price,
    Optional<int> Stock);

// Payload types (wzorzec "mutation payload")
public record CreateProductPayload(Product? Product, UserError? Error);
public record UpdateProductPayload(Product? Product, UserError? Error);
public record UserError(string Message, string Code);

// Typ Mutation
[MutationType]
public class Mutation
{
    public async Task<CreateProductPayload> CreateProductAsync(
        CreateProductInput input,
        IProductService service,
        IValidator<CreateProductInput> validator,
        CancellationToken ct)
    {
        var validation = await validator.ValidateAsync(input, ct);
        if (!validation.IsValid)
            return new CreateProductPayload(null,
                new UserError(
                    string.Join("; ", validation.Errors.Select(e => e.ErrorMessage)),
                    "VALIDATION_ERROR"));

        var product = await service.CreateAsync(input, ct);
        return new CreateProductPayload(product, null);
    }

    [Authorize(Policy = "AdminOnly")]
    public async Task<UpdateProductPayload> UpdateProductAsync(
        UpdateProductInput input,
        IProductService service,
        CancellationToken ct)
    {
        var product = await service.UpdateAsync(input, ct);
        return product is null
            ? new UpdateProductPayload(null,
                new UserError($"Produkt {input.Id} nie istnieje", "NOT_FOUND"))
            : new UpdateProductPayload(product, null);
    }

    [Authorize(Policy = "AdminOnly")]
    public async Task<bool> DeleteProductAsync(
        int id,
        IProductService service,
        CancellationToken ct)
        => await service.DeleteAsync(id, ct);
}

Subscription — real-time

// Typ Subscription
[SubscriptionType]
public class Subscription
{
    [Subscribe]
    [Topic("ProductCreated")]
    public Product OnProductCreated(
        [EventMessage] Product product)
        => product;

    [Subscribe]
    [Topic("{categoryId}")]
    public Product OnProductUpdatedInCategory(
        int categoryId,
        [EventMessage] Product product)
        => product;
}

// Publikowanie zdarzen w Mutation / serwisie
public class ProductService : IProductService
{
    private readonly ITopicEventSender _events;

    public ProductService(ITopicEventSender events)
    {
        _events = events;
    }

    public async Task<Product> CreateAsync(
        CreateProductInput input,
        CancellationToken ct)
    {
        // ... zapis do bazy ...
        var product = new Product { /* ... */ };

        // Wyslij zdarzenie do subskrybentow
        await _events.SendAsync("ProductCreated", product, ct);
        await _events.SendAsync(
            $"{product.CategoryId}", product, ct);  // temat per kategoria

        return product;
    }
}

Type Extensions — rozszerzanie typow

// Rozszerzenie encji Product o pole obliczane i relacje przez DataLoader
[ExtendObjectType<Product>]
public class ProductTypeExtension
{
    // Pole obliczane
    public string GetDisplayPrice([Parent] Product product)
        => $"{product.Price:C2}";

    // Relacja przez DataLoader (rozwiazuje N+1)
    public Task<Category?> GetCategoryAsync(
        [Parent] Product product,
        CategoryByIdDataLoader loader,
        CancellationToken ct)
        => loader.LoadAsync(product.CategoryId, ct);

    // Pole chronione
    [Authorize]
    public async Task<int> GetOrderCountAsync(
        [Parent] Product product,
        AppDbContext db,
        CancellationToken ct)
        => await db.OrderItems
            .CountAsync(i => i.ProductId == product.Id, ct);
}
