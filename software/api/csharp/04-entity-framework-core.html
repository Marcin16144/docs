DbContext — konfiguracja

DbContext jest centralnym punktem EF Core — reprezentuje sesje z baza danych. Kazda klasa encji mapuje sie na tabele. Konfiguracje mozna definiowac przez Data Annotations lub Fluent API (preferowane dla zlozonych przypadkow).

// Encje
public class Product
{
    public int Id { get; set; }
    public string Name { get; set; } = string.Empty;
    public decimal Price { get; set; }
    public int Stock { get; set; }
    public bool IsActive { get; set; } = true;
    public DateTime CreatedAt { get; set; }
    public int CategoryId { get; set; }
    public Category Category { get; set; } = null!;
    public ICollection<OrderItem> OrderItems { get; set; } = [];
    public ProductDetails? Details { get; set; }   // JSON column (EF9)
}

public class ProductDetails  // Complex type / JSON column
{
    public string? Manufacturer { get; set; }
    public string? Sku { get; set; }
    public Dictionary<string, string> Attributes { get; set; } = [];
}

// DbContext
public class AppDbContext : DbContext
{
    public AppDbContext(DbContextOptions<AppDbContext> options)
        : base(options) { }

    public DbSet<Product> Products => Set<Product>();
    public DbSet<Category> Categories => Set<Category>();
    public DbSet<Order> Orders => Set<Order>();
    public DbSet<OrderItem> OrderItems => Set<OrderItem>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.ApplyConfigurationsFromAssembly(
            Assembly.GetExecutingAssembly());
    }

    public override Task<int> SaveChangesAsync(
        CancellationToken ct = default)
    {
        // Auto-set CreatedAt / UpdatedAt
        foreach (var entry in ChangeTracker.Entries<IAuditableEntity>())
        {
            if (entry.State == EntityState.Added)
                entry.Entity.CreatedAt = DateTime.UtcNow;
            if (entry.State is EntityState.Added or EntityState.Modified)
                entry.Entity.UpdatedAt = DateTime.UtcNow;
        }
        return base.SaveChangesAsync(ct);
    }
}

// Konfiguracja encji (IEntityTypeConfiguration)
public class ProductConfiguration : IEntityTypeConfiguration<Product>
{
    public void Configure(EntityTypeBuilder<Product> builder)
    {
        builder.HasKey(p => p.Id);
        builder.Property(p => p.Name)
            .IsRequired()
            .HasMaxLength(200);
        builder.Property(p => p.Price)
            .HasColumnType("decimal(18,2)");
        builder.HasIndex(p => p.Name);
        builder.HasQueryFilter(p => p.IsActive);  // global filter

        // JSON column (EF Core 9)
        builder.OwnsOne(p => p.Details, details =>
        {
            details.ToJson();
        });

        builder.HasOne(p => p.Category)
            .WithMany(c => c.Products)
            .HasForeignKey(p => p.CategoryId)
            .OnDelete(DeleteBehavior.Restrict);
    }
}
// Rejestracja w Program.cs
builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseSqlServer(
        builder.Configuration.GetConnectionString("DefaultConnection"),
        sql => sql.EnableRetryOnFailure(
            maxRetryCount: 3,
            maxRetryDelay: TimeSpan.FromSeconds(5),
            errorNumbersToAdd: null))
    .LogTo(Console.WriteLine, LogLevel.Information)
    .EnableSensitiveDataLogging(
        builder.Environment.IsDevelopment()));

# Dodanie migracji
dotnet ef migrations add InitialCreate --project src/MyApi.Infrastructure --startup-project src/MyApi.Api

# Podglad SQL przed zastosowaniem
dotnet ef migrations script --idempotent

# Zastosowanie migracji
dotnet ef database update

# Cofniecie do konkretnej migracji
dotnet ef database update PreviousMigrationName

# Usuniecie ostatniej migracji (jesli nie zastosowana)

LINQ — lancuchy zapytan

// Podstawowe zapytania
var products = await _context.Products
    .Where(p => p.Price > 100 && p.Stock > 0)
    .OrderBy(p => p.Name)
    .Skip((page - 1) * pageSize)
    .Take(pageSize)
    .Select(p => new ProductDto(p.Id, p.Name, p.Price))
    .ToListAsync(ct);

// Include — eager loading
var orders = await _context.Orders
    .Include(o => o.Customer)
    .Include(o => o.Items)
        .ThenInclude(i => i.Product)
            .ThenInclude(p => p.Category)
    .Where(o => o.Status == OrderStatus.Pending)
    .ToListAsync(ct);

// Projekcja na DTO (unika SELECT *)
var dtos = await _context.Products
    .Where(p => p.CategoryId == categoryId)
    .Select(p => new ProductListDto
    {
        Id = p.Id,
        Name = p.Name,
        Price = p.Price,
        CategoryName = p.Category.Name  // JOIN wygenerowany automatycznie
    })
    .ToListAsync(ct);

// Grupowanie i agregacja
var stats = await _context.OrderItems
    .GroupBy(i => i.ProductId)
    .Select(g => new
    {
        ProductId = g.Key,
        TotalSold = g.Sum(i => i.Quantity),
        Revenue = g.Sum(i => i.Quantity * i.UnitPrice)
    })
    .OrderByDescending(s => s.Revenue)
    .Take(10)
    .ToListAsync(ct);

Strategie ladowania — Eager / Lazy / Explicit

StrategiaJak dzialaKiedy uzywacRyzyko
Eager loading Include() — JOIN w jednym zapytaniu SQL Gdy wiesz z gory, ze potrzebujesz powiazan Duze JOIN-y moga byc wolniejsze niz osobne zapytania
Lazy loading Wlasciwosci nawigacyjne ladowane przy dostepe Rzadko — tylko gdy powiazan uzywa sie opcjonalnie Problem N+1: osobne SQL na kazdy obiekt w petli
Explicit loading Entry().Collection().LoadAsync() Gdy chcesz zaladowac powiazan warunkowo Dwa osobne zapytania SQL
Split query AsSplitQuery() — osobne SQL per kolekcja Kolekcje z duza iloscia danych (unikanie kartezjanu) Wiele zapytan zamiast jednego duzego JOIN 
// Explicit loading — warunkowo
var order = await _context.Orders.FindAsync(orderId);
if (includeItems)
    await _context.Entry(order!)
        .Collection(o => o.Items)
        .Query()
        .Include(i => i.Product)
        .LoadAsync(ct);

// Split query — dla kolekcji z duza iloscia danych
var orders = await _context.Orders
    .Include(o => o.Items)
    .AsSplitQuery()
    .ToListAsync(ct);

AsNoTracking — wydajnosc odczytu

Raw SQL — FromSqlRaw i ExecuteSqlRaw

// FromSqlRaw — zapytania zwracajace encje
var products = await _context.Products
    .FromSqlRaw("SELECT * FROM Products WHERE Price > {0}", minPrice)
    .Include(p => p.Category)
    .Where(p => p.IsActive)  // LINQ dziala na wynikach SQL
    .ToListAsync(ct);

// ZAWSZE uzywaj parametrow (nie interpolacji!) — ochrona przed SQL injection
// ZLE:
var bad = _context.Products
    .FromSqlRaw($"SELECT * FROM Products WHERE Name = '{name}'"); // NIEBEZPIECZNE!

// DOBRZE (parametry):
var good = _context.Products
    .FromSqlRaw("SELECT * FROM Products WHERE Name = {0}", name);

// ExecuteSqlRaw — INSERT/UPDATE/DELETE
var affected = await _context.Database
    .ExecuteSqlRawAsync(
        "UPDATE Products SET Stock = Stock - {0} WHERE Id = {1}",
        quantity, productId, ct);

// Stored procedure
var results = await _context.Products
    .FromSqlRaw("EXEC GetTopSellingProducts @CategoryId = {0}, @Top = {1}",
        categoryId, 10)
    .ToListAsync(ct);

Wydajnosc — compiled queries i batch

// Compiled queries — dla czesto wykonywanych zapytan
private static readonly Func<AppDbContext, int, Task<Product?>>
    GetByIdQuery = EF.CompileAsyncQuery(
        (AppDbContext ctx, int id) =>
            ctx.Products.FirstOrDefault(p => p.Id == id));

// Uzycie
var product = await GetByIdQuery(_context, productId);

// Batch INSERT (EF Core 7+ ExecuteUpdateAsync / ExecuteDeleteAsync)
// Usun wszystkie nieaktywne produkty BEZ ladowania ich do pamieci
var deleted = await _context.Products
    .Where(p => !p.IsActive && p.CreatedAt < cutoffDate)
    .ExecuteDeleteAsync(ct);

// Batch UPDATE bez ladowania encji
var updated = await _context.Products
    .Where(p => p.CategoryId == oldCategoryId)
    .ExecuteUpdateAsync(s => s
        .SetProperty(p => p.CategoryId, newCategoryId)
        .SetProperty(p => p.UpdatedAt, DateTime.UtcNow),
        ct);
