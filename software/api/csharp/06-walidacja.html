Data Annotations

Data Annotations to atrybuty z przestrzeni System.ComponentModel.DataAnnotations. Sa proste w uzyciu, ale maja ograniczenia przy zlozonych regułach i warunkowej walidacji.

public class CreateOrderRequest
{
    [Required(ErrorMessage = "Identyfikator klienta jest wymagany")]
    public int CustomerId { get; set; }

    [Required]
    [MinLength(1, ErrorMessage = "Zamowienie musi zawierac co najmniej jeden produkt")]
    public List<OrderItemRequest> Items { get; set; } = [];

    [StringLength(500, ErrorMessage = "Notatka moze miec max 500 znakow")]
    public string? Note { get; set; }

    [Required]
    public AddressRequest ShippingAddress { get; set; } = null!;
}

public class OrderItemRequest
{
    [Required]
    [Range(1, int.MaxValue, ErrorMessage = "Id produktu musi byc > 0")]
    public int ProductId { get; set; }

    [Range(1, 1000, ErrorMessage = "Ilosc musi byc od 1 do 1000")]
    public int Quantity { get; set; }
}

public class AddressRequest
{
    [Required]
    [StringLength(200)]
    public string Street { get; set; } = string.Empty;

    [Required]
    [StringLength(100)]
    public string City { get; set; } = string.Empty;

    [Required]
    [RegularExpression(@"^\d{2}-\d{3}$", ErrorMessage = "Nieprawidlowy format kodu pocztowego (XX-XXX)")]
    public string PostalCode { get; set; } = string.Empty;

    [Required]
    [StringLength(2, MinimumLength = 2, ErrorMessage = "Kod kraju musi miec 2 znaki (ISO 3166)")]
    public string CountryCode { get; set; } = string.Empty;
}

// Niestandardowy atrybut walidacji
public class FutureDateAttribute : ValidationAttribute
{
    protected override ValidationResult? IsValid(
        object? value, ValidationContext ctx)
    {
        if (value is DateTime date && date <= DateTime.UtcNow)
            return new ValidationResult(
                $"{ctx.DisplayName} musi byc data przyszla");
        return ValidationResult.Success;
    }
}

public class ScheduleRequest
{
    [Required]
    [FutureDate]
    public DateTime ScheduledAt { get; set; }
}

FluentValidation v11

FluentValidation to biblioteka oferujaca plynne API do definiowania regul walidacji. Jest bardziej ekspresywna niz Data Annotations i lepiej radzi sobie z walidacja warunkowa, zagniezdzona i asynchroniczna.

// dotnet add package FluentValidation.AspNetCore

// Validator dla zamowienia
public class CreateOrderValidator : AbstractValidator<CreateOrderRequest>
{
    private readonly ICustomerRepository _customers;
    private readonly IProductRepository _products;

    public CreateOrderValidator(
        ICustomerRepository customers,
        IProductRepository products)
    {
        _customers = customers;
        _products = products;

        RuleFor(x => x.CustomerId)
            .GreaterThan(0).WithMessage("Id klienta musi byc > 0")
            .MustAsync(CustomerExistsAsync)
                .WithMessage("Klient o podanym Id nie istnieje");

        RuleFor(x => x.Items)
            .NotEmpty().WithMessage("Zamowienie musi zawierac produkty")
            .Must(items => items.Count <= 50)
                .WithMessage("Zamowienie moze miec max 50 pozycji");

        RuleForEach(x => x.Items)
            .SetValidator(new OrderItemValidator(_products));

        RuleFor(x => x.ShippingAddress)
            .NotNull()
            .SetValidator(new AddressValidator());

        RuleFor(x => x.Note)
            .MaximumLength(500)
            .When(x => x.Note is not null);
    }

    private async Task<bool> CustomerExistsAsync(
        int id, CancellationToken ct)
        => await _customers.ExistsAsync(id, ct);
}

// Validator pozycji zamowienia
public class OrderItemValidator : AbstractValidator<OrderItemRequest>
{
    private readonly IProductRepository _products;

    public OrderItemValidator(IProductRepository products)
    {
        _products = products;

        RuleFor(x => x.ProductId)
            .GreaterThan(0)
            .MustAsync(ProductExistsAsync)
                .WithMessage("Produkt {PropertyValue} nie istnieje");

        RuleFor(x => x.Quantity)
            .InclusiveBetween(1, 1000)
            .WithMessage("Ilosc musi byc od 1 do 1000");
    }

    private async Task<bool> ProductExistsAsync(
        int id, CancellationToken ct)
        => await _products.ExistsAsync(id, ct);
}

// Validator adresu
public class AddressValidator : AbstractValidator<AddressRequest>
{
    public AddressValidator()
    {
        RuleFor(x => x.Street).NotEmpty().MaximumLength(200);
        RuleFor(x => x.City).NotEmpty().MaximumLength(100);
        RuleFor(x => x.PostalCode)
            .NotEmpty()
            .Matches(@"^\d{2}-\d{3}$")
                .WithMessage("Format kodu pocztowego: XX-XXX");
        RuleFor(x => x.CountryCode)
            .NotEmpty()
            .Length(2)
            .Must(c => c.All(char.IsLetter))
                .WithMessage("Kod kraju musi skladac sie z liter");
    }
}

Integracja z Minimal APIs

// Reczne wywolanie walidatora w Minimal API
app.MapPost("/orders", async Task<Results<Created<Order>, ValidationProblem>>(
    CreateOrderRequest request,
    IValidator<CreateOrderRequest> validator,
    IOrderService orderService,
    CancellationToken ct) =>
{
    var result = await validator.ValidateAsync(request, ct);
    if (!result.IsValid)
        return TypedResults.ValidationProblem(result.ToDictionary());

    var order = await orderService.CreateAsync(request, ct);
    return TypedResults.Created($"/orders/{order.Id}", order);
});

// Filtr walidacji do wielokrotnego uzycia
public class FluentValidationFilter<TRequest> : IEndpointFilter
{
    public async ValueTask<object?> InvokeAsync(
        EndpointFilterInvocationContext ctx,
        EndpointFilterDelegate next)
    {
        var validator = ctx.HttpContext.RequestServices
            .GetService<IValidator<TRequest>>();

        if (validator is not null)
        {
            var model = ctx.Arguments.OfType<TRequest>().FirstOrDefault();
            if (model is not null)
            {
                var result = await validator.ValidateAsync(model,
                    ctx.HttpContext.RequestAborted);
                if (!result.IsValid)
                    return Results.ValidationProblem(result.ToDictionary());
            }
        }
        return await next(ctx);
    }
}

// Metoda rozszerzenia dla czytelnosci
public static RouteHandlerBuilder WithValidation<TRequest>(
    this RouteHandlerBuilder builder)
    => builder.AddEndpointFilter<FluentValidationFilter<TRequest>>();

// Uzycie
app.MapPost("/orders", CreateOrderHandler)
   .WithValidation<CreateOrderRequest>();

Walidacja asynchroniczna

// Async rule — sprawdzenie unikalnosci w bazie
public class CreateProductValidator : AbstractValidator<CreateProductRequest>
{
    public CreateProductValidator(IProductRepository repo)
    {
        RuleFor(x => x.Name)
            .NotEmpty()
            .MaximumLength(200)
            .MustAsync(async (name, ct) =>
                !await repo.ExistsByNameAsync(name, ct))
                .WithMessage("Produkt o tej nazwie juz istnieje");

        RuleFor(x => x.Sku)
            .NotEmpty()
            .MustAsync(async (sku, ct) =>
                !await repo.ExistsBySkuAsync(sku, ct))
                .WithMessage("SKU '{PropertyValue}' jest juz zajety");

        RuleFor(x => x.Price)
            .GreaterThan(0)
            .LessThanOrEqualTo(999_999.99m);

        RuleFor(x => x.CategoryId)
            .MustAsync(async (id, ct) =>
                await repo.CategoryExistsAsync(id, ct))
                .WithMessage("Kategoria nie istnieje");
    }
}

ProblemDetails dla bledow walidacji

// Standardowa odpowiedz walidacji (422 Unprocessable Entity)
// Generowana automatycznie przez [ApiController] lub TypedResults.ValidationProblem()
{
  "type": "https://tools.ietf.org/html/rfc4918#section-11.2",
  "title": "One or more validation errors occurred.",
  "status": 422,
  "errors": {
    "CustomerId": ["Klient o podanym Id nie istnieje"],
    "Items[0].ProductId": ["Produkt 999 nie istnieje"],
    "ShippingAddress.PostalCode": ["Format kodu pocztowego: XX-XXX"]
  }
}

// Dostosowanie odpowiedzi walidacji w [ApiController]
builder.Services.Configure<ApiBehaviorOptions>(options =>
{
    options.InvalidModelStateResponseFactory = ctx =>
    {
        var problem = new ValidationProblemDetails(ctx.ModelState)
        {
            Type = "https://myapi.example.com/errors/validation",
            Title = "Blad walidacji danych",
            Status = StatusCodes.Status422UnprocessableEntity,
            Instance = ctx.HttpContext.Request.Path,
        };
        problem.Extensions["traceId"] =
            ctx.HttpContext.TraceIdentifier;
        return new UnprocessableEntityObjectResult(problem);
    };
});

Data Annotations vs FluentValidation

AspektData AnnotationsFluentValidation v11
Prostota uzyciaAtrybuty inline na modeluOsobna klasa validatora
TestowanieTrudne — wymaga refleksjiLatwe — bezposrednie wywolanie
Walidacja warunkowaOgraniczona (IValidatableObject)Pelna (When, Unless)
Walidacja asyncBrakPelna (MustAsync)
Walidacja zagniezdzonaOgraniczonaPelna (SetValidator, RuleForEach)
Komunikaty bledowAtrybut przy poluFluent chain, placeholdery, i18n
DI w validatorzeBrakPelna (konstruktor)
Kiedy wybracProste DTO, szybkie projektyZlozono logika, async, testowanie
Rekomendacja 2026: Uzyj FluentValidation dla logiki biznesowej i zlozonych modeli. Data Annotations pozostaw dla prostych DTO lub gdy zalezy Ci na kompatybilnosci z EF Core (mapowanie dlugosci kolumn).
