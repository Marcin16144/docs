JWT Bearer — konfiguracja

JWT (JSON Web Token) jest najpopularniejszym mechanizmem uwierzytelniania w REST API. Token zawiera claims (twierdzenia) o uzytkownika i jest podpisany kluczem tajnym lub certyfikatem.

// dotnet add package Microsoft.AspNetCore.Authentication.JwtBearer

// Program.cs
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuer = true,
            ValidIssuer = builder.Configuration["Jwt:Issuer"],

            ValidateAudience = true,
            ValidAudience = builder.Configuration["Jwt:Audience"],

            ValidateIssuerSigningKey = true,
            IssuerSigningKey = new SymmetricSecurityKey(
                Encoding.UTF8.GetBytes(
                    builder.Configuration["Jwt:Secret"]!)),

            ValidateLifetime = true,
            ClockSkew = TimeSpan.FromSeconds(30), // tolerancja zegara

            // Wymagane claims w tokenie
            NameClaimType = JwtRegisteredClaimNames.Sub,
            RoleClaimType = ClaimTypes.Role
        };

        // Zdarzenia — np. logowanie bledow walidacji
        options.Events = new JwtBearerEvents
        {
            OnAuthenticationFailed = ctx =>
            {
                if (ctx.Exception is SecurityTokenExpiredException)
                    ctx.Response.Headers.Append(
                        "Token-Expired", "true");
                return Task.CompletedTask;
            }
        };
    });

Generowanie tokenu JWT

public class TokenService : ITokenService
{
    private readonly JwtOptions _opts;

    public TokenService(IOptions<JwtOptions> opts)
    {
        _opts = opts.Value;
    }

    public string GenerateAccessToken(User user, IList<string> roles)
    {
        var claims = new List<Claim>
        {
            new(JwtRegisteredClaimNames.Sub, user.Id.ToString()),
            new(JwtRegisteredClaimNames.Email, user.Email!),
            new(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString()),
            new(JwtRegisteredClaimNames.Iat,
                DateTimeOffset.UtcNow.ToUnixTimeSeconds().ToString(),
                ClaimValueTypes.Integer64),
        };

        claims.AddRange(roles.Select(r => new Claim(ClaimTypes.Role, r)));

        var key = new SymmetricSecurityKey(
            Encoding.UTF8.GetBytes(_opts.Secret));
        var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: _opts.Issuer,
            audience: _opts.Audience,
            claims: claims,
            notBefore: DateTime.UtcNow,
            expires: DateTime.UtcNow.AddMinutes(_opts.ExpiryMinutes),
            signingCredentials: creds);

        return new JwtSecurityTokenHandler().WriteToken(token);
    }

    public string GenerateRefreshToken()
    {
        var bytes = RandomNumberGenerator.GetBytes(64);
        return Convert.ToBase64String(bytes);
    }
}

Refresh Tokens

Access token powinien miec krotki czas wygasniecia (15-60 minut). Refresh token (dlugo zyjacy, przechowywany w bazie) pozwala pobrac nowy access token bez ponownego logowania.

// Encja refresh tokenu
public class RefreshToken
{
    public int Id { get; set; }
    public string Token { get; set; } = string.Empty;
    public string UserId { get; set; } = string.Empty;
    public DateTime ExpiresAt { get; set; }
    public bool IsRevoked { get; set; }
    public string? ReplacedByToken { get; set; }
    public DateTime CreatedAt { get; set; }
}

// Endpoint odswiezania tokenu
app.MapPost("/auth/refresh", async (
    RefreshRequest req,
    ITokenService tokenService,
    UserManager<AppUser> userManager,
    AppDbContext db) =>
{
    var stored = await db.RefreshTokens
        .FirstOrDefaultAsync(t => t.Token == req.RefreshToken
            && !t.IsRevoked
            && t.ExpiresAt > DateTime.UtcNow);

    if (stored is null)
        return Results.Unauthorized();

    var user = await userManager.FindByIdAsync(stored.UserId);
    if (user is null) return Results.Unauthorized();

    var roles = await userManager.GetRolesAsync(user);

    // Rotacja refresh tokenu (token reuse detection)
    stored.IsRevoked = true;
    var newRefresh = new RefreshToken
    {
        Token = tokenService.GenerateRefreshToken(),
        UserId = user.Id,
        ExpiresAt = DateTime.UtcNow.AddDays(30),
        CreatedAt = DateTime.UtcNow
    };
    stored.ReplacedByToken = newRefresh.Token;
    db.RefreshTokens.Add(newRefresh);
    await db.SaveChangesAsync();

    return Results.Ok(new
    {
        AccessToken = tokenService.GenerateAccessToken(user, roles),
        RefreshToken = newRefresh.Token
    });
});

ASP.NET Core Identity

// dotnet add package Microsoft.AspNetCore.Identity.EntityFrameworkCore

public class AppUser : IdentityUser
{
    public string? FullName { get; set; }
    public DateTime CreatedAt { get; set; }
}

public class AppDbContext : IdentityDbContext<AppUser>
{
    public AppDbContext(DbContextOptions<AppDbContext> options)
        : base(options) { }
}

// Program.cs
builder.Services
    .AddIdentity<AppUser, IdentityRole>(options =>
    {
        options.Password.RequiredLength = 8;
        options.Password.RequireNonAlphanumeric = true;
        options.Lockout.MaxFailedAccessAttempts = 5;
        options.Lockout.DefaultLockoutTimeSpan = TimeSpan.FromMinutes(15);
        options.User.RequireUniqueEmail = true;
        options.SignIn.RequireConfirmedEmail = true;
    })
    .AddEntityFrameworkStores<AppDbContext>()
    .AddDefaultTokenProviders();

Polityki autoryzacji

// Definicja polityk
builder.Services.AddAuthorization(options =>
{
    // Prosta polityka roli
    options.AddPolicy("AdminOnly", policy =>
        policy.RequireRole("Admin"));

    // Polityka z claims
    options.AddPolicy("PremiumUser", policy =>
        policy.RequireClaim("subscription", "premium", "enterprise"));

    // Polityka z wymaganiem uwierzytelnienia i emailem firmowym
    options.AddPolicy("CompanyEmployee", policy =>
        policy.RequireAuthenticatedUser()
              .RequireClaim(ClaimTypes.Email)
              .AddRequirements(new CompanyEmailRequirement("mycompany.com")));

    // Polityka kombinowana
    options.AddPolicy("SeniorEditor", policy =>
        policy.RequireRole("Editor")
              .RequireClaim("experience_years", "5", "6", "7", "8", "9", "10+"));

    // Fallback policy — wszystkie endpointy wymagaja autoryzacji
    options.FallbackPolicy = new AuthorizationPolicyBuilder()
        .RequireAuthenticatedUser()
        .Build();
});

// Uzycie na kontrolerze / akcji

Claims-based RBAC i IAuthorizationHandler

// Wymaganie autoryzacji
public class CompanyEmailRequirement : IAuthorizationRequirement
{
    public string Domain { get; }
    public CompanyEmailRequirement(string domain) => Domain = domain;
}

// Handler wymagania
public class CompanyEmailHandler
    : AuthorizationHandler<CompanyEmailRequirement>
{
    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        CompanyEmailRequirement requirement)
    {
        var emailClaim = context.User
            .FindFirst(ClaimTypes.Email)?.Value;

        if (emailClaim is not null
            && emailClaim.EndsWith($"@{requirement.Domain}",
                StringComparison.OrdinalIgnoreCase))
        {
            context.Succeed(requirement);
        }

        return Task.CompletedTask;
    }
}

// Rejestracja
builder.Services
    .AddSingleton<IAuthorizationHandler, CompanyEmailHandler>();

Resource-based authorization

// Autoryzacja zasobowa — "czy uzytkownik moze edytowac TEN dokument?"
public class DocumentEditRequirement : IAuthorizationRequirement { }

public class DocumentAuthorizationHandler
    : AuthorizationHandler<DocumentEditRequirement, Document>
{
    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        DocumentEditRequirement requirement,
        Document resource)
    {
        var userId = context.User
            .FindFirst(JwtRegisteredClaimNames.Sub)?.Value;

        if (resource.OwnerId == userId
            || context.User.IsInRole("Admin"))
        {
            context.Succeed(requirement);
        }

        return Task.CompletedTask;
    }
}

// Uzycie w akcji kontrolera
[HttpPut("{id:int}")]
public async Task<IActionResult> Update(int id, UpdateDocumentRequest req)
{
    var document = await _repo.GetByIdAsync(id);
    if (document is null) return NotFound();

    var authResult = await _authorizationService
        .AuthorizeAsync(User, document, new DocumentEditRequirement());

    if (!authResult.Succeeded)
        return Forbid();

    var updated = await _repo.UpdateAsync(id, req);
    return Ok(updated);
}

// Serwis pomocniczy — ICurrentUserService
public interface ICurrentUserService
{
    string? UserId { get; }
    string? Email { get; }
    bool IsAuthenticated { get; }
    bool IsInRole(string role);
}

public class CurrentUserService : ICurrentUserService
{
    private readonly IHttpContextAccessor _accessor;

    public CurrentUserService(IHttpContextAccessor accessor)
    {
        _accessor = accessor;
    }

    public string? UserId =>
        _accessor.HttpContext?.User
            .FindFirst(JwtRegisteredClaimNames.Sub)?.Value;

    public string? Email =>
        _accessor.HttpContext?.User
            .FindFirst(JwtRegisteredClaimNames.Email)?.Value;

    public bool IsAuthenticated =>
        _accessor.HttpContext?.User.Identity?.IsAuthenticated ?? false;

    public bool IsInRole(string role) =>
        _accessor.HttpContext?.User.IsInRole(role) ?? false;
}

// Rejestracja
builder.Services.AddHttpContextAccessor();
builder.Services.AddScoped<ICurrentUserService, CurrentUserService>();
Bezpieczenstwo — wazne zasady:
  • Nigdy nie przechowuj klucza JWT w kodzie — uzywaj zmiennych srodowiskowych lub Key Vault.
  • Access token: max 60 minut waznosci. Refresh token: max 30 dni, przechowywany w httpOnly cookie.
  • Implementuj rotacje refresh tokenow (jeden token = jedno uzycie).
  • Dla produkcji rozważ asymetryczne klucze (RSA/ECDSA) zamiast HMAC.
