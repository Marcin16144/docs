WebApplicationFactory — testy integracyjne

// dotnet add package Microsoft.AspNetCore.Mvc.Testing

// Fabryka aplikacji testowej
public class ApiFactory : WebApplicationFactory<Program>,
    IAsyncLifetime
{
    private readonly PostgreSqlContainer _postgres =
        new PostgreSqlBuilder()
            .WithImage("postgres:16-alpine")
            .WithDatabase("testdb")
            .WithUsername("test")
            .WithPassword("test")
            .Build();

    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureTestServices(services =>
        {
            // Zamien prawdziwy DbContext na testowy (wskazujacy na kontener)
            services.RemoveAll<DbContextOptions<AppDbContext>>();
            services.AddDbContext<AppDbContext>(options =>
                options.UseNpgsql(_postgres.GetConnectionString()));

            // Zamien zewnetrzne serwisy na mocki
            services.RemoveAll<IEmailSender>();
            services.AddSingleton<IEmailSender>(
                Substitute.For<IEmailSender>());

            // Zamien autentykacje na testowa
            services
                .AddAuthentication("Test")
                .AddScheme<AuthenticationSchemeOptions,
                    TestAuthHandler>("Test", _ => { });
        });

        builder.UseEnvironment("Test");
    }

    public async Task InitializeAsync()
    {
        await _postgres.StartAsync();

        // Zastosuj migracje na kontenerze testowym
        using var scope = Services.CreateScope();
        var db = scope.ServiceProvider
            .GetRequiredService<AppDbContext>();
        await db.Database.MigrateAsync();
        await SeedTestDataAsync(db);
    }

    public new async Task DisposeAsync()
    {
        await _postgres.StopAsync();
        await base.DisposeAsync();
    }

    private static async Task SeedTestDataAsync(AppDbContext db)
    {
        db.Categories.AddRange(
            new Category { Id = 1, Name = "Elektronika" },
            new Category { Id = 2, Name = "Odziez" });
        db.Products.Add(new Product
        {
            Id = 1, Name = "Laptop", Price = 3999m,
            Stock = 10, CategoryId = 1
        });
        await db.SaveChangesAsync();
    }
}

// Handler autentykacji testowej
public class TestAuthHandler : AuthenticationHandler<AuthenticationSchemeOptions>
{
    public TestAuthHandler(
        IOptionsMonitor<AuthenticationSchemeOptions> options,
        ILoggerFactory logger,
        UrlEncoder encoder)
        : base(options, logger, encoder) { }

    protected override Task<AuthenticateResult> HandleAuthenticateAsync()
    {
        var claims = new[]
        {
            new Claim(ClaimTypes.NameIdentifier, "test-user-id"),
            new Claim(ClaimTypes.Email, "test@example.com"),
            new Claim(ClaimTypes.Role, "Admin")
        };
        var identity = new ClaimsIdentity(claims, "Test");
        var principal = new ClaimsPrincipal(identity);
        var ticket = new AuthenticationTicket(principal, "Test");
        return Task.FromResult(AuthenticateResult.Success(ticket));
    }
}

Testy integracyjne endpointow

public class ProductsEndpointTests
    : IClassFixture<ApiFactory>
{
    private readonly HttpClient _client;
    private readonly ApiFactory _factory;

    public ProductsEndpointTests(ApiFactory factory)
    {
        _factory = factory;
        _client = factory.CreateClient();
    }

    [Fact]
    public async Task GetProducts_ReturnsOkWithList()
    {
        // Act
        var response = await _client.GetAsync("/api/v1/products");

        // Assert
        response.StatusCode.Should().Be(HttpStatusCode.OK);
        var body = await response.Content
            .ReadFromJsonAsync<PagedResult<ProductDto>>();
        body.Should().NotBeNull();
        body!.Items.Should().NotBeEmpty();
    }

    [Fact]
    public async Task GetProduct_WithValidId_ReturnsProduct()
    {
        // Act
        var response = await _client.GetAsync("/api/v1/products/1");

        // Assert
        response.StatusCode.Should().Be(HttpStatusCode.OK);
        var product = await response.Content
            .ReadFromJsonAsync<ProductDto>();
        product!.Id.Should().Be(1);
        product.Name.Should().Be("Laptop");
    }

    [Fact]
    public async Task GetProduct_WithInvalidId_Returns404()
    {
        var response = await _client.GetAsync("/api/v1/products/99999");
        response.StatusCode.Should().Be(HttpStatusCode.NotFound);

        var problem = await response.Content
            .ReadFromJsonAsync<ProblemDetails>();
        problem!.Status.Should().Be(404);
    }

    [Fact]
    public async Task CreateProduct_WithValidData_Returns201()
    {
        // Arrange
        var request = new CreateProductRequest(
            "Nowy Produkt", 299.99m, 50, 1);

        // Act
        var response = await _client.PostAsJsonAsync(
            "/api/v1/products", request);

        // Assert
        response.StatusCode.Should().Be(HttpStatusCode.Created);
        response.Headers.Location.Should().NotBeNull();

        var created = await response.Content
            .ReadFromJsonAsync<ProductDto>();
        created!.Name.Should().Be("Nowy Produkt");
        created.Price.Should().Be(299.99m);
    }

    [Fact]
    public async Task CreateProduct_WithInvalidData_Returns422()
    {
        var request = new { Name = "", Price = -1 };
        var response = await _client.PostAsJsonAsync(
            "/api/v1/products", request);

        response.StatusCode.Should().Be(
            HttpStatusCode.UnprocessableEntity);
        var problem = await response.Content
            .ReadFromJsonAsync<ValidationProblemDetails>();
        problem!.Errors.Should().ContainKey("Name");
    }

    // Izolacja testow — czyszczenie po kazdym tescie
    public async Task DisposeAsync()
    {
        using var scope = _factory.Services.CreateScope();
        var db = scope.ServiceProvider
            .GetRequiredService<AppDbContext>();
        db.Products.RemoveRange(
            db.Products.Where(p => p.Id > 1));
        await db.SaveChangesAsync();
    }
}

Moq — mockowanie zaleznosci

// dotnet add package Moq
// dotnet add package Moq.AutoMock

public class OrderServiceTests
{
    private readonly Mock<IOrderRepository> _repoMock = new();
    private readonly Mock<IProductRepository> _productMock = new();
    private readonly Mock<IEmailSender> _emailMock = new();
    private readonly OrderService _sut;

    public OrderServiceTests()
    {
        _sut = new OrderService(
            _repoMock.Object,
            _productMock.Object,
            _emailMock.Object);
    }

    [Fact]
    public async Task PlaceOrder_WhenProductInStock_CreatesOrder()
    {
        // Arrange
        var product = new Product { Id = 1, Stock = 10, Price = 99m };
        _productMock.Setup(r => r.GetByIdAsync(1, It.IsAny<CancellationToken>()))
            .ReturnsAsync(product);

        var expectedOrder = new Order { Id = 42, TotalAmount = 99m };
        _repoMock.Setup(r => r.CreateAsync(
                It.IsAny<PlaceOrderRequest>(),
                It.IsAny<CancellationToken>()))
            .ReturnsAsync(expectedOrder);

        // Act
        var result = await _sut.PlaceOrderAsync(
            new PlaceOrderRequest(CustomerId: 1, ProductId: 1, Quantity: 1),
            CancellationToken.None);

        // Assert
        result.Should().NotBeNull();
        result.Id.Should().Be(42);

        _emailMock.Verify(e => e.SendOrderConfirmationAsync(
            It.IsAny<Order>(),
            It.IsAny<CancellationToken>()),
            Times.Once);
    }

    [Fact]
    public async Task PlaceOrder_WhenInsufficientStock_ThrowsException()
    {
        // Arrange
        var product = new Product { Id = 1, Stock = 2 };
        _productMock.Setup(r => r.GetByIdAsync(1, It.IsAny<CancellationToken>()))
            .ReturnsAsync(product);

        // Act & Assert
        var act = () => _sut.PlaceOrderAsync(
            new PlaceOrderRequest(1, 1, Quantity: 10),
            CancellationToken.None);

        await act.Should().ThrowAsync<BusinessRuleException>()
            .WithMessage("*stan*");

        _repoMock.Verify(r => r.CreateAsync(
            It.IsAny<PlaceOrderRequest>(),
            It.IsAny<CancellationToken>()),
            Times.Never);
    }
}

Verify — snapshot testing

// dotnet add package Verify.Xunit

// Snapshot test — zapisuje i porownuje odpowiedz JSON
[UsesVerify]
public class ProductSnapshotTests : IClassFixture<ApiFactory>
{
    private readonly HttpClient _client;

    public ProductSnapshotTests(ApiFactory factory)
    {
        _client = factory.CreateClient();
    }

    [Fact]
    public async Task GetProduct_MatchesSnapshot()
    {
        var response = await _client.GetAsync("/api/v1/products/1");
        var json = await response.Content.ReadAsStringAsync();

        // Przy pierwszym uruchomieniu tworzy plik .verified.json
        // Przy kolejnych porownuje z zapisanym
        await Verify(json)
            .ScrubMember("createdAt")  // ignoruj dynamiczne pola
            .ScrubMember("traceId");
    }
}

Dobre praktyki testowania API:
  • Uzywaj Testcontainers zamiast in-memory bazy — testy sa bardziej realistyczne i wykrywaja bledy specyficzne dla SQL.
  • Jeden WebApplicationFactory na kolekcje testow (IClassFixture) — nie twórz nowej instancji per test.
  • Izoluj testy — czyszcz dane po kazdym tescie lub uzywaj transakcji.
  • Testuj kody HTTP, naglowki odpowiedzi i strukture ProblemDetails, nie tylko body.
