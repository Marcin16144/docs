gRPC w ASP.NET Core

Pliki .proto, ProtoBuf, streaming, gRPC-Web, deadlines, interceptory i refleksja

Czym jest gRPC?

gRPC to framework RPC (Remote Procedure Call) stworzony przez Google, oparty na HTTP/2 i Protocol Buffers (ProtoBuf). W porownaniu do REST oferuje silnie typowane kontrakty, wydajniejsza serializacje binarna i natywne wsparcie dla streamingu.

CechagRPCREST/JSON
ProtokolHTTP/2HTTP/1.1 lub HTTP/2
SerializacjaProtoBuf (binarny, ~3-10x mniejszy)JSON (tekstowy, czytelny)
KontraktSilnie typowany (.proto)OpenAPI (opcjonalny)
StreamingNatywny (4 wzorce)SSE lub WebSocket (zewnetrzny)
Browser supportWymaga gRPC-Web proxyNatywny
Kiedy uzywacMikrouslugi backend-to-backend, streaming, real-timePubliczne API, frontend, prostota

Definicja .proto

// products.proto
syntax = "proto3";

option csharp_namespace = "MyApi.Grpc";

package products;

import "google/protobuf/timestamp.proto";
import "google/protobuf/empty.proto";

// Serwis produktow
service ProductService {
  // Unary RPC — jeden request, jedna odpowiedz
  rpc GetProduct (GetProductRequest) returns (ProductResponse);
  rpc CreateProduct (CreateProductRequest) returns (ProductResponse);
  rpc UpdateProduct (UpdateProductRequest) returns (ProductResponse);
  rpc DeleteProduct (DeleteProductRequest) returns (google.protobuf.Empty);

  // Server streaming — jeden request, strumien odpowiedzi
  rpc ListProducts (ListProductsRequest) returns (stream ProductResponse);

  // Client streaming — strumien requestow, jedna odpowiedz
  rpc ImportProducts (stream CreateProductRequest) returns (ImportResult);

  // Bidirectional streaming — strumien requestow i odpowiedzi
  rpc SyncProducts (stream ProductSyncRequest) returns (stream ProductSyncResponse);
}

message GetProductRequest {
  int32 id = 1;
}

message CreateProductRequest {
  string name = 1;
  double price = 2;
  int32 stock = 3;
  int32 category_id = 4;
  optional string description = 5;
}

message UpdateProductRequest {
  int32 id = 1;
  string name = 2;
  double price = 3;
  int32 stock = 4;
}

message DeleteProductRequest {
  int32 id = 1;
}

message ProductResponse {
  int32 id = 1;
  string name = 2;
  double price = 3;
  int32 stock = 4;
  string category_name = 5;
  google.protobuf.Timestamp created_at = 6;
}

message ListProductsRequest {
  int32 page_size = 1;
  string page_token = 2;
  optional int32 category_id = 3;
  optional double min_price = 4;
}

message ImportResult {
  int32 imported = 1;
  int32 failed = 2;
  repeated string errors = 3;
}

message ProductSyncRequest {
  oneof operation {
    CreateProductRequest create = 1;
    UpdateProductRequest update = 2;
    DeleteProductRequest delete = 3;
  }
}

message ProductSyncResponse {
  bool success = 1;
  optional string error = 2;
  int32 processed_id = 3;
}

Konfiguracja projektu (.csproj i Program.cs)

<!-- MyApi.csproj -->
<ItemGroup>
  <PackageReference Include="Grpc.AspNetCore" Version="2.65.0" />
  <PackageReference Include="Grpc.AspNetCore.Web" Version="2.65.0" />
  <PackageReference Include="Grpc.AspNetCore.Server.Reflection" Version="2.65.0" />
</ItemGroup>

<ItemGroup>
  <Protobuf Include="Protos\products.proto" GrpcServices="Server" />
</ItemGroup>
// Program.cs
builder.Services.AddGrpc(options =>
{
    options.EnableDetailedErrors = builder.Environment.IsDevelopment();
    options.MaxReceiveMessageSize = 4 * 1024 * 1024;  // 4 MB
    options.MaxSendMessageSize = 4 * 1024 * 1024;
    options.Interceptors.Add<LoggingInterceptor>();
});

builder.Services.AddGrpcReflection();

var app = builder.Build();

app.UseGrpcWeb(new GrpcWebOptions { DefaultEnabled = true });

app.MapGrpcService<ProductGrpcService>()
   .EnableGrpcWeb()
   .RequireAuthorization();

if (app.Environment.IsDevelopment())
    app.MapGrpcReflectionService();

app.Run();

Implementacja serwisu gRPC

public class ProductGrpcService : ProductService.ProductServiceBase
{
    private readonly IProductRepository _repo;
    private readonly ILogger<ProductGrpcService> _logger;

    public ProductGrpcService(
        IProductRepository repo,
        ILogger<ProductGrpcService> logger)
    {
        _repo = repo;
        _logger = logger;
    }

    // Unary RPC
    public override async Task<ProductResponse> GetProduct(
        GetProductRequest request,
        ServerCallContext context)
    {
        var product = await _repo.GetByIdAsync(
            request.Id, context.CancellationToken);

        if (product is null)
            throw new RpcException(new Status(
                StatusCode.NotFound,
                $"Produkt {request.Id} nie znaleziony"));

        return MapToResponse(product);
    }

    // Server-side streaming
    public override async Task ListProducts(
        ListProductsRequest request,
        IServerStreamWriter<ProductResponse> responseStream,
        ServerCallContext context)
    {
        var products = _repo.GetStreamAsync(
            request.CategoryId,
            request.PageSize,
            context.CancellationToken);

        await foreach (var product in products
            .WithCancellation(context.CancellationToken))
        {
            await responseStream.WriteAsync(
                MapToResponse(product),
                context.CancellationToken);

            // Sprawdz czy klient nie anulowal
            if (context.CancellationToken.IsCancellationRequested)
                break;
        }
    }

    // Client-side streaming
    public override async Task<ImportResult> ImportProducts(
        IAsyncStreamReader<CreateProductRequest> requestStream,
        ServerCallContext context)
    {
        int imported = 0, failed = 0;
        var errors = new List<string>();

        await foreach (var req in requestStream.ReadAllAsync(
            context.CancellationToken))
        {
            try
            {
                await _repo.CreateAsync(req, context.CancellationToken);
                imported++;
            }
            catch (Exception ex)
            {
                failed++;
                errors.Add($"Blad dla '{req.Name}': {ex.Message}");
                _logger.LogWarning(ex, "Import blad dla {Name}", req.Name);
            }
        }

        return new ImportResult
        {
            Imported = imported,
            Failed = failed,
            Errors = { errors }
        };
    }

    // Bidirectional streaming
    public override async Task SyncProducts(
        IAsyncStreamReader<ProductSyncRequest> requestStream,
        IServerStreamWriter<ProductSyncResponse> responseStream,
        ServerCallContext context)
    {
        await foreach (var req in requestStream.ReadAllAsync(
            context.CancellationToken))
        {
            var response = new ProductSyncResponse();
            try
            {
                switch (req.OperationCase)
                {
                    case ProductSyncRequest.OperationOneofCase.Create:
                        var p = await _repo.CreateAsync(
                            req.Create, context.CancellationToken);
                        response.Success = true;
                        response.ProcessedId = p.Id;
                        break;
                    case ProductSyncRequest.OperationOneofCase.Delete:
                        await _repo.DeleteAsync(
                            req.Delete.Id, context.CancellationToken);
                        response.Success = true;
                        response.ProcessedId = req.Delete.Id;
                        break;
                    default:
                        response.Success = false;
                        response.Error = "Nieznana operacja";
                        break;
                }
            }
            catch (Exception ex)
            {
                response.Success = false;
                response.Error = ex.Message;
            }

            await responseStream.WriteAsync(response,
                context.CancellationToken);
        }
    }

    private static ProductResponse MapToResponse(Product p) => new()
    {
        Id = p.Id,
        Name = p.Name,
        Price = (double)p.Price,
        Stock = p.Stock,
        CategoryName = p.Category?.Name ?? string.Empty,
        CreatedAt = Timestamp.FromDateTime(p.CreatedAt.ToUniversalTime())
    };
}

Interceptory

// Interceptor logowania (server-side)
public class LoggingInterceptor : Interceptor
{
    private readonly ILogger<LoggingInterceptor> _logger;

    public LoggingInterceptor(ILogger<LoggingInterceptor> logger)
    {
        _logger = logger;
    }

    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        _logger.LogInformation(
            "gRPC wywolanie: {Method} od {Peer}",
            context.Method, context.Peer);

        try
        {
            var response = await continuation(request, context);
            _logger.LogInformation("gRPC sukces: {Method}", context.Method);
            return response;
        }
        catch (RpcException ex)
        {
            _logger.LogWarning("gRPC blad {Status}: {Method} — {Detail}",
                ex.StatusCode, context.Method, ex.Status.Detail);
            throw;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "gRPC wyjatek: {Method}", context.Method);
            throw new RpcException(new Status(
                StatusCode.Internal, "Wewnetrzny blad serwera"));
        }
    }
}

Deadlines — limity czasu

// Klient z deadline
var channel = GrpcChannel.ForAddress("https://localhost:5001");
var client = new ProductService.ProductServiceClient(channel);

// Deadline — absolutny czas zakonczenia
var deadline = DateTime.UtcNow.AddSeconds(5);
var response = await client.GetProductAsync(
    new GetProductRequest { Id = 1 },
    new CallOptions(deadline: deadline));

// Deadline z CancellationToken
using var cts = new CancellationTokenSource(TimeSpan.FromSeconds(5));
var response2 = await client.GetProductAsync(
    new GetProductRequest { Id = 2 },
    new CallOptions(cancellationToken: cts.Token));

// Streaming z deadline
using var streamCall = client.ListProducts(
    new ListProductsRequest { PageSize = 100 },
    new CallOptions(deadline: DateTime.UtcNow.AddSeconds(30)));

await foreach (var product in streamCall.ResponseStream.ReadAllAsync())
{
    Console.WriteLine(product.Name);
}
gRPC-Web — dostep z przegladarki: Przegladarki nie wspieraja HTTP/2 trailerów wymaganych przez gRPC. Uzyj Grpc.AspNetCore.Web — middleware tlumacza gRPC-Web na gRPC. Alternatywnie: Connect protocol (nowszy standard, wsparcie w .NET 9).
