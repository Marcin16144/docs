OpenAPI / Swagger w .NET 9

Microsoft.AspNetCore.OpenApi, Swashbuckle, NSwag, Scalar UI i generowanie klientow TypeScript

Wbudowane OpenAPI w .NET 9

Od .NET 9 framework zawiera pakiet Microsoft.AspNetCore.OpenApi — natywna implementacja generowania dokumentacji OpenAPI 3.1 bez zewnetrznych zaleznosci. Jest szybszy i lepiej zintegrowany niz Swashbuckle.

// Program.cs — .NET 9 natywne OpenAPI
builder.Services.AddOpenApi(options =>
{
    options.OpenApiVersion = OpenApiSpecVersion.OpenApi3_1;

    // Transformator dokumentu
    options.AddDocumentTransformer((document, context, ct) =>
    {
        document.Info = new OpenApiInfo
        {
            Title = "Products API",
            Version = "v1",
            Description = "API do zarzadzania produktami",
            Contact = new OpenApiContact
            {
                Name = "Zespol API",
                Email = "api@example.com"
            },
            License = new OpenApiLicense
            {
                Name = "MIT",
                Url = new Uri("https://opensource.org/licenses/MIT")
            }
        };
        return Task.CompletedTask;
    });

    // Schemat bezpieczenstwa JWT
    options.AddDocumentTransformer<BearerSecuritySchemeTransformer>();
});

// Udostepnienie dokumentu JSON
app.MapOpenApi();  // /openapi/v1.json

// Wersjonowanie dokumentow
builder.Services.AddOpenApi("v1");
builder.Services.AddOpenApi("v2");
app.MapOpenApi("/openapi/{documentName}.json");

BearerSecuritySchemeTransformer

public class BearerSecuritySchemeTransformer
    : IOpenApiDocumentTransformer
{
    private readonly IAuthenticationSchemeProvider _schemes;

    public BearerSecuritySchemeTransformer(
        IAuthenticationSchemeProvider schemes)
    {
        _schemes = schemes;
    }

    public async Task TransformAsync(
        OpenApiDocument document,
        OpenApiDocumentTransformerContext context,
        CancellationToken ct)
    {
        var authSchemes = await _schemes.GetAllSchemesAsync();
        if (!authSchemes.Any(s =>
            s.Name == JwtBearerDefaults.AuthenticationScheme))
            return;

        document.Components ??= new OpenApiComponents();
        document.Components.SecuritySchemes.Add("Bearer",
            new OpenApiSecurityScheme
            {
                Type = SecuritySchemeType.Http,
                Scheme = "bearer",
                BearerFormat = "JWT",
                Description = "Podaj token JWT w formacie: Bearer {token}"
            });
    }
}

Scalar UI — nastepca Swagger UI

Scalar to nowoczesny, szybki interfejs dla OpenAPI — nastepca Swagger UI. Od .NET 9 jest rekomendowanym UI dla dokumentacji API.

// dotnet add package Scalar.AspNetCore

// Program.cs
if (app.Environment.IsDevelopment())
{
    app.MapOpenApi();
    app.MapScalarApiReference(options =>
    {
        options.Title = "Products API";
        options.Theme = ScalarTheme.Purple;
        options.DefaultHttpClient =
            new(ScalarTarget.CSharp, ScalarClient.HttpClient);
        options.Authentication = new ScalarAuthenticationOptions
        {
            PreferredSecurityScheme = "Bearer"
        };
    });
    // Dostepne pod: /scalar/v1
}

  "runtime": "Net90",

IOperationFilter — Swashbuckle (legacy)

// Swashbuckle nadal uzywa sie w projektach .NET 6-8
// dotnet add package Swashbuckle.AspNetCore

// Niestandardowy filter operacji — dodaje naglowek autoryzacji
public class AuthOperationFilter : IOperationFilter
{
    public void Apply(OpenApiOperation operation, OperationFilterContext context)
    {
        var authAttributes = context.MethodInfo
            .GetCustomAttributes(true)
            .Union(context.MethodInfo.DeclaringType!
                .GetCustomAttributes(true))
            .OfType<AuthorizeAttribute>();

        if (!authAttributes.Any()) return;

        operation.Security.Add(new OpenApiSecurityRequirement
        {
            {
                new OpenApiSecurityScheme
                {
                    Reference = new OpenApiReference
                    {
                        Type = ReferenceType.SecurityScheme,
                        Id = "Bearer"
                    }
                },
                Array.Empty<string>()
            }
        });
    }
}

// Rejestracja
builder.Services.AddSwaggerGen(c =>
{
    c.OperationFilter<AuthOperationFilter>();
    c.AddSecurityDefinition("Bearer", new OpenApiSecurityScheme
    {
        Type = SecuritySchemeType.Http,
        Scheme = "bearer",
        BearerFormat = "JWT"
    });
});
