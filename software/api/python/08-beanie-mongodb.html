Motor i Beanie — stos MongoDB dla async Python

Motor to oficjalny asynchroniczny sterownik MongoDB dla Pythona — wrapper nad PyMongo wykorzystujacy asyncio. Beanie to ODM (Object Document Mapper) zbudowany nad Motor, ktory dodaje Pydantic v2 jako walidacje i intuicyjne API do operacji na dokumentach.

Instalacja

uv add beanie motor fastapi uvicorn
Beanie automatycznie instaluje Motor jako zaleznosc. Wymaga MongoDB 4.4+ lub MongoDB Atlas.

Document — model dokumentu MongoDB

Kazdy dokument MongoDB to klasa dziedziczaca z Document (Beanie). Klasa Document rozszerza Pydantic BaseModel, wiec mozesz uzywac wszystkich walidatorow i adnotacji Pydantic.

from beanie import Document, Indexed, Link, BackLink, before_event, after_event
from beanie import Insert, Update, Delete, Replace
from pydantic import Field, EmailStr, model_validator
from datetime import datetime
from typing import Optional, Annotated
import uuid

# ---- Model kategorii ----
class Kategoria(Document):
    nazwa: Annotated[str, Indexed(unique=True)]  # unikalny index MongoDB
    slug: str
    opis: str = ""
    aktywna: bool = True

    class Settings:
        name = "kategorie"          # nazwa kolekcji w MongoDB
        use_revision = True         # optimistic locking (wersjonowanie)
        indexes = [
            [("slug", 1)],          # index na slug
        ]

# ---- Model produktu z relacja ----
class Produkt(Document):
    id: uuid.UUID = Field(default_factory=uuid.uuid4)
    nazwa: str = Field(min_length=1, max_length=200)
    opis: str = ""
    cena: float = Field(gt=0)
    stan_magazynowy: int = Field(ge=0, default=0)
    tagi: list[str] = Field(default_factory=list)
    metadane: dict = Field(default_factory=dict)

    # Relacja przez Link — lazy loading
    kategoria: Link[Kategoria] | None = None

    # Pola z automatycznymi wartosciami
    utworzono: datetime = Field(default_factory=datetime.utcnow)
    zaktualizowano: datetime = Field(default_factory=datetime.utcnow)
    aktywny: bool = True

    class Settings:
        name = "produkty"
        indexes = [
            [("nazwa", "text"), ("opis", "text")],  # text search index
            [("cena", 1)],
            [("tagi", 1)],
            [("aktywny", 1), ("cena", 1)],          # index zlozony
        ]

    # Hook przed zapisem — aktualizuj timestamp
    @before_event(Update, Replace)
    def ustaw_czas_aktualizacji(self) -> None:
        self.zaktualizowano = datetime.utcnow()

    @model_validator(mode="after")
    def sprawdz_cene_stan(self) -> "Produkt":
        if self.stan_magazynowy == 0 and self.aktywny:
            # mozna automatycznie dezaktywowac
            pass
        return self

# ---- Model uzytkownika ----
class Uzytkownik(Document):
    email: Annotated[EmailStr, Indexed(unique=True)]
    imie: str
    nazwisko: str
    rola: str = "user"
    aktywny: bool = True
    utworzono: datetime = Field(default_factory=datetime.utcnow)

    # Backlink — odwrotna relacja (nie zapisywana w dokumencie)
    zamowienia: list[BackLink["Zamowienie"]] = Field(
        default_factory=list, exclude=True
    )

    class Settings:
        name = "uzytkownicy"

class Zamowienie(Document):
    uzytkownik: Link[Uzytkownik]
    produkty: list[Link[Produkt]]
    suma: float
    status: str = "nowe"
    utworzono: datetime = Field(default_factory=datetime.utcnow)

    class Settings:
        name = "zamowienia"

CRUD — operacje na dokumentach

from beanie import PydanticObjectId
from beanie.operators import In, GTE, LTE, Set, Inc, Push

# ---- CREATE ----
async def przyklad_create() -> None:
    # Sposob 1: insert()
    kategoria = Kategoria(nazwa="Elektronika", slug="elektronika")
    await kategoria.insert()
    print(f"ID: {kategoria.id}")  # MongoDB ObjectId lub UUID

    # Sposob 2: save() — tworzy lub aktualizuje
    produkt = Produkt(
        nazwa="Laptop Pro X",
        cena=3499.99,
        stan_magazynowy=15,
        tagi=["laptop", "komputer", "praca"],
    )
    await produkt.save()

    # Bulk insert
    produkty = [
        Produkt(nazwa=f"Produkt {i}", cena=float(i * 10))
        for i in range(1, 11)
    ]
    await Produkt.insert_many(produkty)

# ---- READ ----
async def przyklad_read() -> None:
    # Pobierz po ID
    produkt = await Produkt.get(PydanticObjectId("507f1f77bcf86cd799439011"))

    # Znajdz jeden dokument
    produkt = await Produkt.find_one(Produkt.nazwa == "Laptop Pro X")

    # Znajdz wiele — z filtrowaniem i sortowaniem
    aktywne = await Produkt.find(
        Produkt.aktywny == True,
        Produkt.cena >= 100.0,
    ).sort(-Produkt.cena).limit(10).to_list()

    # Pobierz z zaladowanymi relacjami (fetch_links)
    produkt_z_kat = await Produkt.find_one(
        Produkt.nazwa == "Laptop Pro X",
        fetch_links=True,  # zaladuje Kategoria przez Link
    )

    # Wyszukiwanie tekstowe (wymaga text index)
    wyniki = await Produkt.find(
        {"$text": {"$search": "laptop wydajny"}}
    ).to_list()

    # Zlicz dokumenty
    liczba = await Produkt.find(Produkt.aktywny == True).count()

    # Sprawdz czy istnieje
    istnieje = await Produkt.find(
        Produkt.nazwa == "Laptop Pro X"
    ).exists()

# ---- UPDATE ----
async def przyklad_update() -> None:
    produkt = await Produkt.find_one(Produkt.nazwa == "Laptop Pro X")
    if produkt:
        # Aktualizuj pole i zapisz
        produkt.cena = 2999.99
        await produkt.save()

        # Aktualizuj operatorem (bez pobierania dokumentu)
        await Produkt.find(
            Produkt.stan_magazynowy == 0
        ).update(Set({Produkt.aktywny: False}))

        # Inkrementacja
        await produkt.update(Inc({Produkt.stan_magazynowy: 5}))

        # Dodaj do tablicy
        await produkt.update(Push({Produkt.tagi: "wyprzedaz"}))

# ---- DELETE ----
async def przyklad_delete() -> None:
    produkt = await Produkt.find_one(Produkt.nazwa == "Stary Produkt")
    if produkt:
        await produkt.delete()

    # Bulk delete
    await Produkt.find(Produkt.aktywny == False).delete()

Aggregation Pipeline

Porownanie sterownikow MongoDB dla Python

Cecha Beanie (ODM) MongoEngine pymongo (raw)
Walidacja Pydantic v2 (pelna) Wlasna (slabsza) Brak
Async Tak (Motor) Nie (sync only) Motor (oddzielny)
Type hints Pelne Czesciowe Brak (dict)
Agregacje Pelny pipeline Ograniczone Pelny pipeline
Relacje Link[], BackLink[] ReferenceField Reczne
FastAPI integracja Doskonala Slaba (sync) Wymaga opakowywania
Idealny dla Nowe projekty async Stare projekty Django Pelna kontrola, perf

Inicjalizacja z FastAPI lifespan

from contextlib import asynccontextmanager
from fastapi import FastAPI
from beanie import init_beanie
from motor.motor_asyncio import AsyncIOMotorClient
import os

MONGO_URL = os.environ.get("MONGO_URL", "mongodb://localhost:27017")
DB_NAME = os.environ.get("DB_NAME", "mojabaza")

@asynccontextmanager
async def lifespan(app: FastAPI):
    # STARTUP — polacz z MongoDB i inicjalizuj Beanie
    client = AsyncIOMotorClient(MONGO_URL)
    await init_beanie(
        database=client[DB_NAME],
        document_models=[
            Kategoria,
            Produkt,
            Uzytkownik,
            Zamowienie,
        ],
    )
    print(f"Polaczono z MongoDB: {DB_NAME}")
    yield
    # SHUTDOWN — zamknij polaczenie
    client.close()
    print("Rozlaczono z MongoDB")

app = FastAPI(lifespan=lifespan)

@app.get("/produkty/")
async def lista_produktow(limit: int = 20) -> list[Produkt]:
    return await Produkt.find(
        Produkt.aktywny == True
    ).sort(-Produkt.utworzono).limit(limit).to_list()

Indexed — indeksy w Beanie

Uzyj Annotated[str, Indexed(unique=True)] dla pol z indeksem unikalnym lub Indexed(index_type=pymongo.TEXT) dla wyszukiwania pelnotekstowego. Beanie tworzy indeksy automatycznie podczas init_beanie() — nie musisz tworzyc ich recznie w MongoDB.
