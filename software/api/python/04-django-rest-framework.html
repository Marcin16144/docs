Django REST Framework

Serializers, ViewSets, Routers, uprawnienia, autentykacja JWT, paginacja, filtrowanie

Wprowadzenie do DRF

Django REST Framework (DRF) to najpopularniejsza biblioteka do budowania REST API w Django. Oferuje dojrzaly ekosystem, swietna dokumentacje i mnogo gotowych komponentow. Jest wyborem numer jeden dla zespolow ktore juz uzywaja Django i chca szybko wystawic API.

Instalacja

pip install djangorestframework djangorestframework-simplejwt django-filter
Dodaj 'rest_framework' i 'django_filters' do INSTALLED_APPS.

Serializers — walidacja i serializacja

Serializatory DRF pelnia role analogiczna do Pydantic w FastAPI — konwertuja zlozone obiekty (modele Django) na Python dict/JSON i odwrotnie, oraz waliduja dane wejsciowe.

# models.py
from django.db import models
from django.contrib.auth.models import User

class Kategoria(models.Model):
    nazwa = models.CharField(max_length=100)
    slug = models.SlugField(unique=True)

    class Meta:
        verbose_name_plural = "kategorie"

class Artykul(models.Model):
    tytul = models.CharField(max_length=300)
    tresc = models.TextField()
    autor = models.ForeignKey(User, on_delete=models.CASCADE, related_name="artykuly")
    kategoria = models.ForeignKey(Kategoria, on_delete=models.SET_NULL, null=True)
    opublikowano = models.BooleanField(default=False)
    utworzono = models.DateTimeField(auto_now_add=True)
    zaktualizowano = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-utworzono"]

# serializers.py
from rest_framework import serializers
from .models import Artykul, Kategoria

class KategoriaSerializer(serializers.ModelSerializer):
    class Meta:
        model = Kategoria
        fields = ["id", "nazwa", "slug"]

class ArtykulListSerializer(serializers.ModelSerializer):
    """Lekki serializer do list — bez pelnej tresci."""
    autor_nazwa = serializers.SerializerMethodField()
    kategoria = KategoriaSerializer(read_only=True)

    class Meta:
        model = Artykul
        fields = ["id", "tytul", "autor_nazwa", "kategoria", "opublikowano", "utworzono"]

    def get_autor_nazwa(self, obj) -> str:
        return f"{obj.autor.first_name} {obj.autor.last_name}".strip() or obj.autor.username

class ArtykulDetailSerializer(ArtykulListSerializer):
    """Pelny serializer — do widokow detali."""
    class Meta(ArtykulListSerializer.Meta):
        fields = ArtykulListSerializer.Meta.fields + ["tresc", "zaktualizowano"]

class ArtykulTworzSerializer(serializers.ModelSerializer):
    """Serializer do tworzenia/edycji — przyjmuje ID kategorii."""
    class Meta:
        model = Artykul
        fields = ["tytul", "tresc", "kategoria", "opublikowany"]

    def validate_tytul(self, value: str) -> str:
        if len(value) < 5:
            raise serializers.ValidationError("Tytul musi miec min. 5 znakow.")
        return value.strip()

    def create(self, validated_data: dict) -> Artykul:
        # autor jest ustawiany automatycznie z request.user
        validated_data["autor"] = self.context["request"].user
        return super().create(validated_data)

ViewSets i Routers

ViewSet laczy wiele powiazan widokow (list, detail, create, update, delete) w jednej klasie. Router automatycznie generuje URL-e dla ViewSet.

# views.py
from rest_framework import viewsets, permissions, status, filters
from rest_framework.decorators import action
from rest_framework.request import Request
from rest_framework.response import Response
from django_filters.rest_framework import DjangoFilterBackend
from .models import Artykul, Kategoria
from .serializers import (
    ArtykulListSerializer, ArtykulDetailSerializer,
    ArtykulTworzSerializer, KategoriaSerializer,
)
from .permissions import IsAuthorOrReadOnly

class ArtykulViewSet(viewsets.ModelViewSet):
    """
    CRUD dla artykulow.
    list:   GET  /artykuly/
    create: POST /artykuly/
    retrieve: GET /artykuly/{id}/
    update: PUT  /artykuly/{id}/
    partial_update: PATCH /artykuly/{id}/
    destroy: DELETE /artykuly/{id}/
    """
    queryset = Artykul.objects.select_related("autor", "kategoria").all()
    permission_classes = [permissions.IsAuthenticatedOrReadOnly, IsAuthorOrReadOnly]

    # Filtrowanie, sortowanie, wyszukiwanie
    filter_backends = [DjangoFilterBackend, filters.SearchFilter, filters.OrderingFilter]
    filterset_fields = ["kategoria", "opublikowano", "autor"]
    search_fields = ["tytul", "tresc"]
    ordering_fields = ["utworzono", "tytul"]
    ordering = ["-utworzono"]

    def get_serializer_class(self):
        """Rozny serializer dla listy, detalu i tworzenia."""
        if self.action == "list":
            return ArtykulListSerializer
        if self.action in ("create", "update", "partial_update"):
            return ArtykulTworzSerializer
        return ArtykulDetailSerializer

    def get_queryset(self):
        qs = super().get_queryset()
        if self.action == "list" and not self.request.user.is_staff:
            qs = qs.filter(opublikowano=True)
        return qs

    # Niestandardowa akcja: POST /artykuly/{id}/opublikuj/
    @action(detail=True, methods=["post"], permission_classes=[permissions.IsAdminUser])
    def opublikuj(self, request: Request, pk=None) -> Response:
        artykul = self.get_object()
        artykul.opublikowano = True
        artykul.save(update_fields=["opublikowano"])
        return Response({"status": "opublikowano"})

    # Niestandardowa akcja: GET /artykuly/moje/
    @action(detail=False, methods=["get"], permission_classes=[permissions.IsAuthenticated])
    def moje(self, request: Request) -> Response:
        qs = Artykul.objects.filter(autor=request.user)
        serializer = ArtykulListSerializer(qs, many=True, context={"request": request})
        return Response(serializer.data)

class KategoriaViewSet(viewsets.ReadOnlyModelViewSet):
    """Tylko odczyt — brak tworzenia/edycji przez API."""
    queryset = Kategoria.objects.all()
    serializer_class = KategoriaSerializer
    permission_classes = [permissions.AllowAny]

# urls.py
from rest_framework.routers import DefaultRouter
from django.urls import path, include

router = DefaultRouter()
router.register("artykuly", ArtykulViewSet, basename="artykul")
router.register("kategorie", KategoriaViewSet, basename="kategoria")

urlpatterns = [
    path("api/v1/", include(router.urls)),
]

Uprawnienia — permissions

DRF umozliwia tworzenie niestandardowych klas uprawnien poprzez dziedziczenie z BasePermission.

Klasa uprawnienia Opis Typowe uzycie
AllowAnyBez ograniczenPubliczne endpointy
IsAuthenticatedWymaga zalogowaniaWiekzosc API
IsAdminUserTylko staff/adminPanel administracyjny
IsAuthenticatedOrReadOnlyZapis: auth, odczyt: wszyscyBlogi, CMS
DjangoModelPermissionsUprawnienia modelu DjangoIntegracja z admin
Wlasna klasaDowolna logikaNp. IsAuthorOrReadOnly

Paginacja i filtrowanie

DRF oferuje wbudowana paginacje — wystarczy skonfigurowac w settings.py:

# settings.py — globalna konfiguracja DRF
REST_FRAMEWORK = {
    # Autentykacja
    "DEFAULT_AUTHENTICATION_CLASSES": [
        "rest_framework_simplejwt.authentication.JWTAuthentication",
        "rest_framework.authentication.SessionAuthentication",
    ],
    # Uprawnienia domyslne
    "DEFAULT_PERMISSION_CLASSES": [
        "rest_framework.permissions.IsAuthenticated",
    ],
    # Paginacja
    "DEFAULT_PAGINATION_CLASS": "rest_framework.pagination.PageNumberPagination",
    "PAGE_SIZE": 20,
    # Filtrowanie
    "DEFAULT_FILTER_BACKENDS": [
        "django_filters.rest_framework.DjangoFilterBackend",
        "rest_framework.filters.SearchFilter",
        "rest_framework.filters.OrderingFilter",
    ],
    # Throttling (rate limiting)
    "DEFAULT_THROTTLE_CLASSES": [
        "rest_framework.throttling.AnonRateThrottle",
        "rest_framework.throttling.UserRateThrottle",
    ],
    "DEFAULT_THROTTLE_RATES": {
        "anon": "100/day",
        "user": "1000/day",
    },
    # Format odpowiedzi
    "DEFAULT_RENDERER_CLASSES": [
        "rest_framework.renderers.JSONRenderer",
    ],
}

Wydajnosc — select_related i prefetch_related

Zawsze uzywaj select_related() dla ForeignKey i prefetch_related() dla ManyToMany w get_queryset(). Bez tego kazdyv rekord wygeneruje osobne zapytanie SQL (problem N+1).
