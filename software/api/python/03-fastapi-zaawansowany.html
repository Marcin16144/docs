Middleware

Middleware to warstwa kodu wykonywana dla kazdego zadania przed i po obsludze przez endpoint. FastAPI/Starlette oferuje dwa sposoby definiowania middleware.

Sposob Kiedy uzywac Uwagi
@app.middleware("http") proste przypadki, logging, timing Funkcja async, latwiejszy dostep do request/response
BaseHTTPMiddleware zlozony middleware jako klasa Latwa do testowania i wielokrotnego uzycia
app.add_middleware() gotowe middleware (CORS, GZip itd.) Standardowe Starlette middleware 
from fastapi import FastAPI, Request, Response
from fastapi.middleware.cors import CORSMiddleware
from fastapi.middleware.gzip import GZipMiddleware
from starlette.middleware.base import BaseHTTPMiddleware
import time
import uuid

app = FastAPI()

# ---- Middleware przez dekorator ----
@app.middleware("http")
async def dodaj_czas_odpowiedzi(request: Request, call_next) -> Response:
    start = time.perf_counter()
    response = await call_next(request)
    czas = time.perf_counter() - start
    response.headers["X-Process-Time"] = f"{czas:.4f}s"
    return response

# ---- Middleware jako klasa ----
class RequestIdMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next) -> Response:
        request_id = str(uuid.uuid4())
        request.state.request_id = request_id
        response = await call_next(request)
        response.headers["X-Request-ID"] = request_id
        return response

app.add_middleware(RequestIdMiddleware)

# ---- CORS ----
app.add_middleware(
    CORSMiddleware,
    allow_origins=["https://moja-aplikacja.pl", "http://localhost:3000"],
    allow_credentials=True,
    allow_methods=["GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"],
    allow_headers=["Authorization", "Content-Type", "X-Request-ID"],
    expose_headers=["X-Process-Time", "X-Request-ID"],
    max_age=3600,  # cache preflight przez 1h
)

# ---- GZip kompresja odpowiedzi ----
app.add_middleware(GZipMiddleware, minimum_size=1000)

Background Tasks

BackgroundTasks pozwala uruchamiac zadania po wyslaniu odpowiedzi do klienta — bez blokowania. Dobre dla: wysylania emaili, logowania, powiadomien. Do ciezkich zadan (kolejki) uzyj Celery lub ARQ.

from fastapi import BackgroundTasks
import asyncio

async def wyslij_email_powitalny(email: str, imie: str) -> None:
    """Symulacja wysylania emaila — moze trwac dlugo."""
    await asyncio.sleep(2)
    print(f"Email wyslany do {email}: Witaj {imie}!")

async def zapisz_log_akcji(akcja: str, user_id: int) -> None:
    await asyncio.sleep(0.1)
    print(f"LOG: user={user_id} akcja={akcja}")

class RejestrujDTO(BaseModel):
    email: str
    imie: str

@app.post("/rejestruj", status_code=201)
async def rejestruj(
    dane: RejestrujDTO,
    background_tasks: BackgroundTasks,
) -> dict[str, str]:
    # Zapisz uzytkownika do bazy...
    nowy_id = 42

    # Zaplanuj zadania w tle (wykonaja sie po zwroceniu odpowiedzi)
    background_tasks.add_task(wyslij_email_powitalny, dane.email, dane.imie)
    background_tasks.add_task(zapisz_log_akcji, "rejestracja", nowy_id)

    # Odpowiedz wraca NATYCHMIAST, email i log w tle
    return {"status": "zarejestrowano", "user_id": str(nowy_id)}

WebSocket

FastAPI wspiera WebSocket poprzez Starlette. Endpoint @app.websocket() utrzymuje polaczenie dwukierunkowe — idealne dla czatow, powiadomien real-time, dashboardow.

File Upload — UploadFile

FastAPI uzywa UploadFile do obslugi przesilanych plikow. Pliki sa przesylane jako multipart/form-data, nie JSON.

from fastapi import WebSocket, WebSocketDisconnect, UploadFile, File, Form
from typing import Annotated
import aiofiles
import os

# ---- WebSocket endpoint ----
class ConnectionManager:
    def __init__(self):
        self.active: list[WebSocket] = []

    async def connect(self, ws: WebSocket):
        await ws.accept()
        self.active.append(ws)

    def disconnect(self, ws: WebSocket):
        self.active.remove(ws)

    async def broadcast(self, message: str):
        for connection in self.active:
            await connection.send_text(message)

manager = ConnectionManager()

@app.websocket("/ws/{room_id}")
async def websocket_endpoint(websocket: WebSocket, room_id: str):
    await manager.connect(websocket)
    try:
        while True:
            data = await websocket.receive_text()
            await manager.broadcast(f"[{room_id}] {data}")
    except WebSocketDisconnect:
        manager.disconnect(websocket)
        await manager.broadcast(f"[{room_id}] Uzytkownik rozlaczony")

# ---- File upload ----
UPLOAD_DIR = "/tmp/uploads"
os.makedirs(UPLOAD_DIR, exist_ok=True)

@app.post("/upload-plik/")
async def upload_plik(
    plik: Annotated[UploadFile, File(description="Plik do wgrania")],
    opis: Annotated[str, Form()] = "",
) -> dict:
    # Walidacja rozszerzenia
    dozwolone = {".jpg", ".jpeg", ".png", ".pdf"}
    ext = os.path.splitext(plik.filename or "")[1].lower()
    if ext not in dozwolone:
        from fastapi import HTTPException
        raise HTTPException(400, f"Niedozwolone rozszerzenie: {ext}")

    sciezka = os.path.join(UPLOAD_DIR, plik.filename or "plik")
    async with aiofiles.open(sciezka, "wb") as f:
        while chunk := await plik.read(1024 * 64):  # 64KB chunk
            await f.write(chunk)

    return {
        "nazwa": plik.filename,
        "rozmiar": plik.size,
        "typ": plik.content_type,
        "opis": opis,
    }

@app.post("/upload-wiele/")
async def upload_wiele(
    pliki: Annotated[list[UploadFile], File()],
) -> dict:
    wyniki = []
    for p in pliki:
        wyniki.append({"nazwa": p.filename, "rozmiar": p.size})
    return {"przeslane": len(wyniki), "pliki": wyniki}

OAuth2 i bezpieczenstwo

OAuth2PasswordBearer to schemat bezpieczenstwa FastAPI do obslugi tokenow Bearer. Automatycznie pojawia sie w Swagger UI jako pole autoryzacji.

Lifespan — startup i shutdown

Nowoczesny sposob inicjalizacji aplikacji (zalecany od FastAPI 0.93+). Zastepuje przestarzale @app.on_event("startup").

Custom Exception Handlers

Mozesz definiowac globalne handlery dla dowolnych wyjatkow za pomoca @app.exception_handler(MyException). FastAPI domyslnie zwraca 422 dla bledow walidacji Pydantic i 404 dla brakujacych zasobow — mozesz je nadpisac aby dostosowac format odpowiedzi bledu.

Starlette — integracja niskopoziomowa

FastAPI jest zbudowany na Starlette — mozesz uzywac bezposrednio wszystkich komponentow Starlette: Request, Response, JSONResponse, StreamingResponse, FileResponse, StaticFiles, Mount. FastAPI to Starlette z dodatkiem Pydantic i automatycznej dokumentacji.
