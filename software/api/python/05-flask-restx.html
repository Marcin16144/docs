Flask + Flask-RESTX

App factory, Blueprint, Flask-RESTX, marshmallow, Flask-SQLAlchemy, Swagger UI

Flask — kiedy uzywac

Flask to mikro-framework — daje minimalne narzedzia i pozwala samodzielnie dobrac reszte stosu. Idealny gdy potrzebujesz pelnej kontroli, masz specyficzne wymagania integracyjne albo budujesz maly serwis bez potrzeby pelnego frameworka jak Django.

Kryterium Flask + Flask-RESTX FastAPI
Paradygmat WSGI (synchroniczny), opcjonalny async ASGI (async natywny)
Wydajnosc Srednia (WSGI) Wysoka (ASGI + async)
Dokumentacja OpenAPI Automatyczna przez Flask-RESTX Automatyczna (wbudowana)
Walidacja danych marshmallow lub wtyczki Pydantic (wbudowany)
ORM Flask-SQLAlchemy (popularny) Dowolny (SQLAlchemy, Tortoise)
Dojrzalosc ekosystemu Bardzo dojrzaly, duzo wtyczek Szybko rosnacy
Krzywa uczenia sie Niska — prosty poczatek Srednia — wymaga znajom. async
Idealny dla Male API, mikroserwisy, legacy Nowe projekty, high-perf API

App Factory Pattern

Wzorzec fabryki aplikacji pozwala tworzyc wiele instancji Flask z rozna konfiguracja (np. testowa vs produkcyjna) i unika globalnego stanu. Jest standardem w wiekszych projektach Flask.

# config.py
import os
from dataclasses import dataclass

@dataclass
class Config:
    SECRET_KEY: str = os.environ.get("SECRET_KEY", "dev-secret-zmien-na-produkcji")
    SQLALCHEMY_DATABASE_URI: str = os.environ.get("DATABASE_URL", "sqlite:///dev.db")
    SQLALCHEMY_TRACK_MODIFICATIONS: bool = False
    JWT_SECRET_KEY: str = os.environ.get("JWT_SECRET_KEY", "jwt-secret")
    DEBUG: bool = False

@dataclass
class DevelopmentConfig(Config):
    DEBUG: bool = True
    SQLALCHEMY_ECHO: bool = True

@dataclass
class ProductionConfig(Config):
    SQLALCHEMY_DATABASE_URI: str = os.environ["DATABASE_URL"]

config_map = {
    "development": DevelopmentConfig,
    "production": ProductionConfig,
    "default": DevelopmentConfig,
}

# extensions.py — obiekty rozszerzen bez aplikacji (late binding)
from flask_sqlalchemy import SQLAlchemy
from flask_migrate import Migrate
from flask_jwt_extended import JWTManager

db = SQLAlchemy()
migrate = Migrate()
jwt = JWTManager()

# app/__init__.py — fabryka aplikacji
from flask import Flask
from .extensions import db, migrate, jwt
from .config import config_map
import os

def create_app(config_name: str | None = None) -> Flask:
    if config_name is None:
        config_name = os.environ.get("FLASK_ENV", "default")

    app = Flask(__name__)
    app.config.from_object(config_map[config_name])

    # Inicjalizacja rozszerzen
    db.init_app(app)
    migrate.init_app(app, db)
    jwt.init_app(app)

    # Rejestracja Blueprint i API
    from .api import api_bp
    app.register_blueprint(api_bp, url_prefix="/api/v1")

    return app

Flask-RESTX — Api, Resource, Namespace

Flask-RESTX (nastepca Flask-RESTPlus) dodaje do Flaska klasy Resource zamiast funkcji widoku oraz automatyczna generacje Swagger UI bez zadnej dodatkowej konfiguracji.

# api/__init__.py
from flask import Blueprint
from flask_restx import Api

api_bp = Blueprint("api", __name__)

api = Api(
    api_bp,
    version="1.0",
    title="Moje Flask API",
    description="API zbudowane z Flask-RESTX",
    doc="/docs",  # Swagger UI pod /api/v1/docs
)

# Importuj namespace po inicjalizacji api
from .produkty import ns_produkty
from .uzytkownicy import ns_uzytkownicy

api.add_namespace(ns_produkty, path="/produkty")
api.add_namespace(ns_uzytkownicy, path="/uzytkownicy")

# api/produkty.py
from flask_restx import Namespace, Resource, fields
from flask import request
from marshmallow import Schema, fields as mf, validate, ValidationError
from ..extensions import db

ns_produkty = Namespace("produkty", description="Operacje na produktach")

# ---- Model do dokumentacji Swagger ----
produkt_model = ns_produkty.model("Produkt", {
    "id": fields.Integer(readonly=True),
    "nazwa": fields.String(required=True, description="Nazwa produktu"),
    "cena": fields.Float(required=True, description="Cena brutto"),
    "kategoria": fields.String(description="Kategoria"),
    "dostepny": fields.Boolean(default=True),
})

produkt_input_model = ns_produkty.model("ProduktInput", {
    "nazwa": fields.String(required=True),
    "cena": fields.Float(required=True),
    "kategoria": fields.String(),
})

# ---- Marshmallow Schema do walidacji ----
class ProduktSchema(Schema):
    nazwa = mf.Str(required=True, validate=validate.Length(min=1, max=200))
    cena = mf.Float(required=True, validate=validate.Range(min=0.01))
    kategoria = mf.Str(load_default="ogolne")
    dostepny = mf.Bool(load_default=True)

produkt_schema = ProduktSchema()
produkty_schema = ProduktSchema(many=True)

# ---- Resource — lista i tworzenie ----
@ns_produkty.route("/")
class ProduktLista(Resource):
    @ns_produkty.marshal_list_with(produkt_model)
    def get(self):
        """Pobierz liste produktow."""
        return []  # db.session.execute(select(Produkt)).scalars().all()

    @ns_produkty.expect(produkt_input_model, validate=True)
    @ns_produkty.marshal_with(produkt_model, code=201)
    def post(self):
        """Utworz nowy produkt."""
        try:
            dane = produkt_schema.load(request.json)
        except ValidationError as err:
            ns_produkty.abort(400, str(err.messages))
        # produkt = Produkt(**dane); db.session.add(produkt); db.session.commit()
        return {"id": 1, **dane}, 201

# ---- Resource — pojedynczy zasob ----
@ns_produkty.route("/")
@ns_produkty.param("produkt_id", "ID produktu")
@ns_produkty.response(404, "Nie znaleziono")
class ProduktZasob(Resource):
    @ns_produkty.marshal_with(produkt_model)
    def get(self, produkt_id: int):
        """Pobierz produkt po ID."""
        # produkt = db.get_or_404(Produkt, produkt_id)
        return {"id": produkt_id, "nazwa": "Przykladowy", "cena": 99.99, "dostepny": True}

    @ns_produkty.expect(produkt_input_model)
    @ns_produkty.marshal_with(produkt_model)
    def put(self, produkt_id: int):
        """Zastap produkt."""
        dane = produkt_schema.load(request.json)
        return {"id": produkt_id, **dane}

    @ns_produkty.response(204, "Usunieto")
    def delete(self, produkt_id: int):
        """Usun produkt."""
        return "", 204

Flask-SQLAlchemy i Flask-Migrate

Flask-SQLAlchemy integruje SQLAlchemy z kontekstem aplikacji Flask. Flask-Migrate (oparty na Alembic) zarzadza migracjami schematu bazy danych.

Komendy migracji

  • flask db init — inicjalizacja katalogu migrations/
  • flask db migrate -m "Dodaj tabele produktow" — wygeneruj migrace
  • flask db upgrade — zastosuj migracje
  • flask db downgrade — cofnij ostatnia migrace

reqparse vs marshmallow

reqparse (wbudowany w Flask-RESTX) jest przestarzaly i zostanie usuniecie w przyszlych wersjach. Uzywaj marshmallow do walidacji — oferuje bardziej elastyczne schematy, zagniezdzanie, niestandardowe typy i lepsza obsluge bledow.
