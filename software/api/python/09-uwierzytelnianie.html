Uwierzytelnianie Python API

JWT (python-jose), OAuth2, refresh token, Auth0, Azure AD (MSAL), passlib, rate limiting (slowapi)

JWT — JSON Web Tokens

JWT to standard tokenow nosnych (Bearer tokens) zawierajacych zakodowane claims (twierdzenia) o uzytkownikach. Skladaja sie z trzech czesci: header, payload, signature — oddzielonych kropkami. Serwer nie musi przechowywac sesji — token jest weryfikowany przez podpis kryptograficzny.

Typ tokenu Czas zycia Zastosowanie Przechowywanie
Access Token 15-60 minut Autoryzacja requestow API Pamiec (memory), nie localStorage
Refresh Token 7-30 dni Odnowienie access tokenu HttpOnly cookie (bezpieczne)
ID Token (OIDC) jednorazowy Informacje o uzytkownikuu Klient (po weryfikacji)

JWT z python-jose w FastAPI

from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer, OAuth2PasswordRequestForm
from jose import JWTError, jwt
from passlib.context import CryptContext
from pydantic import BaseModel
from datetime import datetime, timedelta, timezone
from typing import Annotated
import os

app = FastAPI()

# ---- Konfiguracja ----
SECRET_KEY = os.environ.get("SECRET_KEY", "zmien-na-produkcji-min-32-znaki!!!")
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 30
REFRESH_TOKEN_EXPIRE_DAYS = 7

# ---- Haszowanie hasel (bcrypt przez passlib) ----
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def haszuj_haslo(haslo: str) -> str:
    return pwd_context.hash(haslo)

def weryfikuj_haslo(haslo_plain: str, haslo_hash: str) -> bool:
    return pwd_context.verify(haslo_plain, haslo_hash)

# ---- Schemat OAuth2 — wskazuje endpoint logowania ----
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/auth/token")

# ---- Schematy Pydantic ----
class TokenOdpowiedz(BaseModel):
    access_token: str
    refresh_token: str
    token_type: str = "bearer"
    expires_in: int = ACCESS_TOKEN_EXPIRE_MINUTES * 60

class TokenData(BaseModel):
    sub: str | None = None
    rola: str = "user"

class UzytkownikDTO(BaseModel):
    id: int
    email: str
    imie: str
    rola: str

# ---- Tworzenie tokenow ----
def utworz_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
    payload = data.copy()
    expire = datetime.now(timezone.utc) + (
        expires_delta or timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    )
    payload.update({
        "exp": expire,
        "iat": datetime.now(timezone.utc),
        "type": "access",
    })
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

def utworz_refresh_token(sub: str) -> str:
    payload = {
        "sub": sub,
        "exp": datetime.now(timezone.utc) + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS),
        "iat": datetime.now(timezone.utc),
        "type": "refresh",
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

def dekoduj_token(token: str) -> TokenData:
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        sub: str | None = payload.get("sub")
        if sub is None:
            raise HTTPException(status_code=401, detail="Nieprawidlowy token")
        return TokenData(sub=sub, rola=payload.get("rola", "user"))
    except JWTError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Nie mozna zweryfikowac tokenu",
            headers={"WWW-Authenticate": "Bearer"},
        )

# ---- Dependency: biezacy uzytkownik ----
async def get_current_user(
    token: Annotated[str, Depends(oauth2_scheme)]
) -> UzytkownikDTO:
    token_data = dekoduj_token(token)
    # W realnej aplikacji: pobranie uzytkownika z bazy po token_data.sub
    return UzytkownikDTO(
        id=int(token_data.sub or "0"),
        email="jan@example.com",
        imie="Jan",
        rola=token_data.rola,
    )

async def wymaga_admina(
    uzytkownik: Annotated[UzytkownikDTO, Depends(get_current_user)]
) -> UzytkownikDTO:
    if uzytkownik.rola != "admin":
        raise HTTPException(status_code=403, detail="Wymagane uprawnienia admina")
    return uzytkownik

# ---- Endpointy autentykacji ----
from fastapi import APIRouter
from fastapi.responses import JSONResponse

router_auth = APIRouter(prefix="/auth", tags=["auth"])

@router_auth.post("/token", response_model=TokenOdpowiedz)
async def login(form_data: Annotated[OAuth2PasswordRequestForm, Depends()]) -> TokenOdpowiedz:
    # W realnej aplikacji: weryfikacja z baza danych
    if form_data.username != "jan@example.com" or form_data.password != "tajne":
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Nieprawidlowe dane logowania",
            headers={"WWW-Authenticate": "Bearer"},
        )
    user_id = "1"
    access_token = utworz_access_token({"sub": user_id, "rola": "user"})
    refresh_token = utworz_refresh_token(user_id)
    return TokenOdpowiedz(access_token=access_token, refresh_token=refresh_token)

@router_auth.post("/refresh", response_model=TokenOdpowiedz)
async def refresh(refresh_token: str) -> TokenOdpowiedz:
    token_data = dekoduj_token(refresh_token)
    new_access = utworz_access_token({"sub": token_data.sub, "rola": token_data.rola})
    new_refresh = utworz_refresh_token(token_data.sub or "")
    return TokenOdpowiedz(access_token=new_access, refresh_token=new_refresh)

@router_auth.get("/me", response_model=UzytkownikDTO)
async def me(uzytkownik: Annotated[UzytkownikDTO, Depends(get_current_user)]) -> UzytkownikDTO:
    return uzytkownik

app.include_router(router_auth)

Rate Limiting — slowapi

slowapi to biblioteka rate limiting dla FastAPI oparta na limits. Oferuje podobne API do Flask-Limiter.

from slowapi import Limiter, _rate_limit_exceeded_handler
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded
from fastapi import Request

# Inicjalizacja limitera — klucz po IP
limiter = Limiter(key_func=get_remote_address)
app.state.limiter = limiter
app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)

# Zastosowanie limitow na endpointy
@app.get("/publiczny/dane")
@limiter.limit("100/minute")
async def publiczne_dane(request: Request) -> dict:
    return {"dane": "publiczne"}

@app.post("/auth/token")
@limiter.limit("5/minute;20/hour")  # zlozony limit
async def login_z_limitem(request: Request, form_data: OAuth2PasswordRequestForm = Depends()):
    return {"token": "..."}

# Limit globalny dla calej aplikacji
@app.middleware("http")
async def global_rate_limit(request: Request, call_next):
    # mozna dodac globalny limit przez middleware
    return await call_next(request)

Auth0 i Azure AD (MSAL)

Dla produkcyjnych systemow warto rozwazyc zewnetrznych dostawcow tozsamosci zamiast implementowac autentykacje od zera.

Auth0 — weryfikacja JWT

Auth0 wystawia tokeny JWT podpisane kluczem RSA. Weryfikacja polega na pobraniu publicznego klucza z https://TWOJA_DOMENA.auth0.com/.well-known/jwks.json i weryfikacji podpisu. Biblioteka python-jose[cryptography] obsluguje RSA.

Azure AD — MSAL Python

Pakiet msal (Microsoft Authentication Library) dostarcza gotowe flow dla Azure AD: ConfidentialClientApplication dla serwer-serwer (client credentials), PublicClientApplication dla logowania uzytkownikow. Tokeny Azure AD weryfikuj przez jwks_uri z OpenID configuration endpoint.

Bezpieczenstwo — wazne zasady

  • Nigdy nie przechowuj secret key w kodzie — uzyj zmiennych srodowiskowych
  • Uzyj bcrypt (przez passlib) do hasel — nigdy MD5, SHA1, SHA256
  • Refresh tokeny przechowuj w HttpOnly, Secure, SameSite=Strict cookies
  • Implementuj blackliste tokenow przy wylogowaniu (Redis lub baza danych)
  • Ustaw krotki czas zycia access tokena (15-30 min)
  • Zawsze uzyj HTTPS na produkcji — tokeny Bearer sa jawne
