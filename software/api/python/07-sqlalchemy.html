SQLAlchemy 2.0 — nowy styl

SQLAlchemy 2.0 wprowadzilo kompletnie nowe API dla modeli ORM oparte na Mapped[] i mapped_column(). Stary styl z Column() nadal dziala, ale nowy jest bardziej czytelny, w pelni type-safe i zalecany do nowych projektow.

Instalacja

uv add sqlalchemy[asyncio] asyncpg greenlet alembic
Dla PostgreSQL async uzywaj asyncpg. Dla SQLite async: aiosqlite.

Modele ORM — nowy styl 2.0

from sqlalchemy import String, Integer, Float, Boolean, DateTime, ForeignKey, Text
from sqlalchemy import func, select, update, delete
from sqlalchemy.orm import (
    DeclarativeBase, Mapped, mapped_column, relationship, selectinload
)
from sqlalchemy.ext.asyncio import (
    AsyncSession, AsyncEngine, create_async_engine, async_sessionmaker
)
from datetime import datetime
from typing import Optional
import uuid

# ---- Baza dla wszystkich modeli ----
class Base(DeclarativeBase):
    pass

# ---- Mixin z wspolnymi polami ----
class TimestampMixin:
    utworzono: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), server_default=func.now(), nullable=False
    )
    zaktualizowano: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), server_default=func.now(),
        onupdate=func.now(), nullable=False
    )

# ---- Model uzytkownika ----
class Uzytkownik(TimestampMixin, Base):
    __tablename__ = "uzytkownicy"

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    email: Mapped[str] = mapped_column(String(255), unique=True, nullable=False, index=True)
    imie: Mapped[str] = mapped_column(String(100), nullable=False)
    nazwisko: Mapped[str] = mapped_column(String(100), nullable=False)
    aktywny: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)

    # Relacja one-to-many
    artykuly: Mapped[list["Artykul"]] = relationship(
        back_populates="autor", cascade="all, delete-orphan", lazy="select"
    )

    def __repr__(self) -> str:
        return f"Uzytkownik(id={self.id}, email={self.email!r})"

# ---- Model artykulu ----
class Artykul(TimestampMixin, Base):
    __tablename__ = "artykuly"

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    tytul: Mapped[str] = mapped_column(String(300), nullable=False)
    tresc: Mapped[str] = mapped_column(Text, nullable=False)
    opublikowano: Mapped[bool] = mapped_column(Boolean, default=False)
    autor_id: Mapped[int] = mapped_column(ForeignKey("uzytkownicy.id"), nullable=False)

    # Relacja many-to-one
    autor: Mapped["Uzytkownik"] = relationship(back_populates="artykuly")

    # Relacja many-to-many przez tabele asocjacyjna
    tagi: Mapped[list["Tag"]] = relationship(
        secondary="artykul_tagi", back_populates="artykuly"
    )

class Tag(Base):
    __tablename__ = "tagi"
    id: Mapped[int] = mapped_column(primary_key=True)
    nazwa: Mapped[str] = mapped_column(String(50), unique=True)
    artykuly: Mapped[list["Artykul"]] = relationship(
        secondary="artykul_tagi", back_populates="tagi"
    )

# Tabela asocjacyjna many-to-many
from sqlalchemy import Table, Column
artykul_tagi = Table(
    "artykul_tagi", Base.metadata,
    Column("artykul_id", ForeignKey("artykuly.id"), primary_key=True),
    Column("tag_id", ForeignKey("tagi.id"), primary_key=True),
)

AsyncEngine i AsyncSession

from contextlib import asynccontextmanager
from typing import AsyncGenerator

# ---- Tworzenie silnika async ----
DATABASE_URL = "postgresql+asyncpg://user:password@localhost:5432/mojabaza"

engine: AsyncEngine = create_async_engine(
    DATABASE_URL,
    echo=False,           # True = loguj SQL (tylko dev)
    pool_size=10,         # stale polaczenia w puli
    max_overflow=20,      # dodatkowe polaczenia ponad pool_size
    pool_timeout=30,      # max czas oczekiwania na polaczenie
    pool_pre_ping=True,   # sprawdzaj polaczenie przed uzyciem
)

# ---- Fabryka sesji ----
AsyncSessionLocal = async_sessionmaker(
    bind=engine,
    class_=AsyncSession,
    expire_on_commit=False,  # wazne: nie inwaliduj obiektow po commit
    autoflush=False,
)

# ---- Dependency dla FastAPI ----
async def get_db() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionLocal() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise

# ---- Inicjalizacja tabel (dev) ----
async def init_db() -> None:
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

# ---- Repozytorium — wzorzec dostep do danych ----
class UzytkownikRepo:
    def __init__(self, session: AsyncSession) -> None:
        self.session = session

    async def pobierz_po_id(self, user_id: int) -> Uzytkownik | None:
        stmt = (
            select(Uzytkownik)
            .where(Uzytkownik.id == user_id)
            .options(selectinload(Uzytkownik.artykuly))  # eager load
        )
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def lista(self, skip: int = 0, limit: int = 20) -> list[Uzytkownik]:
        stmt = select(Uzytkownik).offset(skip).limit(limit).order_by(Uzytkownik.id)
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def utworz(self, email: str, imie: str, nazwisko: str) -> Uzytkownik:
        uzytkownik = Uzytkownik(email=email, imie=imie, nazwisko=nazwisko)
        self.session.add(uzytkownik)
        await self.session.flush()  # generuje ID bez commit
        return uzytkownik

    async def aktualizuj(self, user_id: int, **dane) -> int:
        stmt = (
            update(Uzytkownik)
            .where(Uzytkownik.id == user_id)
            .values(**dane)
            .returning(Uzytkownik.id)
        )
        result = await self.session.execute(stmt)
        return result.scalar_one()

    async def usun(self, user_id: int) -> bool:
        stmt = delete(Uzytkownik).where(Uzytkownik.id == user_id)
        result = await self.session.execute(stmt)
        return result.rowcount > 0

Alembic — migracje schematu

Alembic to narzedzie do migracji bazy danych stworzone przez tworcow SQLAlchemy. Generuje pliki migracji na podstawie roznic miedzy modelem ORM a aktualnym schematem bazy.

Komendy Alembic

  • alembic init alembic — inicjalizacja, tworzy alembic.ini i katalog alembic/
  • alembic revision --autogenerate -m "Dodaj tabele uzytkownikow" — generuj migrace
  • alembic upgrade head — zastosuj wszystkie migracje
  • alembic downgrade -1 — cofnij ostatnia migrace
  • alembic history — historia migracji
  • alembic current — aktualna wersja bazy

expire_on_commit=False

Domyslnie SQLAlchemy po commit() inwaliduje wszystkie obiekty w sesji. W aplikacjach async to problem — dostep do atrybutu po commit generuje nowe zapytanie SQL. Ustaw expire_on_commit=False w async_sessionmaker aby tego uniknac.
