Siec i firewall w Linuxie

ip, ss, dig, ufw, nftables, SSH hardening, WireGuard

Konfiguracja sieci - polecenia ip

Pakiet iproute2 zastapil stare narzedzia (ifconfig, route, arp). W nowoczesnym Linuxie uzywa sie ip jako uniwersalnego narzedzia.

# Lista interfejsow z adresami
ip a
ip address show

# Tylko interfejsy bez adresow (warstwa link)
ip link show

# Tablica routingu
ip route
ip r

# Tablica ARP/sasiadow
ip neigh

# Statystyki interfejsu
ip -s link show eth0

# Konkretne IP/interfejs
ip a show dev eth0

# Tymczasowe dodanie/usuniecie adresu
sudo ip addr add 192.168.1.100/24 dev eth0
sudo ip addr del 192.168.1.100/24 dev eth0

# Wlacz/wylacz interfejs
sudo ip link set eth0 up
sudo ip link set eth0 down

# Dodanie trasy statycznej
sudo ip route add 10.0.0.0/24 via 192.168.1.1 dev eth0

# Domyslna brama
sudo ip route add default via 192.168.1.1

Trwala konfiguracja - Netplan (Ubuntu)

Ubuntu 18+ uzywa Netplan jako abstrakcji nad NetworkManager/systemd-networkd. Konfiguracja w plikach YAML w /etc/netplan/.

# /etc/netplan/01-netcfg.yaml
network:
  version: 2
  renderer: networkd
  ethernets:
    eth0:
      dhcp4: true
    eth1:
      addresses:
        - 192.168.1.100/24
      routes:
        - to: default
          via: 192.168.1.1
      nameservers:
        addresses:
          - 1.1.1.1
          - 9.9.9.9
# Walidacja konfiguracji
sudo netplan generate

# Test (cofa zmiany po 120 sekundach)
sudo netplan try

# Zastosowanie zmian
sudo netplan apply

# Tryb debug
sudo netplan --debug apply

ss - nowoczesny netstat

ss (socket statistics) zastapil netstat. Jest szybszy i pokazuje wiecej informacji.

# Wszystkie nasluchujace porty TCP/UDP z PID procesu
sudo ss -tulpn

# Tylko TCP
ss -tn

# Tylko nasluchujace
ss -lt

# Wszystkie polaczenia (ESTABLISHED + LISTEN)
ss -tan

# Tylko polaczenia w stanie ESTABLISHED
ss state established

# Statystyki
ss -s

# Polaczenia do konkretnego portu
ss -tn dst :443
ss -tn src :22

# Filtr po procesie
ss -tlnp '( sport = :80 or sport = :443 )'

DNS - dig, host, nslookup

# Instalacja dig (Ubuntu)
sudo apt install dnsutils

# Podstawowe zapytanie A
dig example.com

# Krotka odpowiedz
dig +short example.com

# Konkretny rekord
dig example.com A
dig example.com AAAA      # IPv6
dig example.com MX        # mail
dig example.com TXT       # SPF, DKIM, DMARC
dig example.com NS        # serwery nazw
dig example.com SOA       # autorytet domeny

# Zapytanie do konkretnego serwera DNS
dig @1.1.1.1 example.com
dig @8.8.8.8 example.com

# Reverse DNS (PTR)
dig -x 1.1.1.1

# Sledzenie odpowiedzi (od root serverow)
dig +trace example.com

# host - prostszy interfejs
host example.com
host -t MX example.com

# nslookup (legacy, ale wciaz uzywany)
nslookup example.com

UFW - prosty firewall (Ubuntu)

UFW (Uncomplicated Firewall) to nakladka na iptables/nftables, domyslna w Ubuntu. Idealny dla pojedynczych serwerow i desktopow.

# Status
sudo ufw status verbose
sudo ufw status numbered

# Wlaczenie/wylaczenie
sudo ufw enable
sudo ufw disable

# Domyslne polityki (zalecane)
sudo ufw default deny incoming
sudo ufw default allow outgoing

# Zezwol na konkretny port
sudo ufw allow 22/tcp
sudo ufw allow 80,443/tcp
sudo ufw allow ssh           # uzywa /etc/services

# Zezwol z konkretnego IP
sudo ufw allow from 192.168.1.100
sudo ufw allow from 192.168.1.0/24 to any port 22

# Rate limiting (chroni przed brute-force)
sudo ufw limit ssh

# Zablokowanie portu
sudo ufw deny 23

# Usuniecie reguly
sudo ufw delete allow 80/tcp
sudo ufw delete 3            # po numerze (z status numbered)

# Reset wszystkich regul
sudo ufw reset

# Logi
sudo ufw logging on
sudo tail -f /var/log/ufw.log

iptables (legacy)

iptables to klasyczne narzedzie konfiguracji firewalla. W Ubuntu 22+ jest stopniowo zastepowane przez nftables, ale wciaz dostepne dla kompatybilnosci.

# Lista wszystkich regul
sudo iptables -L -v -n

# Lista regul w lancuchu INPUT z numerami
sudo iptables -L INPUT --line-numbers

# Domyslna polityka DROP
sudo iptables -P INPUT DROP
sudo iptables -P FORWARD DROP

# Zezwol na ESTABLISHED i RELATED (krytyczne!)
sudo iptables -A INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT

# Zezwol loopback
sudo iptables -A INPUT -i lo -j ACCEPT

# Zezwol SSH
sudo iptables -A INPUT -p tcp --dport 22 -j ACCEPT

# Zezwol HTTP/HTTPS
sudo iptables -A INPUT -p tcp -m multiport --dports 80,443 -j ACCEPT

# Zapis regul (Ubuntu)
sudo apt install iptables-persistent
sudo netfilter-persistent save

nftables - nowoczesny standard

nftables to nastepca iptables - prostsza skladnia, lepsza wydajnosc, jeden binary dla IPv4 i IPv6. W Ubuntu 22+ to domyslny backend.

# Lista regul
sudo nft list ruleset

# Plik konfiguracyjny /etc/nftables.conf
#!/usr/sbin/nft -f
flush ruleset

table inet filter {
    chain input {
        type filter hook input priority 0; policy drop;

        # Zezwol na loopback
        iif lo accept

        # ESTABLISHED i RELATED
        ct state established,related accept

        # ICMP
        ip protocol icmp accept
        ip6 nexthdr icmpv6 accept

        # SSH z rate limiting
        tcp dport 22 ct state new limit rate 5/minute accept

        # HTTP/HTTPS
        tcp dport { 80, 443 } accept

        # Drop wszystko inne (nie loguj zwykle)
        # log prefix "nft drop: " counter drop
        counter drop
    }

    chain forward {
        type filter hook forward priority 0; policy drop;
    }

    chain output {
        type filter hook output priority 0; policy accept;
    }
}
# Zaladowanie konfiguracji
sudo nft -f /etc/nftables.conf

# Wlaczenie autostartu
sudo systemctl enable --now nftables

SSH - hardening

SSH jest pierwszym celem atakow. Konfiguracja w /etc/ssh/sshd_config. Po zmianach: sudo systemctl reload ssh.

# /etc/ssh/sshd_config - przyklad bezpiecznej konfiguracji

# Zmien port (utrudnia automatyczne skanery)
Port 2222

# Wylacz logowanie roota
PermitRootLogin no

# Wylacz logowanie haslem - tylko klucze
PasswordAuthentication no
PubkeyAuthentication yes
ChallengeResponseAuthentication no

# Tylko nowoczesne klucze
HostKeyAlgorithms ssh-ed25519,rsa-sha2-512,rsa-sha2-256
KexAlgorithms curve25519-sha256,curve25519-sha256@libssh.org
Ciphers chacha20-poly1305@openssh.com,aes256-gcm@openssh.com

# Limit prob logowania
MaxAuthTries 3
MaxSessions 10

# Timeout sesji bez aktywnosci (15 min)
ClientAliveInterval 300
ClientAliveCountMax 3

# Tylko konkretni uzytkownicy
AllowUsers jkowalski deploy
AllowGroups ssh-users

# Wylacz X11 forwarding jesli niepotrzebne
X11Forwarding no

# Banner przy logowaniu
Banner /etc/issue.net

Generowanie kluczy SSH

# ed25519 (zalecany - szybki, nowoczesny)
ssh-keygen -t ed25519 -C "kontakt@i4n.pl"

# RSA jesli wymagana kompatybilnosc (min. 4096 bit)
ssh-keygen -t rsa -b 4096 -C "kontakt@i4n.pl"

# Skopiowanie klucza publicznego na serwer
ssh-copy-id -p 2222 jkowalski@serwer.pl

# Test polaczenia
ssh -p 2222 jkowalski@serwer.pl

# Konfiguracja w ~/.ssh/config dla wygody
Host moj-serwer
    HostName serwer.pl
    Port 2222
    User jkowalski
    IdentityFile ~/.ssh/id_ed25519

fail2ban - ochrona przed brute-force

sudo apt install fail2ban

# Konfiguracja w /etc/fail2ban/jail.local (NIE jail.conf)
[DEFAULT]
bantime = 1h
findtime = 10m
maxretry = 5

[sshd]
enabled = true
port = 2222
filter = sshd
logpath = /var/log/auth.log

# Restart i sprawdzenie statusu
sudo systemctl restart fail2ban
sudo fail2ban-client status
sudo fail2ban-client status sshd

# Odbanowanie IP
sudo fail2ban-client set sshd unbanip 1.2.3.4

WireGuard VPN

WireGuard to nowoczesny VPN wbudowany w jadro Linuxa od 5.6. Prostszy, szybszy i bezpieczniejszy niz OpenVPN/IPsec.

# Instalacja
sudo apt install wireguard

# Generowanie kluczy serwera
wg genkey | sudo tee /etc/wireguard/private.key | wg pubkey | sudo tee /etc/wireguard/public.key
sudo chmod 600 /etc/wireguard/private.key

# Konfiguracja serwera /etc/wireguard/wg0.conf
[Interface]
Address = 10.0.0.1/24
ListenPort = 51820
PrivateKey = SERWER_PRIVATE_KEY

PostUp = iptables -A FORWARD -i wg0 -j ACCEPT; iptables -t nat -A POSTROUTING -o eth0 -j MASQUERADE
PostDown = iptables -D FORWARD -i wg0 -j ACCEPT; iptables -t nat -D POSTROUTING -o eth0 -j MASQUERADE

[Peer]
PublicKey = KLIENT_PUBLIC_KEY
AllowedIPs = 10.0.0.2/32

# Wlaczenie forwardowania pakietow
echo 'net.ipv4.ip_forward=1' | sudo tee -a /etc/sysctl.conf
sudo sysctl -p

# Uruchomienie
sudo wg-quick up wg0
sudo systemctl enable wg-quick@wg0

# Status
sudo wg show
Checklist hardeningu serwera:
  • UFW skonfigurowany i wlaczony (default deny incoming)
  • SSH tylko na klucze, alternatywny port, fail2ban
  • Wszystkie publiczne porty maja uzasadnienie biznesowe
  • Aktualizacje bezpieczenstwa automatyczne (unattended-upgrades)
  • Monitoring otwartych portow (cron z ss -tulpn do logu)
  • Regularne audyty: sudo lynis audit system
