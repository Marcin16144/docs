Anatomia bezpiecznego skryptu

Kazdy produkcyjny skrypt bash powinien zaczynac sie od kilku elementow ochronnych. Bez nich blad w jednej linii moze spowodowac katastrofe (np. rm -rf $UNDEFINED/).

Funkcje

# Definicja
log_info() {
    local msg="$1"
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    echo "[INFO] $timestamp - $msg"
}

log_error() {
    local msg="$1"
    echo "[ERROR] $msg" >&2     # stderr
    return 1
}

# Funkcja z wartoscia zwrotna
is_root() {
    [ "$EUID" -eq 0 ]
}

# Uzycie
if is_root; then
    log_info "Skrypt uruchomiony jako root"
else
    log_error "Wymagany root"
    exit 1
fi

# Funkcja zwracajaca wartosc przez echo (jak return wartosci)
get_cpu_count() {
    nproc
}

CPU=$(get_cpu_count)

# Wiele wartosci - przez tablice lub global
parse_url() {
    local url="$1"
    PROTOCOL="${url%%://*}"
    REST="${url#*://}"
    HOST="${REST%%/*}"
    PATH_PART="/${REST#*/}"
}

parse_url "https://example.com/api/users"
echo "Protokol: $PROTOCOL"
echo "Host: $HOST"
echo "Sciezka: $PATH_PART"

# Zamiana w pliku (dru4kuje na stdout)

# Zamiana wszystkich wystapien w linii

# Edycja w miejscu (tworzy backup .bak)

# Wiele zamian

# Zamiana tylko w konkretnych liniach

sed '/^#/s/old/new/g' file.txt   # tylko komentarze

# Usuniecie pustych linii
sed '/^$/d' file.txt

# Usuniecie linii zaczynajacej sie od #

# Wstawienie tekstu przed linia

# Wyswietlenie tylko konkretnych linii

awk - przetwarzanie pol

# Wyswietl drugie pole z kazdej linii
awk '{print $2}' file.txt

# Pola oddzielone : (np. /etc/passwd)
awk -F: '{print $1, $7}' /etc/passwd

# Filtruj linie i wyswietl pola
awk '$3 > 1000 {print $1}' /etc/passwd  # uzytkownicy z UID > 1000

# Sumowanie kolumny
awk '{sum += $1} END {print sum}' numbers.txt

# Suma rozmiarow plikow w katalogu
ls -l | awk '{sum += $5} END {printf "%.2f MB\n", sum/1024/1024}'

# Operacje na ciagach
awk '{print toupper($0)}' file.txt
awk 'NF > 0 {print NR": "$0}' file.txt   # numerowane niepuste

# Wbudowane zmienne: NF (liczba pol), NR (numer linii), $0 (cala linia)

# Bardziej zlozony - statystyki uzycia CPU
ps aux | awk '{cpu += $3} END {printf "Total CPU: %.1f%%\n", cpu}'

find - wyszukiwanie plikow

# Po nazwie
find /etc -name "*.conf"
find . -iname "*.JPG"           # case-insensitive

# Po typie
find / -type f                  # zwykle pliki
find / -type d                  # katalogi
find / -type l                  # symlinki

# Po rozmiarze
find . -size +100M              # ponad 100 MB
find . -size -1k                # ponizej 1 KB

# Po czasie
find /var/log -mtime -1         # zmodyfikowane ostatnia doba
find /tmp -atime +30            # nieuzywane od 30 dni
find /home -mmin -60            # ostatnia godzina

# Po uprawnieniach
find / -perm 4000 -type f       # SUID
find / -perm /u=s               # to samo (symbolicznie)

# Wykonanie komendy na znalezionych
find /tmp -type f -name "*.tmp" -exec rm {} \;

# BEZPIECZNIEJ - z null-separated (radzi sobie ze spacjami)
find /tmp -type f -print0 | xargs -0 rm

# Z exec + (szybsze - jeden proces)
find /tmp -type f -name "*.tmp" -exec rm {} +

# Filtrowanie
find / -type f -name "*.log" -size +10M 2>/dev/null

# Wykluczenie katalogow
find / -path /proc -prune -o -name "*.conf" -print

# Tylko nazwy plikow z dopasowaniem
grep -l "wzorzec" *.txt

# Extended regex

# Kontekst (linie przed/po)
grep -A 3 "ERROR" log.txt        # 3 linie po
grep -B 3 "ERROR" log.txt        # 3 linie przed
grep -C 3 "ERROR" log.txt        # 3 linie kontekstu

# Dokladne dopasowanie slowa

# Tylko dopasowanie (nie cala linia)

Error handling i trap

#!/usr/bin/env bash
set -euo pipefail

# Funkcja wywolywana przy bledzie
error_handler() {
    local line=$1
    local code=$2
    echo "BLAD na linii $line (kod $code)" >&2
    # Tutaj cleanup
    rm -f /tmp/lockfile
    exit "$code"
}

# Funkcja przy normalnym wyjsciu
cleanup() {
    echo "Sprzatanie..."
    rm -f /tmp/tempfile.$$
}

# Trap - obsluga sygnalow i bledow
trap 'error_handler $LINENO $?' ERR
trap cleanup EXIT
trap 'echo "Przerwane"; exit 130' INT TERM

# Glowna logika
main() {
    echo "Start"

    # Probojemy 3 razy
    local retries=0
    until curl -sf https://example.com > /dev/null; do
        retries=$((retries + 1))
        if [ "$retries" -ge 3 ]; then
            echo "Nie udalo sie po 3 probach" >&2
            return 1
        fi
        echo "Proba $retries nieudana, czekam..."
        sleep 5
    done

    echo "Sukces"
}

main "$@"
